Process Manager

Linux process identity, creation, execution, termination, credentials, namespaces, cgroups, and the /proc filesystem from kernel internals outward.

Overview

Every schedulable entity in Linux is a struct task_struct . This article follows the lifetime of that structure from fork() through execve() to do_exit() , covering credentials, namespaces, cgroups, and the /proc/PID/ files that expose each field. After reading it you can map kernel structures to the files you inspect during an incident.

task_struct Descriptor

Every schedulable entity in Linux is represented by a struct task_struct defined in include/linux/sched.h . This structure is the single largest and most important data structure in the kernel and can exceed 6 KB on a modern kernel with all features enabled. It is the anchor for process identity, credentials, memory, namespaces, and cgroup membership.

Process States

The kernel tracks process state in task_struct->__state (formerly state ). The exact state values are defined in include/linux/sched.h :

State Value Meaning
TASK_RUNNING 0x000 On CPU or on runqueue ready to run
TASK_INTERRUPTIBLE 0x001 Sleeping, can be woken by signal
TASK_UNINTERRUPTIBLE 0x002 Sleeping, ignores signals (D state)
TASK_STOPPED 0x004 Stopped by SIGSTOP/ptrace
TASK_TRACED 0x008 Being traced by ptrace
EXIT_DEAD 0x010 Exit almost complete, about to reap
EXIT_ZOMBIE 0x020 Terminated, waiting for parent wait()
TASK_DEAD 0x040 Task_struct about to be freed
TASK_WAKEKILL 0x080 Like TASK_UNINTERRUPTIBLE + fatal sig
TASK_WAKING 0x100 Being woken up (transient)
TASK_NOLOAD 0x200 Don't count toward load average
TASK_NEW 0x400 Freshly forked, not yet running

Process Creation

fork() (and vfork() , clone() ) all converge on kernel_clone() which calls copy_process() . This function copies (or shares) every subsystem of the parent process: 

// kernel/fork.c — simplified copy_process() flow
struct task_struct *copy_process(
    struct pid *pid,
    int trace,
    int node,
    struct kernel_clone_args *args)
{
    struct task_struct *p;

    // 1. Allocate and zero task_struct
    p = dup_task_struct(current, node);
    if (!p)
        return ERR_PTR(-ENOMEM);

    // 2. Resource accounting (RLIMIT_NPROC, cgroup checks)
    retval = copy_creds(p, args);               // Copy or share credentials
    retval = copy_semundo(args, p);             // SysV semaphore undo
    retval = copy_files(args, p);               // Copy or share file descriptor table
    retval = copy_fs(args, p);                  // Copy or share fs_struct (root, pwd)
    retval = copy_sighand(args, p);             // Copy or share signal handlers
    retval = copy_signal(args, p);              // Copy or share signal_struct
    retval = copy_mm(args, p);                  // Copy or share mm_struct (virtual memory)
    retval = copy_namespaces(args, p);          // Copy or share namespace proxy
    retval = copy_io(args, p);                  // Copy or share I/O context
    retval = copy_thread(p, args);              // Arch-specific: set up thread, stack

    // 3. Assign PID
    pid = alloc_pid(p->nsproxy->pid_ns_for_children);
    p->pid = pid_nr(pid);                       // TID in init namespace

    // 4. Integrate into data structures
    hlist_add_head_rcu(&p->pid_links[PIDTYPE_PID], ...);
    list_add_tail_rcu(&p->thread_group, ...);

    // 5. Set state
    p->__state = TASK_NEW;
    return p;
}

Image Loading with execve()

When a process calls execve() , the kernel replaces the entire virtual address space with a new program image. The old address space is discarded; only the task identity (PID, credentials, namespaces) persists. 

// fs/exec.c → arch/x86/kernel/process_64.c flow:
SYSCALL_DEFINE3(execve, const char __user *, filename,
                const char __user *const __user *, argv,
                const char __user *const __user *, envp)
{
    return do_execve(getname(filename), argv, envp);
}

Process Termination

// kernel/exit.c — simplified
void __noreturn do_exit(long code)
{
    struct task_struct *tsk = current;

    profile_task_exit(tsk);

    // 1. Set the exit code
    tsk->exit_code = (code & 0xff) << 8;

    // 2. Release resources
    exit_signals(tsk);          // Set all pending signals to ignored
    exit_mm(tsk);               // Release mm_struct (mmput)
    exit_sem(tsk);              // Undo SysV semaphores
    exit_shm(tsk);              // Detach SysV shared memory
    exit_files(tsk);            // Close file descriptors, put files_struct
    exit_fs(tsk);               // Release fs_struct
    exit_task_namespaces(tsk);  // Put namespace references
    exit_io_context(tsk);       // Release I/O context
    exit_thread(tsk);           // Arch-specific thread cleanup

    // 3. Notify parent
    exit_notify(tsk, group_dead);
    //    │
    //    ├─ If group_dead (last thread in process):
    //    │    signal parent with SIGCHLD (or set parent-death signal)
    //    │    reparent orphaned children to init (PID 1) or subreaper
    //    │
    //    └─ Set state to EXIT_ZOMBIE

    // 4. Final schedule — never returns
    tsk->__state = EXIT_ZOMBIE;
    do_task_dead();
}

Zombie Reaping

  • The parent must call wait() ( wait4 , waitpid , waitid ) to read the exit status.
  • Until the parent waits, the zombie holds a task_struct and PID entry.
  • If the parent exits before reaping, init (PID 1) inherits the orphan zombies and wait() s automatically.
  • A process can use prctl(PR_SET_CHILD_SUBREAPER, 1) to become a subreaper.
# Find zombie processes
ps aux | grep 'Z'

# Zombie's exit code
cat /proc/<zombie_pid>/status | grep -E 'State|SigCgt|SigPnd'

# Check init's children
sudo ps -o pid,ppid,state,comm --ppid 1

# Subreaper identification
sudo grep subreaper /proc/*/status 2>/dev/null

Daemonizing with setsid()

// Classic daemonization sequence
void daemonize(void)
{
    pid_t pid = fork();
    if (pid > 0) exit(0);  // Parent exits

    setsid();              // Create new session, detach from terminal
    // Now: SID = PID, no controlling terminal

    pid = fork();
    if (pid > 0) exit(0);  // Session leader exits (can't reacquire CTTY)

    chdir("/");            // Don't keep mount points busy
    umask(0);               // Inherited umask can break file creation
    close(0); close(1); close(2);  // Or redirect to /dev/null
}

Process Credentials

The struct cred encapsulates a task's security context. It is reference-counted and shared among threads that don't need distinct credentials. 

// include/linux/cred.h
struct cred {
    atomic_long_t   usage;
    kuid_t          uid;            // Real UID
    kgid_t          gid;            // Real GID
    kuid_t          suid;           // Saved UID
    kgid_t          sgid;           // Saved GID
    kuid_t          euid;           // Effective UID
    kgid_t          egid;           // Effective GID
    kuid_t          fsuid;          // FS UID (for file access checks)
    kgid_t          fsgid;          // FS GID
    unsigned        securebits;
    kernel_cap_t    cap_inheritable;// Inheritable capabilities
    kernel_cap_t    cap_permitted;  // Permitted capabilities
    kernel_cap_t    cap_effective;  // Effective capabilities
    kernel_cap_t    cap_bset;       // Bounding set
    kernel_cap_t    cap_ambient;    // Ambient set
    ...
    struct user_struct *user;       // Per-user accounting
    struct user_namespace *user_ns; // User namespace
    ...
};

// Task holds two cred pointers:
//  task->cred     = subjective (used for access checks)
//  task->real_cred= objective  (stores the actual identity)

PID Namespaces and Translation

A process has different PIDs in different namespaces. The kernel maintains a struct pid with multiple struct upid entries: 

// include/linux/pid.h
struct pid {
    refcount_t          count;
    unsigned int        level;          // Depth in namespace hierarchy
    spinlock_t          lock;
    struct hlist_head   tasks[PIDTYPE_MAX];  // Hash: type → task
    struct upid         numbers[];           // Flexible array: one per ns layer
};

struct upid {
    int                 nr;             // PID number in this namespace
    struct pid_namespace *ns;           // Which namespace
};

Challenge

Write a Go tool that builds the process tree from /proc . For each numeric directory, read /proc/<pid>/stat for the PID, PPID, process group, session, and state, and /proc/<pid>/status for Tgid and NSpid . Print each process indented under its parent and mark entries whose Tgid differs from Pid as threads. 

package main

import (
	"bufio"
	"fmt"
	"os"
	"sort"
	"strconv"
	"strings"
)

type proc struct {
	pid, ppid, pgid, sid int
	tgid, nspid          int
	state, comm          string
}

func parseProc(pid int) proc {
	p := proc{pid: pid}
	raw, err := os.ReadFile(fmt.Sprintf("/proc/%d/stat", pid))
	if err != nil {
		return proc{}
	}
	s := string(raw)
	open := strings.Index(s, "(")
	end := strings.LastIndex(s, ")")
	if open < 0 || end < 0 {
		return proc{}
	}
	p.comm = s[open+1 : end]
	fields := strings.Fields(s[end+1:])
	p.state = fields[0]
	p.ppid, _ = strconv.Atoi(fields[1])
	p.pgid, _ = strconv.Atoi(fields[2])
	p.sid, _ = strconv.Atoi(fields[3])

	f, err := os.Open(fmt.Sprintf("/proc/%d/status", pid))
	if err != nil {
		return p
	}
	defer f.Close()
	sc := bufio.NewScanner(f)
	for sc.Scan() {
		line := sc.Text()
		switch {
		case strings.HasPrefix(line, "Tgid:"):
			p.tgid, _ = strconv.Atoi(strings.TrimSpace(strings.TrimPrefix(line, "Tgid:")))
		case strings.HasPrefix(line, "NSpid:"):
			ns := strings.Fields(strings.TrimPrefix(line, "NSpid:"))
			p.nspid, _ = strconv.Atoi(ns[len(ns)-1])
		}
	}
	return p
}

func procList() []proc {
	entries, _ := os.ReadDir("/proc")
	out := make([]proc, 0)
	for _, e := range entries {
		pid, err := strconv.Atoi(e.Name())
		if err != nil {
			continue
		}
		if p := parseProc(pid); p.pid != 0 {
			out = append(out, p)
		}
	}
	sort.Slice(out, func(i, j int) bool { return out[i].pid < out[j].pid })
	return out
}

func main() {
	byParent := make(map[int][]proc)
	for _, p := range procList() {
		byParent[p.ppid] = append(byParent[p.ppid], p)
	}

	var walk func(pid, depth int)
	walk = func(pid, depth int) {
		for _, p := range byParent[pid] {
			kind := "process"
			if p.tgid != p.pid {
				kind = "thread"
			}
			fmt.Printf("%s%-6d %-7s ppid=%-6d pgid=%-6d sid=%-6d ns=%-6d %s (%s)\n",
				strings.Repeat("  ", depth), p.pid, p.state, p.ppid,
				p.pgid, p.sid, p.nspid, p.comm, kind)
			walk(p.pid, depth+1)
		}
	}
	walk(1, 0)
}

Build and run it, then compare with ps : 

go build -o proctree proctree.go
./proctree | head -30
ps -eo pid,ppid,pgid,sid,state,comm --forest | head -30

Start a process that spawns threads, then confirm that its extra entries share one Tgid : 

stress-ng --cpu 2 --timeout 5s &
sleep 1
./proctree | grep -A 4 stress-ng
