Persistence Analysis

Identifying and analyzing persistence mechanisms in Linux malware: crontab, systemd, shell profiles, SSH keys, and LD_PRELOAD.

Overview

Persistence is how malware survives reboots, user logouts, and process kills. A dropper that infects a machine, runs once, and disappears after reboot is a failed infection, so malware invests code in creating a trigger the OS honors automatically. This article maps the persistence vectors on Linux, shows how strace and an LD_PRELOAD spy library expose them, walks through crontab, systemd, SSH key, PAM , and kernel module mechanisms, and ends with a Go tool that audits a system for the file writes these mechanisms leave behind.

Required Reading

  • Tracing - tracing binary execution with strace, ltrace, ftrace
  • LD_PRELOAD - intercepting function calls with LD_PRELOAD

The Persistence Problem

A Linux process dies when the user logs out (SIGHUP to the session), when the system reboots, when an admin runs kill , or when a competing infection kills it. To survive, the malware must create a trigger that the OS itself honors: something that runs automatically on boot, login, or on a schedule.

Every persistence mechanism reduces to writing a file to a well-known location that the OS reads at startup, then waiting for the OS to execute its contents.

Tracing Persistence Setup with strace

When analyzing an unknown binary, filter strace for writes to known persistence paths:

# Broad persistence monitor — watch all file writes to suspicious paths
strace -e trace=open,openat,write,creat,rename,mkdir -f \
  -o trace.log ./malware_sample

# Post-analysis: filter for persistence-relevant paths
grep -E '/(etc|home/.*/\.(bashrc|profile|zshrc|ssh|config/autostart|config/systemd))' trace.log

# Specific filter patterns:
PATHS=(
  '/etc/crontab'
  '/etc/cron.d/'
  '/etc/cron.hourly/'
  '/etc/cron.daily/'
  '/etc/cron.weekly/'
  '/etc/cron.monthly/'
  '/etc/systemd/system/'
  '/etc/init.d/'
  '/etc/rc.local'
  '/etc/profile'
  '/etc/bash.bashrc'
  '/etc/ld.so.preload'
  '/etc/modules-load.d/'
  '/etc/pam.d/'
  '/etc/xdg/autostart/'
  '/etc/update-motd.d/'
  '/etc/udev/rules.d/'
  'bashrc'
  '.profile'
  '.zshrc'
  '.ssh/authorized_keys'
  '.config/autostart/'
  '.config/systemd/user/'
  '/var/spool/cron/crontabs/'
)

for p in "${PATHS[@]}"; do
  grep "$p" trace.log && echo "^^^ FOUND PERSISTENCE: $p"
done

Persistence Detection with LD_PRELOAD

A persistent spy library that only logs writes to suspicious locations:

/*
 * persist_spy.c — LD_PRELOAD library for persistence detection
 * Compile: gcc -shared -fPIC -fno-builtin -o persist_spy.so persist_spy.c -ldl
 * Usage:   LD_PRELOAD=./persist_spy.so ./malware_sample
 */

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdarg.h>
#include <sys/stat.h>
#include <sys/types.h>

static const char *PERSIST_PATHS[] = {
    "/etc/crontab",
    "/etc/cron.d/",
    "/etc/cron.hourly/",
    "/etc/cron.daily/",
    "/etc/cron.weekly/",
    "/etc/cron.monthly/",
    "/etc/systemd/system/",
    "/etc/init.d/",
    "/etc/rc.local",
    "/etc/rc.d/",
    "/etc/profile",
    "/etc/bash.bashrc",
    "/etc/ld.so.preload",
    "/etc/modules-load.d/",
    "/etc/pam.d/",
    "/etc/xdg/autostart/",
    "/etc/update-motd.d/",
    "/etc/udev/rules.d/",
    ".bashrc",
    ".profile",
    ".zshrc",
    ".ssh/authorized_keys",
    ".config/autostart/",
    ".config/systemd/user/",
    "/var/spool/cron/crontabs/",
    NULL
};

static int is_persist_path(const char *path)
{
    if (!path) return 0;
    for (int i = 0; PERSIST_PATHS[i]; i++) {
        if (strstr(path, PERSIST_PATHS[i]))
            return 1;
    }
    return 0;
}

/* Hook: open / open64 */
typedef int (*orig_open_t)(const char *, int, ...);

int open(const char *pathname, int flags, ...)
{
    orig_open_t f = (orig_open_t)dlsym(RTLD_NEXT, "open");
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list ap;
        va_start(ap, flags);
        mode = va_arg(ap, mode_t);
        va_end(ap);
    }

    if (is_persist_path(pathname)) {
        const char *op = (flags & O_WRONLY || flags & O_RDWR) ? "WRITE" : "READ";
        fprintf(stderr, "[!] PERSISTENCE DETECTED: open(%s %s)\n", op, pathname);

        /* Print backtrace — who opened this file? */
        void *bt[32];
        int n = backtrace(bt, 32);
        char **syms = backtrace_symbols(bt, n);
        fprintf(stderr, "[!] Backtrace:\n");
        for (int i = 1; i < n && i < 8; i++)  /* skip first frame (self) */
            fprintf(stderr, "    #%d  %s\n", i, syms[i]);
        free(syms);
    }

    return f ? f(pathname, flags, mode) : -1;
}

/* Hook: write — detect writes to already-open persistence FDs */
typedef ssize_t (*orig_write_t)(int, const void *, size_t);

ssize_t write(int fd, const void *buf, size_t count)
{
    orig_write_t f = (orig_write_t)dlsym(RTLD_NEXT, "write");

    /* Check if this FD points to a persistence path */
    char fdpath[256];
    snprintf(fdpath, sizeof(fdpath), "/proc/self/fd/%d", fd);
    char link[1024];
    ssize_t n = readlink(fdpath, link, sizeof(link) - 1);
    if (n > 0) {
        link[n] = '\0';
        if (is_persist_path(link)) {
            fprintf(stderr, "[!] PERSISTENCE WRITE: fd=%d → %s (%zu bytes)\n",
                    fd, link, count);
            /* Print first 128 bytes of what was written */
            if (count > 0) {
                fprintf(stderr, "[!] Data: ");
                for (size_t i = 0; i < count && i < 128; i++)
                    fprintf(stderr, "%c", ((char *)buf)[i] >= 0x20 &&
                            ((char *)buf)[i] < 0x7f ? ((char *)buf)[i] : '.');
                fprintf(stderr, "\n");
            }
        }
    }

    return f ? f(fd, buf, count) : -1;
}

/* Hook: execve — detect installation commands */
typedef int (*orig_execve_t)(const char *, char *const[], char *const[]);

int execve(const char *pathname, char *const argv[], char *const envp[])
{
    orig_execve_t f = (orig_execve_t)dlsym(RTLD_NEXT, "execve");

    /* Detect persistence-related command execution */
    const char *persist_cmds[] = {
        "crontab", "systemctl", "useradd", "usermod", "passwd",
        "chkconfig", "update-rc.d", "rc-update", "apt", "pacman",
        "yum", "dnf", "ssh-keygen", "ssh", "scp", "mkdir",
        "chmod", "chown", NULL
    };

    /* Extract just the command name from pathname */
    const char *basename_cmd = strrchr(pathname, '/');
    basename_cmd = basename_cmd ? basename_cmd + 1 : pathname;

    for (int i = 0; persist_cmds[i]; i++) {
        if (strcmp(basename_cmd, persist_cmds[i]) == 0) {
            fprintf(stderr, "[!] PERSISTENCE CMD: %s", pathname);
            if (argv) {
                for (int j = 0; argv[j]; j++)
                    fprintf(stderr, " \"%s\"", argv[j]);
            }
            fprintf(stderr, "\n");
            break;
        }
    }

    return f ? f(pathname, argv, envp) : -1;
}

Compile and link with backtrace support:

gcc -shared -fPIC -fno-builtin -o persist_spy.so persist_spy.c -ldl -rdynamic

Persistence Mechanism Analysis

1. Shell Configuration Injection

Most common in user-level malware: append a command to ~/.bashrc or ~/.profile so it executes on every shell login.

Detection in disassembly

// Ghidra decompilation of a shell-config injection routine:
void install_persistence(void) {
    char *home = getenv("HOME");
    char bashrc[256];
    snprintf(bashrc, sizeof(bashrc), "%s/.bashrc", home);
    int fd = open(bashrc, O_WRONLY | O_APPEND);
    if (fd > 0) {
        write(fd, "\n# system update check\n/path/to/malware &\n", 43);
        close(fd);
    }
}

What strace reveals

openat(AT_FDCWD, "/root/.bashrc", O_WRONLY|O_APPEND) = 3
write(3, "\n# system update check\n/path/to/malware &\n", 43) = 43
close(3)

Manual inspection

# Check all user profiles:
grep -rln 'malware_multiple_backtick\|# system\|# update' /home/*/.bashrc /home/*/.profile /home/*/.zshrc 2>/dev/null
grep -rln 'malware_multiple_backtick\|# system\|# update' /root/.bashrc /root/.profile 2>/dev/null

# The malware often disguises itself with benign-sounding comments:
# "# system service check"
# "# kernel module loader"
# "# license manager daemon"

2. Cron-Based Persistence

User crontab

# Malware executes: crontab -l | { cat; echo "*/5 * * * * /tmp/.hidden/malware"; } | crontab -
# Appends a cron job that runs every 5 minutes

# strace shows:
execve("/usr/bin/crontab", ["crontab", "-"], [...])  # reading/installing crontab
write(1, "*/5 * * * * /tmp/.hidden/malware\n", 33)

System crontab (root only)

# Malware (as root) writes directly to /etc/crontab:
echo "*/10 * * * * root /usr/lib/.hidden/update-daemon" >> /etc/crontab

# strace shows:
openat(AT_FDCWD, "/etc/crontab", O_WRONLY|O_APPEND) = 4
write(4, "*/10 * * * * root /usr/lib/.hidden/update-daemon\n", 50) = 50

cron.d drop-in (root only)

# More sophisticated malware creates a dedicated cron.d file:
echo '*/5 * * * * root /lib/.sys/daemon >/dev/null 2>&1' > /etc/cron.d/system-update

# strace shows:
openat(AT_FDCWD, "/etc/cron.d/system-update", O_WRONLY|O_CREAT|O_TRUNC, 0644) = 4
write(4, "*/5 * * * * root /lib/.sys/daemon >/dev/null 2>&1\n", 51) = 51

How cron files are structured

# /etc/crontab format:
# m h dom mon dow user command
*/15 * *   *   *   root /usr/lib/system-update >/dev/null 2>&1

# /etc/cron.d/* format (same, but filename is arbitrary):
# m h dom mon dow user command
0   3 *   *   *   root /usr/lib/cleanup.sh

Analyzing suspicious crontabs

# Dump all cron entries system-wide:
for u in $(cut -d: -f1 /etc/passwd); do echo "=== $u ==="; crontab -u "$u" -l 2>/dev/null; done

# Check all cron-spooled files:
for f in /var/spool/cron/crontabs/*; do echo "=== $f ==="; cat "$f"; done

# Check /etc/cron.* directories:
for d in /etc/cron.d /etc/cron.hourly /etc/cron.daily /etc/cron.weekly /etc/cron.monthly; do
    echo "=== $d ==="
    ls -la "$d" 2>/dev/null
done

# Common suspicious entries:
# - Commands running from /tmp, /dev/shm, /var/tmp
# - Periods shorter than 1 hour
# - Files in hidden directories (prefixed with .)

3. Systemd Service Persistence

Systemd is the default init on ArchLinux and most modern distributions. Malware creates a .service file that systemd starts at boot.

Minimal malicious service file

[Unit]
Description=System Update Service
After=network.target

[Service]
Type=simple
ExecStart=/usr/lib/.hidden/system-update
Restart=always
RestartSec=10

[Install]
WantedBy=multi-user.target

Malware's installation sequence

# 1. Write the service file
cat > /etc/systemd/system/system-update.service << 'EOF'
[Unit]
Description=System Update Service
After=network.target

[Service]
Type=simple
ExecStart=/usr/lib/.hidden/system-update
Restart=always
RestartSec=10

[Install]
WantedBy=multi-user.target
EOF

# 2. Drop the payload binary
cp /tmp/payload /usr/lib/.hidden/system-update
chmod +x /usr/lib/.hidden/system-update

# 3. Enable and start
systemctl daemon-reload
systemctl enable system-update.service
systemctl start system-update.service

strace reveals the whole chain

# File creates:
openat(AT_FDCWD, "/etc/systemd/system/system-update.service", O_WRONLY|O_CREAT, 0644)
write(4, "[Unit]\nDescription=System Update Service...", ...)

openat(AT_FDCWD, "/usr/lib/.hidden/system-update", O_WRONLY|O_CREAT, 0755)
write(5, <ELF binary contents>, ...)

# Command execution:
execve("/usr/bin/systemctl", ["systemctl", "daemon-reload"], [...])
execve("/usr/bin/systemctl", ["systemctl", "enable", "system-update.service"], [...])
execve("/usr/bin/systemctl", ["systemctl", "start", "system-update.service"], [...])

Analyzing suspicious systemd services

# List all enabled services and look for unknown ones:
systemctl list-unit-files --state=enabled | grep -vE '^(systemd|dbus|NetworkManager|sshd|cron|docker|podman)'
# Any unknown service is suspect

# Check the ExecStart path:
systemctl cat suspicious-service.service
# Read the service file directly:
cat /etc/systemd/system/suspicious-service.service

# Look for services in user scope (malware running as non-root):
ls -la ~/.config/systemd/user/
# User-level systemd services start at user login, not system boot.

# Check for timers (cron-like scheduling):
systemctl list-timers --all
# Timers can trigger services periodically — check any unknown timer's linked service
systemctl cat suspicious.timer

User-Level systemd Units

mkdir -p ~/.config/systemd/user/
cat > ~/.config/systemd/user/persist-daemon.service << 'EOF'
[Unit]
Description=User Daemon

[Service]
ExecStart=/home/user/.local/bin/malware
Restart=always

[Install]
WantedBy=default.target
EOF

systemctl --user daemon-reload
systemctl --user enable persist-daemon.service
systemctl --user start persist-daemon.service

4. SSH Authorized Keys

A backdoor that adds the attacker's public key to ~/.ssh/authorized_keys : 

# Detection in strace:
openat(AT_FDCWD, "/root/.ssh/authorized_keys", O_WRONLY|O_CREAT|O_APPEND, 0600)
write(4, "ssh-rsa AAAA.... attacker@c2-server\n", ...)
chmod("/root/.ssh/authorized_keys", 0600)
chmod("/root/.ssh", 0700)

# If user has no .ssh directory:
mkdir("/root/.ssh", 0700)

Analysis

# Check for unauthorized SSH keys:
for h in /root /home/*; do
    keyfile="$h/.ssh/authorized_keys"
    if [ -f "$keyfile" ]; then
        echo "=== $keyfile ==="
        cat "$keyfile"
    fi
done

# Look for comments in keys referencing unknown hosts
# Also check: the .ssh directory's creation timestamp
stat ~/.ssh/authorized_keys  # mtime should match when the key was installed

5. LD_PRELOAD Persistence (System-Wide)

Writing to /etc/ld.so.preload injects the specified library into every subsequently started process. This is a rootkit-level persistence mechanism. 

# Detection:
cat /etc/ld.so.preload
# /usr/local/lib/libsystem.so   ← SUSPICIOUS: no standard library has this path/name

# Any entry in ld.so.preload that does not come from a known package is malicious:
pacman -Qo /usr/local/lib/libsystem.so                         # Arch: check ownership
dpkg -S /usr/local/lib/libsystem.so                            # Debian
rpm -qf /usr/local/lib/libsystem.so                            # RHEL/Fedora

Malware implementation

void install_ld_preload_persistence(const char *lib_path) {
    int fd = open("/etc/ld.so.preload", O_WRONLY | O_CREAT | O_APPEND, 0644);
    if (fd < 0) return;
    write(fd, lib_path, strlen(lib_path));
    write(fd, "\n", 1);
    close(fd);
    // Now every process maps this library
}

Effect

If the library hooks fopen , stat , opendir , etc., it can hide files, processes, and network connections from every tool on the system. ls uses opendir -> hook filters directory entries. ps reads /proc -> hook filters PID entries. ss reads /proc/net/tcp -> hook filters connection lines.

6. PAM Module Persistence

PAM persistence replaces or extends authentication logic: 

# Malware adds a line to /etc/pam.d/sshd:
# auth required pam_unix.so
# auth sufficient pam_malware.so    ← added by malware

# The pam_malware.so module:
# - Accepts any password that matches a backdoor password
# - Logs all passwords to a file
# - Opens a reverse shell on "special" authentication

# Detection:
find /lib/security /usr/lib/security -name 'pam_*.so' -type f | while read f; do
    pacman -Qo "$f" 2>/dev/null || rpm -qf "$f" 2>/dev/null || echo "ORPHAN: $f"
done

7. Desktop Autostart

GUI -targeted malware persists via the freedesktop.org autostart specification: 

# ~/.config/autostart/system-helper.desktop
[Desktop Entry]
Type=Application
Name=System Helper
Exec=/home/user/.local/share/system-helper
Hidden=false
NoDisplay=true
X-GNOME-Autostart-enabled=true
						
# Detection:
ls -la ~/.config/autostart/
# Look for entries without corresponding packages:
pacman -Qo ~/.config/autostart/*.desktop

# System-wide autostart:
ls -la /etc/xdg/autostart/

8. Kernel Module Autoload

Root-level malware can install a kernel module that loads on every boot:

# Malware writes:
echo "malicious_module" > /etc/modules-load.d/malware.conf

# And places the .ko file:
cp malicious_module.ko /lib/modules/$(uname -r)/kernel/drivers/net/malicious.ko
depmod -a

# Detection:
cat /etc/modules-load.d/*.conf
# Any module not from an official package:
for mod in $(cat /etc/modules-load.d/*.conf 2>/dev/null); do
    modinfo "$mod" 2>/dev/null | grep -E 'filename|author|description'
done

Persistence Detection Script

A comprehensive system audit script:

#!/bin/bash
# persist_audit.sh — audit a system for persistence mechanisms

echo "=== Shell Profiles ==="
for f in /root/.bashrc /root/.profile /home/*/.bashrc /home/*/.profile; do
    [ -f "$f" ] && echo "$f: $(wc -l < "$f") lines"
done

echo "=== /etc/ld.so.preload ==="
cat /etc/ld.so.preload 2>/dev/null || echo "(none)"

echo "=== crontabs ==="
for u in $(cut -d: -f1 /etc/passwd); do
    crontab -u "$u" -l 2>/dev/null | grep -v '^#' | grep -v '^$' | while read line; do
        echo "  USER=$u: $line"
    done
done

for f in /etc/cron.d/* /etc/cron.hourly/* /etc/cron.daily/* /etc/cron.weekly/* /etc/cron.monthly/*; do
    [ -f "$f" ] && echo "  SYSTEM: $f"
done

echo "=== systemd services ==="
systemctl list-unit-files --state=enabled 2>/dev/null | grep -vE '^(systemd|dbus|NetworkManager|sshd|cron|getty|polkit|accounts-daemon|udisks2|upower|wpa_supplicant|ModemManager|bluetooth|avahi-daemon|cups|docker|podman)'
for u in /home/*/.config/systemd/user/*.service; do
    echo "  USER: $u"
done

echo "=== SSH keys ==="
for h in /root /home/*; do
    [ -f "$h/.ssh/authorized_keys" ] && echo "  $h/.ssh/authorized_keys: $(head -1 "$h/.ssh/authorized_keys")"
done

echo "=== Desktop autostart ==="
for d in /root/.config/autostart /home/*/.config/autostart /etc/xdg/autostart; do
    [ -d "$d" ] && ls -la "$d/"
done

echo "=== Kernel modules ==="
cat /etc/modules-load.d/*.conf 2>/dev/null || echo "(none)"

echo "=== MOTD ==="
ls -la /etc/update-motd.d/ 2>/dev/null

echo "=== PAM oddities ==="
grep -r 'auth.*sufficient' /etc/pam.d/ 2>/dev/null | grep -v 'pam_unix\|pam_ldap\|pam_sss\|pam_fprintd'

Dropper Analysis Pattern

A dropper is malware whose purpose is installing other malware. The pattern in strace is characteristic:

1. write("/tmp/.hidden/payload", <ELF binary bytes>)    # Drop payload file
2. chmod("/tmp/.hidden/payload", 0755)                    # Make executable
3. write("/etc/systemd/system/bad.service", <service file>) # Set up persistence
4. execve("/usr/bin/systemctl", ["systemctl", "enable", "bad.service"])  # Enable
5. execve("/usr/bin/systemctl", ["systemctl", "start", "bad.service"])   # Start
6. unlink("/tmp/dropper")                                 # Clean up self

The dropper itself is ephemeral - it runs once, drops the real payload, installs persistence, and deletes itself. The persistence mechanism is what keeps the payload alive after the dropper exits.

Persistence Across Container Escapes

Malware targeting containerized environments may attempt host-level persistence:

# Malware in container writes to a mounted host directory:
echo "0 */6 * * * root /usr/lib/systemd/systemd-update" >> /host/etc/crontab

# Or via container socket:
podman run --restart=always -d --name system-update malicious-image

# Detection in the container:
# Look for writes to paths that are likely host mounts:
strace -e openat,write -f ./container_malware 2>&1 | grep -E '/(host|rootfs|mnt)'

Post-Infection Cleanup

When persistence is confirmed, cleanup follows the reverse chain:

# 1. Kill the active process
kill -9 $(pgrep -f '/usr/lib/.hidden/system-update')

# 2. Disable and remove the systemd service
systemctl stop system-update.service
systemctl disable system-update.service
rm /etc/systemd/system/system-update.service
systemctl daemon-reload

# 3. Remove cron entries
crontab -u root -l | grep -v '/usr/lib/.hidden/system-update' | crontab -u root -

# 4. Clean shell profiles
sed -i '/system-update/d' /root/.bashrc /etc/profile /etc/bash.bashrc

# 5. Remove the payload
rm -rf /usr/lib/.hidden/

# 6. Check LD_PRELOAD
: > /etc/ld.so.preload  # clear if malicious

# 7. Verify cleanup
bash /path/to/persist_audit.sh

Summary

Persistence is malware's anchor in the system. Every persistence mechanism writes a file to a well-known location that the OS reads at startup. These writes are visible in strace logs, interceptable with LD_PRELOAD hooks, and identifiable in disassembly. Shell profiles, crontabs, systemd services, SSH keys, and LD_PRELOAD preloads form the complete vector surface. When analyzing unknown malware, filter strace output for writes to persistence paths first - this reveals the infection chain immediately. The cleanup procedure reverses the chain: kill the process, disable the trigger, remove the trigger file, delete the payload binary.

Challenge

Write a Go tool that audits a Linux filesystem for persistence artifacts. Walk the paths from the vectors table, skip missing entries, and print each existing artifact with its modification time and size. Treat /etc/ld.so.preload , ~/.ssh/authorized_keys , and the autostart and systemd directories as suspicious, and exit with a non-zero status when at least one is found. 

package main

import (
	"fmt"
	"os"
	"path/filepath"
)

type candidate struct {
	path       string
	suspicious bool
}

func main() {
	home, _ := os.UserHomeDir()

	targets := []candidate{
		{"/etc/ld.so.preload", true},
		{"/etc/crontab", true},
		{"/etc/rc.local", true},
		{"/etc/profile", false},
		{"/etc/bash.bashrc", false},
		{filepath.Join(home, ".bashrc"), false},
		{filepath.Join(home, ".profile"), false},
		{filepath.Join(home, ".ssh/authorized_keys"), true},
		{filepath.Join(home, ".config/autostart"), true},
		{filepath.Join(home, ".config/systemd/user"), true},
		{"/etc/cron.d", true},
		{"/etc/systemd/system", true},
		{"/etc/pam.d", false},
		{"/etc/modules-load.d", true},
		{"/etc/udev/rules.d", false},
		{"/etc/xdg/autostart", true},
		{"/etc/update-motd.d", false},
	}

	found := 0

	for _, target := range targets {
		info, err := os.Stat(target.path)
		if err != nil {
			continue
		}

		label := "present"
		if target.suspicious {
			label = "SUSPICIOUS"
			found++
		}

		mode := "file"
		if info.IsDir() {
			mode = "dir"
		}
		fmt.Printf("%-10s %-4s %s (%s, %d bytes)\n",
			label, mode, target.path, info.ModTime().UTC(), info.Size())
	}

	if found > 0 {
		fmt.Printf("\n%d suspicious artifact(s) found\n", found)
		os.Exit(2)
	}

	fmt.Println("no suspicious persistence artifacts found")
}

go run persist_audit.go
echo "exit: $?"
