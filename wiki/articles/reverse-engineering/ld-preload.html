LD_PRELOAD Hooking

Intercepting function calls at the dynamic linker level to trace, modify, and hijack library calls in Linux binaries.

Overview

LD_PRELOAD exploits the dynamic linker's symbol resolution order: when resolving a function name, ld.so searches the libraries named in LD_PRELOAD before libc and other system libraries, so a hook that exports the same symbol shadows the real implementation. Any dynamically linked binary - malware, closed-source tools, a CI runner - calls your code instead. This requires no ptrace, no breakpoints, and no kernel modules, only a .so file and an environment variable. After reading, you can hook file, network, and execution calls, chain to the original functions with dlsym , and recognize when malware uses the same trick for persistence.

Required Reading

  • Dynamic Linking - the dynamic linker, symbol resolution order, PLT/GOT mechanics
  • GDB for Malware Analysis - debugging malware dynamically

Symbol Resolution Order

When a binary calls open("/etc/shadow", O_RDONLY) , the compiler emits a call open@plt instruction. On first call, the PLT stub invokes _dl_runtime_resolve which walks the symbol tables of all loaded shared objects in order:

  1. LD_PRELOAD libraries (searched first, left to right)
  2. DT_NEEDED libraries from the binary itself (linked at compile time)
  3. The dynamic linker itself ( ld-linux.so )
  4. libc.so and other system libraries

The first matching symbol wins. If libpreload.so exports open , that definition is used. No relocation, no GOT overwrite - the resolver simply returns your function's address.

This is not a security bypass; it's a deliberate debugging/extension mechanism of the ELF dynamic linking model.

The Minimal Hook Library

A hook library exports functions with the same symbol name as the target. To chain to the original, use dlsym(RTLD_NEXT, "function_name") : 

/*
 * hook.c - minimal LD_PRELOAD hook for open(2)
 *
 * Compile:
 *   gcc -shared -fPIC -o hook.so hook.c -ldl
 * Usage:
 *   LD_PRELOAD=./hook.so /path/to/binary
 */

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <fcntl.h>
#include <stdarg.h>
#include <unistd.h>
#include <string.h>
#include <sys/types.h>

/* Type definition matching the real open() signature */
typedef int (*orig_open_t)(const char *pathname, int flags, ...);

int open(const char *pathname, int flags, ...)
{
    /* Handle variadic mode argument */
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list args;
        va_start(args, flags);
        mode = va_arg(args, mode_t);
        va_end(args);
    }

    /* Resolve the original function - must be per-call or cached */
    orig_open_t orig_open = (orig_open_t)dlsym(RTLD_NEXT, "open");
    if (!orig_open) {
        fprintf(stderr, "[-] dlsym(open) failed: %s\n", dlerror());
        return -1;
    }

    fprintf(stderr, "[HOOK] open(\"%s\", 0x%x, 0%o)\n", pathname, flags, mode);

    int fd = orig_open(pathname, flags, mode);

    fprintf(stderr, "[HOOK]   -> fd = %d\n", fd);
    return fd;
}

Every call to open() - from the binary, from any library it loads, from any thread - passes through this hook. The fprintf output goes to stderr so it doesn't contaminate normal output pipes.

Constructors and Destructors

__attribute__((constructor)) and __attribute__((destructor)) run automatically at load and unload time: 

__attribute__((constructor))
static void hook_init(void)
{
    fprintf(stderr, "[*] hook.so loaded - intercepting functions\n");
    fprintf(stderr, "[*] PID: %d\n", getpid());
}

__attribute__((destructor))
static void hook_fini(void)
{
    fprintf(stderr, "[*] hook.so unloaded - total syscalls intercepted: %lu\n",
            (unsigned long)intercept_count);
}

Constructors run before main() , after LD_PRELOAD libraries are mapped into the process address space. They fire even if the binary calls execve - each new process inherits LD_PRELOAD and runs constructors again.

File Access Monitor

A general-purpose file-access spy reveals every file a binary reads or writes:

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <stdarg.h>

/* open / open64 / openat - all file-opening variants */
typedef int (*orig_open_t)(const char *, int, ...);
typedef int (*orig_open64_t)(const char *, int, ...);
typedef int (*orig_openat_t)(int, const char *, int, ...);

static orig_open_t   real_open   = NULL;
static orig_open64_t real_open64 = NULL;
static orig_openat_t real_openat = NULL;

__attribute__((constructor))
static void init(void)
{
    real_open   = (orig_open_t)dlsym(RTLD_NEXT, "open");
    real_open64 = (orig_open64_t)dlsym(RTLD_NEXT, "open64");
    real_openat = (orig_openat_t)dlsym(RTLD_NEXT, "openat");
}

int open(const char *path, int flags, ...)
{
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list ap;
        va_start(ap, flags);
        mode = va_arg(ap, mode_t);
        va_end(ap);
    }

    const char *op = (flags & O_WRONLY || flags & O_RDWR) ? "WRITE" : "READ";
    fprintf(stderr, "[OPEN] %-6s | %s\n", op, path);

    return real_open ? real_open(path, flags, mode) : -1;
}

int open64(const char *path, int flags, ...)
{
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list ap;
        va_start(ap, flags);
        mode = va_arg(ap, mode_t);
        va_end(ap);
    }
    fprintf(stderr, "[OPEN64] %s\n", path);
    return real_open64 ? real_open64(path, flags, mode) : -1;
}

int openat(int dirfd, const char *path, int flags, ...)
{
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list ap;
        va_start(ap, flags);
        mode = va_arg(ap, mode_t);
        va_end(ap);
    }
    fprintf(stderr, "[OPENAT] dirfd=%d path=%s\n", dirfd, path);
    return real_openat ? real_openat(dirfd, path, flags, mode) : -1;
}

Network Connection Monitor

Intercept all outbound connections, revealing C2 infrastructure: 

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <string.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>
#include <unistd.h>

typedef int (*orig_connect_t)(int, const struct sockaddr *, socklen_t);

static orig_connect_t real_connect = NULL;

__attribute__((constructor))
static void init(void)
{
    real_connect = (orig_connect_t)dlsym(RTLD_NEXT, "connect");
}

int connect(int sockfd, const struct sockaddr *addr, socklen_t addrlen)
{
    char ip_str[INET6_ADDRSTRLEN] = {0};
    int port = 0;

    if (addr->sa_family == AF_INET) {
        struct sockaddr_in *sin = (struct sockaddr_in *)addr;
        inet_ntop(AF_INET, &sin->sin_addr, ip_str, sizeof(ip_str));
        port = ntohs(sin->sin_port);
    } else if (addr->sa_family == AF_INET6) {
        struct sockaddr_in6 *sin6 = (struct sockaddr_in6 *)addr;
        inet_ntop(AF_INET6, &sin6->sin6_addr, ip_str, sizeof(ip_str));
        port = ntohs(sin6->sin6_port);
    }

    fprintf(stderr, "[CONNECT] fd=%d -> %s:%d\n", sockfd, ip_str, port);

    int ret = real_connect ? real_connect(sockfd, addr, addrlen) : -1;
    fprintf(stderr, "[CONNECT]   result=%d\n", ret);
    return ret;
}

Command Execution Interceptor

Malware often calls execve , system , or popen to spawn shells or install additional payloads. Hook all three: 

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <signal.h>

typedef int    (*orig_execve_t)(const char *, char *const[], char *const[]);
typedef int    (*orig_system_t)(const char *);
typedef FILE * (*orig_popen_t)(const char *, const char *);

static orig_execve_t real_execve = NULL;
static orig_system_t real_system = NULL;
static orig_popen_t  real_popen  = NULL;

__attribute__((constructor))
static void init(void)
{
    real_execve = (orig_execve_t)dlsym(RTLD_NEXT, "execve");
    real_system = (orig_system_t)dlsym(RTLD_NEXT, "system");
    real_popen  = (orig_popen_t)dlsym(RTLD_NEXT, "popen");
}

int execve(const char *pathname, char *const argv[], char *const envp[])
{
    fprintf(stderr, "[EXECVE] %s", pathname);
    if (argv) {
        for (int i = 0; argv[i]; i++)
            fprintf(stderr, " [%d]=\"%s\"", i, argv[i]);
    }
    fprintf(stderr, "\n");
    return real_execve ? real_execve(pathname, argv, envp) : -1;
}

int system(const char *command)
{
    fprintf(stderr, "[SYSTEM] %s\n", command ? command : "(null)");
    return real_system ? real_system(command) : -1;
}

FILE *popen(const char *command, const char *type)
{
    fprintf(stderr, "[POPEN] mode=%s cmd=%s\n", type, command);
    return real_popen ? real_popen(command, type) : NULL;
}

Testing against a simple C program:

/* test_target.c */
#include <stdlib.h>
#include <unistd.h>
int main(void) {
    system("id");
    execl("/bin/ls", "ls", "-la", NULL);
    return 0;
}
						
gcc -o test_target test_target.c
gcc -shared -fPIC -o cmd_hook.so cmd_hook.c -ldl
LD_PRELOAD=./cmd_hook.so ./test_target

# Output:
# [SYSTEM] id
# uid=1000(user) gid=1000(user) ...
# [EXECVE] /bin/ls [0]="ls" [1]="-la"
# total 64
# drwxr-xr-x ...

Read/Write Data Capture

Intercept read() and write() to capture data flowing over file descriptors: 

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <ctype.h>

typedef ssize_t (*orig_read_t)(int, void *, size_t);
typedef ssize_t (*orig_write_t)(int, const void *, size_t);

static orig_read_t  real_read  = NULL;
static orig_write_t real_write = NULL;

__attribute__((constructor))
static void init(void)
{
    real_read  = (orig_read_t)dlsym(RTLD_NEXT, "read");
    real_write = (orig_write_t)dlsym(RTLD_NEXT, "write");
}

static void hexdump(const void *data, size_t len, const char *prefix)
{
    const unsigned char *p = data;
    fprintf(stderr, "%s len=%zu\n", prefix, len);
    for (size_t i = 0; i < len; i += 16) {
        fprintf(stderr, "  %04zx  ", i);
        for (size_t j = 0; j < 16 && (i + j) < len; j++)
            fprintf(stderr, "%02x ", p[i + j]);
        for (size_t j = 16 - (len - i < 16 ? len - i : 0); j > 0; j--)
            fprintf(stderr, "   ");
        fprintf(stderr, " |");
        for (size_t j = 0; j < 16 && (i + j) < len; j++)
            fprintf(stderr, "%c", isprint(p[i + j]) ? p[i + j] : '.');
        fprintf(stderr, "|\n");
    }
}

ssize_t read(int fd, void *buf, size_t count)
{
    ssize_t ret = real_read ? real_read(fd, buf, count) : -1;
    if (ret > 0 && ret < 1024) {
        fprintf(stderr, "[READ] fd=%d ", fd);
        hexdump(buf, (size_t)ret, "");
    }
    return ret;
}

ssize_t write(int fd, const void *buf, size_t count)
{
    fprintf(stderr, "[WRITE] fd=%d ", fd);
    hexdump(buf, count, "");
    return real_write ? real_write(fd, buf, count) : -1;
}

RTLD_NEXT vs RTLD_DEFAULT

Two search modes for dlsym :

Constant Behavior
RTLD_NEXT Find NEXT definition after the current library. Finds the real function.
RTLD_DEFAULT Search in default library search order. Starts from the beginning.

RTLD_NEXT is the correct choice in hook libraries. RTLD_DEFAULT would find your own hook function (an infinite loop otherwise avoided by caching). The distinction matters when multiple LD_PRELOAD libraries stack: 

# Stacking hooks: hook_A intercepts open, hook_B also intercepts open
LD_PRELOAD=./hook_A.so:./hook_B.so ./target
# Resolution order: hook_A, hook_B, libc
# hook_A's dlsym(RTLD_NEXT, "open") -> finds hook_B's open
# hook_B's dlsym(RTLD_NEXT, "open") -> finds libc's open

Compilation Gotchas

Hook libraries have strict compilation requirements:

# Correct compilation flags for x86_64:
gcc -shared -fPIC -o hook.so hook.c -ldl -fno-builtin

# CRUCIAL flags explained:
# -shared     : produce a shared object (.so), not executable
# -fPIC       : position-independent code (required for shared libs)
# -ldl        : link against libdl (provides dlopen/dlsym/dlclose/dlerror)
# -fno-builtin: do not replace standard function calls with builtins
#               without this, gcc may inline strlen() instead of calling it,
#               causing hook functions to break

Without '-fno-builtin' : GCC recognizes memset and memcpy as builtins and may inline them, bypassing your hook. Your hook's own internal calls to these functions might also be optimized away, making dlsym resolution fail silently.

Without '-ldl' : undefined reference to 'dlsym' - you must link against libdl explicitly.

Caching is mandatory : Calling dlsym(RTLD_NEXT, ...) on every invocation of the hooked function is correct but slow. Cache the result in a constructor or a static variable: 

static orig_open_t real_open = NULL;
static pthread_once_t once = PTHREAD_ONCE_INIT;

static void resolve_once(void) {
    real_open = (orig_open_t)dlsym(RTLD_NEXT, "open");
}

int open(const char *path, int flags, ...) {
    pthread_once(&once, resolve_once);
    // ... use real_open
}

For single-threaded targets, constructor-based caching suffices. For multi-threaded malware (botnets), use pthread_once for thread safety.

Hooking Variadic Functions

Variadic functions like printf , open (with O_CREAT), ioctl , fcntl are tricky because the number and types of arguments vary. Use va_list and explicitly check flags: 

int open(const char *pathname, int flags, ...)
{
    mode_t mode = 0;
    if (flags & O_CREAT) {
        va_list ap;
        va_start(ap, flags);
        mode = va_arg(ap, mode_t);
        va_end(ap);
    }

    /* The callee only reads mode when O_CREAT is set, so passing it always is safe. */
    return real_open(pathname, flags, mode);
}

Read the variadic argument exactly once. The kernel ignores the mode argument unless O_CREAT is set, so forward it unconditionally and never call va_arg twice.

Setuid and Capabilities

LD_PRELOAD is disabled for setuid binaries and binaries with filesystem capabilities. This is a deliberate security measure in glibc : 

$ sudo chmod u+s ./setuid_binary
$ LD_PRELOAD=./hook.so ./setuid_binary
# Hook NOT loaded. glibc ignores LD_PRELOAD for secure-execution mode.

The kernel exposes secure-execution mode through the auxiliary vector. This tiny program prints the AT_SECURE flag, which is non-zero for setuid binaries and binaries with file capabilities: 

/* secure-check.c - getauxval(AT_SECURE) is 0 normally, 1 in secure-execution mode. */
#include <stdio.h>
#include <sys/auxv.h>

int main(void)
{
    printf("AT_SECURE = %lu\n", getauxval(AT_SECURE));
    return 0;
}
						
gcc -o secure-check secure-check.c
./secure-check            # AT_SECURE = 0
sudo chmod u+s ./secure-check
./secure-check            # AT_SECURE = 1

For setuid binaries, use ptrace -based hooking (GDB) or kernel-level tracing (BPF/ftrace) instead of LD_PRELOAD.

Static Binaries

Statically linked binaries have no dynamic linker - ld.so is never invoked, LD_PRELOAD is never processed: 

$ file ./static_binary
static_binary: ELF 64-bit LSB executable, x86-64, statically linked
$ LD_PRELOAD=./hook.so ./static_binary
# Hook library is silently ignored. No constructor runs.
# Use strace, Frida, or patching techniques instead.

Detect static linking:

readelf -d ./binary | grep NEEDED   # empty -> statically linked, or
ldd ./binary | grep "not a dynamic" # explicitly says "not a dynamic executable"

Raw Syscalls

Malware may bypass libc wrappers entirely by invoking syscalls directly. An LD_PRELOAD hook on write() cannot intercept: 

/* Bypasses LD_PRELOAD hooks on write() */
#include <unistd.h>
#include <sys/syscall.h>

int main(void) {
    const char *msg = "hidden message\n";
    syscall(SYS_write, 1, msg, 15); /* raw syscall - no libc wrapper */
    return 0;
}
Interception Layer Catches libc wrappers Catches raw syscalls
LD_PRELOAD yes no
strace (ptrace) yes yes
Frida yes yes (via syscall)
eBPF tracepoints no yes

Combine LD_PRELOAD with strace for full coverage.

General-Purpose Spy Library

A production-ready spy library that logs every significant call:

/*
 * spy.c - General-purpose LD_PRELOAD interception library
 *
 * Compile: gcc -shared -fPIC -fno-builtin -o spy.so spy.c -ldl
 * Usage:   LD_PRELOAD=./spy.so SPY_LOG=./spy.log ./target
 *          SPY_FILTER=open,execve ./target  # only these functions
 */

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <stdarg.h>
#include <time.h>
#include <fcntl.h>
#include <sys/socket.h>
#include <netinet/in.h>
#include <arpa/inet.h>

static FILE *spy_log = NULL;
static int spy_count = 0;

__attribute__((constructor))
static void spy_init(void)
{
    const char *logpath = getenv("SPY_LOG");
    if (logpath) {
        spy_log = fopen(logpath, "a");
    }
    if (!spy_log) spy_log = stderr;

    time_t now = time(NULL);
    fprintf(spy_log, "=== SPY START PID=%d TIME=%s", getpid(), ctime(&now));
}

__attribute__((destructor))
static void spy_fini(void)
{
    fprintf(spy_log, "=== SPY END total_calls=%d\n", spy_count);
    if (spy_log != stderr) fclose(spy_log);
}

/* -- open ---------------------------------------------------------------- */

typedef int (*orig_open_t)(const char *, int, ...);
int open(const char *pathname, int flags, ...)
{
    orig_open_t f = (orig_open_t)dlsym(RTLD_NEXT, "open");

    va_list ap;
    va_start(ap, flags);
    mode_t mode = (flags & O_CREAT) ? va_arg(ap, mode_t) : 0;
    va_end(ap);

    int ret = f ? f(pathname, flags, mode) : -1;
    fprintf(spy_log, "[%d] open(\"%s\", 0x%x) = %d\n", ++spy_count, pathname, flags, ret);
    return ret;
}

/* -- connect ------------------------------------------------------------- */

typedef int (*orig_connect_t)(int, const struct sockaddr *, socklen_t);
int connect(int sockfd, const struct sockaddr *addr, socklen_t addrlen)
{
    orig_connect_t f = (orig_connect_t)dlsym(RTLD_NEXT, "connect");

    char ip[INET6_ADDRSTRLEN] = "?";
    int port = 0;
    if (addr->sa_family == AF_INET) {
        struct sockaddr_in *s = (struct sockaddr_in *)addr;
        inet_ntop(AF_INET, &s->sin_addr, ip, sizeof(ip));
        port = ntohs(s->sin_port);
    }

    int ret = f ? f(sockfd, addr, addrlen) : -1;
    fprintf(spy_log, "[%d] connect(%d, \"%s\", %d) = %d\n",
            ++spy_count, sockfd, ip, port, ret);
    return ret;
}

/* -- execve -------------------------------------------------------------- */

typedef int (*orig_execve_t)(const char *, char *const[], char *const[]);
int execve(const char *pathname, char *const argv[], char *const envp[])
{
    orig_execve_t f = (orig_execve_t)dlsym(RTLD_NEXT, "execve");

    fprintf(spy_log, "[%d] execve(\"%s\", [", ++spy_count, pathname);
    if (argv)
        for (int i = 0; argv[i]; i++)
            fprintf(spy_log, "\"%s\"%s", argv[i], argv[i+1] ? ", " : "");
    fprintf(spy_log, "])\n");

    return f ? f(pathname, argv, envp) : -1;
}

/* -- system -------------------------------------------------------------- */

typedef int (*orig_system_t)(const char *);
int system(const char *command)
{
    orig_system_t f = (orig_system_t)dlsym(RTLD_NEXT, "system");
    fprintf(spy_log, "[%d] system(\"%s\")\n", ++spy_count, command ? command : "");
    return f ? f(command) : -1;
}

/* -- dlopen / dlsym (detect dynamic API resolution) ---------------------- */

typedef void * (*orig_dlopen_t)(const char *, int);
void *dlopen(const char *filename, int flags)
{
    orig_dlopen_t f = (orig_dlopen_t)dlsym(RTLD_NEXT, "dlopen");
    fprintf(spy_log, "[%d] dlopen(\"%s\", 0x%x)\n", ++spy_count, filename, flags);
    return f ? f(filename, flags) : NULL;
}

typedef void * (*orig_dlsym_t)(void *, const char *);
void *dlsym(void *handle, const char *symbol)
{
    orig_dlsym_t f = (orig_dlsym_t)dlsym(RTLD_NEXT, "dlsym");
    fprintf(spy_log, "[%d] dlsym(%p, \"%s\")\n", ++spy_count, handle, symbol);
    return f ? f(handle, symbol) : NULL;
}

/* -- socket / bind / listen (backdoor detection) ------------------------- */

typedef int (*orig_socket_t)(int, int, int);
int socket(int domain, int type, int protocol)
{
    orig_socket_t f = (orig_socket_t)dlsym(RTLD_NEXT, "socket");
    int ret = f ? f(domain, type, protocol) : -1;
    fprintf(spy_log, "[%d] socket(%d, %d, %d) = %d\n",
            ++spy_count, domain, type, protocol, ret);
    return ret;
}

typedef int (*orig_bind_t)(int, const struct sockaddr *, socklen_t);
int bind(int sockfd, const struct sockaddr *addr, socklen_t addrlen)
{
    orig_bind_t f = (orig_bind_t)dlsym(RTLD_NEXT, "bind");
    int port = 0;
    if (addr->sa_family == AF_INET)
        port = ntohs(((struct sockaddr_in *)addr)->sin_port);
    fprintf(spy_log, "[%d] bind(%d, *, %d)\n", ++spy_count, sockfd, port);
    return f ? f(sockfd, addr, addrlen) : -1;
}

Combining With strace

For complete coverage where LD_PRELOAD misses raw syscalls, run both simultaneously:

# LD_PRELOAD catches libc-level functions, strace catches everything else
LD_PRELOAD=./spy.so strace -o strace.log -f ./target

# Filter strace for raw syscalls your LD_PRELOAD hooks don't see:
grep '^write\|^read\|^openat' strace.log

Library Injection

LD_PRELOAD only affects newly started processes. To inject into an already-running process, use gdb : 

gdb -p $(pgrep target_process)
(gdb) call (int)dlopen("/path/to/hook.so", 2)   # RTLD_NOW = 2
(gdb) detach
(gdb) quit

# The hook library's constructor runs upon dlopen.
# Hook functions are now active in the running process.
# WARNING: This crashes if the process is mid-syscall or holds locks.

Safer: use Frida for runtime injection into production processes.

ld.so.preload Persistence

Malware often writes to /etc/ld.so.preload for persistence - this file is read by the dynamic linker on every process start, applying the specified libraries system-wide: 

# Detect LD_PRELOAD-based persistence:
cat /etc/ld.so.preload
# If it contains suspicious paths, a userland rootkit may be installed.

# The ld.so.preload file allows one library path per line:
# /lib/librootkit.so
# /usr/local/lib/libhide.so

The analysis technique (LD_PRELOAD interception) and the malware technique (ld.so.preload persistence) are the same mechanism used for different purposes. When analyzing a binary that writes to /etc/ld.so.preload , you've found a userland rootkit installer.

Summary

LD_PRELOAD hooks intercept every dynamically-linked function call - open , connect , execve , system , dlopen , read , write - giving you complete observability of a binary's interaction with the OS. It is the first tool to reach for when analyzing unknown Linux binaries. No kernel modules, no ptrace overhead, no performance penalty beyond your logging code. Combine with strace for syscall-level gaps, with GDB for interactive debugging, and with Frida for runtime injection. The compiled .so is portable across any Linux system with the same architecture and glibc version.

Challenge

Write a Go tool that prints the effective symbol search order for a binary without running it. Parse the ELF dynamic section for DT_NEEDED , DT_RPATH , and DT_RUNPATH , then combine that with the LD_PRELOAD environment variable and /etc/ld.so.preload . Run it with and without LD_PRELOAD set and confirm that the preload libraries are searched before the DT_NEEDED libraries. 

package main

import (
	"bufio"
	"debug/elf"
	"fmt"
	"os"
	"strings"
)

func main() {
	if len(os.Args) != 2 {
		fmt.Fprintln(os.Stderr, "usage: loadorder <elf-binary>")
		os.Exit(1)
	}

	f, err := elf.Open(os.Args[1])
	if err != nil {
		fmt.Fprintln(os.Stderr, "open:", err)
		os.Exit(1)
	}
	defer f.Close()

	fmt.Println("preload sources (searched before DT_NEEDED):")
	for _, lib := range strings.Split(os.Getenv("LD_PRELOAD"), ":") {
		if lib = strings.TrimSpace(lib); lib != "" {
			fmt.Printf("  LD_PRELOAD      %s\n", lib)
		}
	}
	if file, err := os.Open("/etc/ld.so.preload"); err == nil {
		scanner := bufio.NewScanner(file)
		for scanner.Scan() {
			line := strings.TrimSpace(scanner.Text())
			if line != "" && !strings.HasPrefix(line, "#") {
				fmt.Printf("  ld.so.preload   %s\n", line)
			}
		}
		file.Close()
	}

	fmt.Println("DT_NEEDED (breadth-first link order):")
	for _, lib := range f.ImportedLibraries() {
		fmt.Printf("  %s\n", lib)
	}

	for _, tag := range []elf.DynTag{elf.DT_RPATH, elf.DT_RUNPATH} {
		if paths, err := f.DynString(tag); err == nil && len(paths) > 0 {
			fmt.Printf("%s:\n", tag)
			for _, p := range paths {
				fmt.Printf("  %s\n", p)
			}
		}
	}
}

Build it and compare the two runs:

go build -o loadorder loadorder.go
./loadorder /bin/ls

gcc -shared -fPIC -o hook.so hook.c -ldl
LD_PRELOAD=$PWD/hook.so ./loadorder /bin/ls
