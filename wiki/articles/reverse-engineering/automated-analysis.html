Automated Analysis Pipelines

Building malware analysis pipelines with triage, static analysis, dynamic sandboxing, and Ghidra headless scripting.

Overview

A single malware sample can take hours to analyze by hand. A pipeline automates the repetitive first passes so that only interesting samples reach an analyst. This article builds a staged Linux pipeline: directory ingestion, fast triage, headless Ghidra analysis, a podman sandbox, similarity clustering, and a small results dashboard. After reading it you can run the whole pipeline end to end and extend any stage.

Required Reading

  • Sandbox Analysis - contained execution environments, strace, tcpdump
  • Ghidra Scripting - headless Ghidra, Python and Java scripting
  • Symbolic Execution - angr integration for constraint solving

Pipeline Architecture

A production malware analysis pipeline processes samples through discrete stages, each enriching the analysis with additional data. The stages are designed so early fast filters reduce load on expensive later stages:

INGESTION  →  TRIAGE  →  STATIC  →  DYNAMIC  →  CORRELATION  →  REPORT
  (API)      (<5s)      (5-60s)    (60-180s)     (async)        (JSON/HTML)
Stage Input Output Time Budget Tools
Ingestion Sample file Queued job <1s HTTP API, inotify watch
Triage Raw file Hash, type, entropy, strings, YARA hits <5s Python, ssdeep, YARA, magic
Static ELF/PE file Functions, imports, crypto, capabilities 30-120s Ghidra headless, readelf
Dynamic ELF/PE file Behavioral trace, syscalls, network 60-300s podman, strace, tcpdump
Correlation All reports Clusters, family attribution Async ssdeep, Jaccard, API hashes
Report Analysis data JSON, dashboard, alerts Async Flask, Elastic

Stage 1: Ingestion

The simplest ingestion is a watched directory. Any file dropped to /opt/samples/incoming triggers analysis:

#!/usr/bin/env python3
"""ingest.py — Watch directory and queue samples for analysis"""

import os
import json
import time
import hashlib
import shutil
from pathlib import Path
from datetime import datetime

INCOMING_DIR  = Path('/opt/samples/incoming')
PROCESSED_DIR = Path('/opt/samples/processed')
FAILED_DIR    = Path('/opt/samples/failed')
QUEUE_FILE    = Path('/opt/samples/queue.jsonl')

for d in [INCOMING_DIR, PROCESSED_DIR, FAILED_DIR]:
    d.mkdir(parents=True, exist_ok=True)


def compute_hashes(filepath: Path) -> dict:
    """Compute MD5, SHA1, SHA256, and ssdeep for a file."""
    hashes = {'md5': hashlib.md5(), 'sha1': hashlib.sha1(),
              'sha256': hashlib.sha256()}

    with open(filepath, 'rb') as f:
        while chunk := f.read(65536):
            for h in hashes.values():
                h.update(chunk)

    result = {name: h.hexdigest() for name, h in hashes.items()}

    # ssdeep requires full file read — compute separately
    try:
        import ssdeep
        with open(filepath, 'rb') as f:
            result['ssdeep'] = ssdeep.hash(f.read())
    except ImportError:
        result['ssdeep'] = None

    return result


def ingest_sample(filepath: Path) -> dict:
    """Ingest one sample → compute hashes → move to processed → queue job."""
    hashes = compute_hashes(filepath)

    # Check if already processed (deduplicate by SHA256)
    for processed in PROCESSED_DIR.rglob('*.meta.json'):
        try:
            meta = json.loads(processed.read_text())
            if meta.get('sha256') == hashes['sha256']:
                print(f"[SKIP] Duplicate: {filepath.name} (SHA256 match)")
                filepath.unlink()
                return None
        except:
            pass

    # Move to processed with metadata
    dest = PROCESSED_DIR / filepath.name
    shutil.move(str(filepath), str(dest))

    meta = {
        'original_name': filepath.name,
        'processed_path': str(dest),
        'ingested_at': datetime.utcnow().isoformat(),
        **hashes
    }

    meta_path = dest.with_suffix(dest.suffix + '.meta.json')
    meta_path.write_text(json.dumps(meta, indent=2))

    # Write to queue
    with open(QUEUE_FILE, 'a') as q:
        q.write(json.dumps(meta) + '\n')

    print(f"[INGEST] {filepath.name} → {dest.name} "
          f"(SHA256: {hashes['sha256'][:16]})")
    return meta


def watch_directory():
    """Poll incoming directory for new files."""
    print(f"[*] Watching {INCOMING_DIR} for new samples...")
    seen = set()

    while True:
        for filepath in INCOMING_DIR.iterdir():
            if filepath.is_file() and filepath.name not in seen:
                seen.add(filepath.name)
                try:
                    ingest_sample(filepath)
                except Exception as e:
                    print(f"[ERROR] Failed to ingest {filepath}: {e}")
                    FAILED_DIR.mkdir(parents=True, exist_ok=True)
                    shutil.move(str(filepath), str(FAILED_DIR / filepath.name))

        time.sleep(2)


if __name__ == '__main__':
    watch_directory()

Stage 2: Triage - Fast Pre-Filtering

Triage extracts maximum information in minimum time. The goal: determine whether the sample warrants deep analysis.

#!/usr/bin/env python3
"""triage.py — Fast malware triage (<5 seconds per sample)"""

import os
import re
import json
import math
import struct
import subprocess
import tempfile
from pathlib import Path
from collections import Counter
from typing import Dict, List, Optional


class SampleTriage:
    def __init__(self, filepath: Path):
        self.path = Path(filepath)
        self.data = self.path.read_bytes() if self.path.stat().st_size < 100_000_000 else b''
        self.results = {
            'file': str(self.path),
            'size': self.path.stat().st_size,
            'name': self.path.name,
        }

    def entropy(self, data: bytes) -> float:
        """Shannon entropy."""
        if not data:
            return 0.0
        counts = Counter(data)
        length = len(data)
        return -sum((c / length) * math.log2(c / length) for c in counts.values())

    def check_file_type(self) -> Dict:
        """Determine file type via magic bytes and file command."""
        result = {}

        # Run file command
        try:
            proc = subprocess.run(['file', '-b', str(self.path)],
                                  capture_output=True, text=True)
            result['file_type'] = proc.stdout.strip()
        except:
            result['file_type'] = 'unknown'

        # Check ELF header
        if len(self.data) >= 4 and self.data[:4] == b'\x7fELF':
            result['format'] = 'ELF'
            if len(self.data) >= 18:
                result['bits'] = 64 if self.data[4] == 2 else 32
                endian = 'LE' if self.data[5] == 1 else 'BE'
                machine_map = {
                    0x03: 'i386', 0x3E: 'x86-64', 0x28: 'ARM',
                    0xB7: 'AArch64', 0x08: 'MIPS', 0x14: 'PPC',
                    0xF3: 'RISC-V'
                }
                machine = struct.unpack_from('<H', self.data, 18)[0]
                result['arch'] = f"{machine_map.get(machine, hex(machine))}-{endian}"

                # Linked type
                e_type = struct.unpack_from('<H', self.data, 16)[0]
                type_map = {0: 'NONE', 1: 'REL', 2: 'EXEC', 3: 'DYN', 4: 'CORE'}
                result['elf_type'] = type_map.get(e_type, str(e_type))

        elif len(self.data) >= 2 and self.data[:2] == b'MZ':
            result['format'] = 'PE'
        else:
            result['format'] = 'unknown'

        return result

    def compute_entropy_profile(self) -> Dict:
        """Compute entropy for the whole file and per-256-byte window."""
        profile = {}

        # Overall entropy
        sample = self.data[:min(len(self.data), 1_000_000)]
        profile['overall'] = round(self.entropy(sample), 4)

        # Windowed entropy (detect packed/encrypted regions)
        window_size = 256
        entropies = []
        for i in range(0, min(len(self.data), 100_000), window_size):
            entropies.append(self.entropy(self.data[i:i + window_size]))

        if entropies:
            profile['mean'] = round(sum(entropies) / len(entropies), 4)
            profile['max']  = round(max(entropies), 4)
            profile['min']  = round(min(entropies), 4)

        return profile

    def extract_strings(self, min_len: int = 6) -> List[str]:
        """Extract printable ASCII strings."""
        pattern = re.compile(rb'[\x20-\x7e]{%d,}' % min_len)
        strings = []
        seen = set()

        for match in pattern.finditer(self.data):
            s = match.group().decode('ascii', errors='ignore')
            if s not in seen and len(s) <= 200:
                seen.add(s)
                strings.append(s)
                if len(strings) > 10000:
                    break

        return strings

    def detect_packer(self, strings: List[str]) -> Dict:
        """Detect packing based on strings and section names."""
        result = {'packed': False, 'packer': None}

        packer_signatures = {
            'UPX': ['UPX0', 'UPX1', 'UPX!', 'UPX'],
            'MPRESS': ['MPRESS', '.MPRESS'],
            'ASPack': ['.aspack', '.adata'],
        }

        for i, s in enumerate(strings):
            for packer, sigs in packer_signatures.items():
                for sig in sigs:
                    if sig in s:
                        result['packed'] = True
                        result['packer'] = packer
                        result['matched_string'] = s
                        return result

        # High entropy + few strings = probably packed
        if len(strings) < 50 and len(self.data) > 10000:
            ent = self.entropy(self.data[:100000])
            if ent > 7.5:
                result['packed'] = True
                result['packer'] = 'unknown (high entropy)'

        return result

    def detect_crypto_constants(self, strings: List[str]) -> List[str]:
        """Detect crypto implementations from strings."""
        crypto_strings = {
            'AES': ['aes', 'rijndael', 'sbox'],
            'RC4': ['arcfour', 'rc4'],
            'MD5': ['md5', 'MD5_Init', 'MD5_Update', 'MD5_Final'],
            'SHA1': ['sha1', 'SHA1_Init'],
            'SHA256': ['sha256', 'SHA256_Init'],
            'RSA': ['rsa', 'RSA_', 'bigint'],
            'DES': ['des_', 'DES_'],
            'ChaCha': ['chacha', 'chacha20'],
            'Blowfish': ['blowfish', 'BF_'],
            'Base64': ['base64', 'B64_'],
            'Zlib': ['zlib', 'deflate', 'inflate'],
        }

        detected = []
        found = set()
        for s in strings:
            s_lower = s.lower()
            for algo, keywords in crypto_strings.items():
                if algo not in found:
                    for kw in keywords:
                        if kw in s_lower:
                            detected.append(algo)
                            found.add(algo)
                            break

        return detected

    def scan_yara(self) -> List[Dict]:
        """Run YARA rules against the sample."""
        yara_hits = []

        # Try standard rule paths
        rule_paths = [
            Path('/opt/yara/rules'),
            Path('/usr/share/yara'),
            Path.home() / '.yara/rules',
        ]

        for rules_dir in rule_paths:
            if not rules_dir.exists():
                continue

            for rule_file in rules_dir.rglob('*.yar'):
                try:
                    proc = subprocess.run(
                        ['yara', '-C', str(rule_file), str(self.path)],
                        capture_output=True, text=True, timeout=30
                    )
                    if proc.stdout.strip():
                        for line in proc.stdout.strip().split('\n'):
                            parts = line.split(' ', 1)
                            if len(parts) == 2:
                                yara_hits.append({
                                    'rule': parts[0],
                                    'file': rule_file.name,
                                    'match': parts[1]
                                })
                except subprocess.TimeoutExpired:
                    continue
                except FileNotFoundError:
                    break  # yara not installed

        return yara_hits

    def extract_ip_urls(self, strings: List[str]) -> Dict:
        """Extract IPs and URLs from strings."""
        ips = set()
        urls = set()

        ip_pattern = re.compile(r'\b(?:\d{1,3}\.){3}\d{1,3}\b')
        url_pattern = re.compile(r'https?://[^\s"\'<>]+')

        for s in strings:
            for match in ip_pattern.finditer(s):
                ips.add(match.group())
            for match in url_pattern.finditer(s):
                urls.add(match.group()[:120])

        return {
            'ips': sorted(ips)[:50],
            'urls': sorted(urls)[:50],
            'ip_count': len(ips),
            'url_count': len(urls),
        }

    def run(self) -> Dict:
        """Execute all triage checks."""
        print(f"[TRIAGE] {self.path.name} ({self.results['size']} bytes)")

        # File type
        self.results['type'] = self.check_file_type()

        # Entropy
        self.results['entropy'] = self.compute_entropy_profile()

        # Strings
        strings = self.extract_strings()
        self.results['string_count'] = len(strings)
        self.results['top_strings'] = strings[:100]

        # Packer detection
        self.results['packer'] = self.detect_packer(strings)

        # Crypto detection
        self.results['crypto_hints'] = self.detect_crypto_constants(strings)

        # Network indicators
        self.results['network'] = self.extract_ip_urls(strings)

        # YARA
        if self.path.stat().st_size < 50_000_000:
            self.results['yara'] = self.scan_yara()

        # Triage verdict
        verdict = self._compute_verdict()
        self.results['verdict'] = verdict

        return self.results

    def _compute_verdict(self) -> str:
        """Heuristic verdict based on triage results."""
        score = 0

        if self.results['packer']['packed']:
            score += 30
        if self.results['entropy'].get('overall', 0) > 7.5:
            score += 20
        if self.results['network']['ip_count'] > 5:
            score += 10
        if self.results.get('crypto_hints'):
            score += 10
        if self.results.get('yara') and len(self.results['yara']) > 0:
            score += 25
        if self.results['string_count'] < 30 and self.results['size'] > 50000:
            score += 15

        if score >= 50:
            return 'MALICIOUS'
        elif score >= 25:
            return 'SUSPICIOUS'
        else:
            return 'BENIGN'


if __name__ == '__main__':
    import sys
    for path in sys.argv[1:]:
        triage = SampleTriage(Path(path))
        results = triage.run()

        out_path = Path(path).with_suffix('.triage.json')
        out_path.write_text(json.dumps(results, indent=2))
        print(f"  → {out_path}")
        print(f"  Verdict: {results['verdict']}")
        print(f"  Packer:  {results['packer']}")
        print(f"  Entropy: {results['entropy'].get('overall')}")
        print(f"  Strings: {results['string_count']}")

Stage 3: Headless Ghidra

Ghidra's analyzeHeadless command runs scripts without GUI :

$ analyzeHeadless /opt/ghidra/projects MalwareProj \
    -import /opt/samples/sample.elf \
    -postScript ExportAnalysis.java \
    -deleteProject \
    -noanalysis

The -postScript runs after auto-analysis. A production export script:

/* ExportAnalysis.java — Export all analysis data to JSON */
import ghidra.app.script.GhidraScript;
import ghidra.program.model.listing.*;
import ghidra.program.model.symbol.*;
import ghidra.program.model.pcode.*;
import com.google.gson.*;
import java.io.*;
import java.util.*;

public class ExportAnalysis extends GhidraScript {

    @Override
    public void run() throws Exception {
        JsonObject report = new JsonObject();

        // Basic info
        report.addProperty("name", currentProgram.getName());
        report.addProperty("md5", getMD5());
        AddressIterator entryPoints = currentProgram.getSymbolTable()
            .getExternalEntryPointIterator();
        report.addProperty("entry_point",
            entryPoints.hasNext() ? entryPoints.next().toString() : "unknown");

        // Functions
        JsonArray functions = new JsonArray();
        FunctionIterator funcs = currentProgram.getFunctionManager()
            .getFunctions(true);
        int funcCount = 0;

        while (funcs.hasNext() && funcCount < 5000) {
            Function func = funcs.next();
            JsonObject fobj = new JsonObject();
            fobj.addProperty("name", func.getName());
            fobj.addProperty("addr", func.getEntryPoint().toString());
            fobj.addProperty("size", (int)func.getBody().getNumAddresses());

            // Signature (instructions count, calls count)
            fobj.addProperty("is_thunk", func.isThunk());
            fobj.addProperty("has_loop", hasLoop(func));

            // Called functions
            JsonArray calls = new JsonArray();
            Set<Function> called = func.getCalledFunctions(monitor);
            for (Function callee : called) {
                calls.add(new JsonPrimitive(callee.getEntryPoint().toString()));
            }
            fobj.add("calls", calls);

            functions.add(fobj);
            funcCount++;
        }
        report.addProperty("function_count", funcCount);
        report.add("functions", functions);

        // Imports (external functions)
        JsonArray imports = new JsonArray();
        SymbolIterator syms = currentProgram.getSymbolTable()
            .getExternalSymbols();
        while (syms.hasNext()) {
            Symbol sym = syms.next();
            JsonObject imp = new JsonObject();
            imp.addProperty("name", sym.getName());
            imp.addProperty("lib", sym.getParentNamespace().getName());
            imports.add(imp);
        }
        report.add("imports", imports);

        // Strings (first 10000)
        JsonArray strings = new JsonArray();
        DataIterator data = currentProgram.getListing()
            .getDefinedData(true);
        int strCount = 0;

        while (data.hasNext() && strCount < 10000) {
            Data d = data.next();
            if (d.hasStringValue()) {
                String s = d.getDefaultValueRepresentation();
                if (s.length() >= 4) {
                    JsonObject strobj = new JsonObject();
                    strobj.addProperty("addr", d.getAddress().toString());
                    strobj.addProperty("value", s);
                    strings.add(strobj);
                    strCount++;
                }
            }
        }
        report.addProperty("string_count", strCount);
        report.add("strings", strings);

        // Write to output file
        String outPath = getScriptArgs()[0];
        FileWriter writer = new FileWriter(outPath);
        Gson gson = new GsonBuilder().setPrettyPrinting().create();
        gson.toJson(report, writer);
        writer.close();

        printf("Exported to %s\n", outPath);
    }

    private String getMD5() {
        String md5 = currentProgram.getExecutableMD5();
        return md5 != null ? md5 : "unknown";
    }

    private boolean hasLoop(Function func) {
        // Check if CFG has loops (back edges)
        // Simplified — use PcodeBlockBasic iteration
        return false;
    }
}

Python wrapper for headless Ghidra

#!/usr/bin/env python3
"""ghidra_headless.py — Run Ghidra analysis from Python"""

import os
import json
import subprocess
import tempfile
from pathlib import Path

GHIDRA_HOME = Path(os.environ.get('GHIDRA_HOME', '/opt/ghidra'))
HEADLESS_BIN = GHIDRA_HOME / 'support' / 'analyzeHeadless'

def analyze_sample(sample_path: Path, project_dir: Path = None) -> Path:
    """
    Run headless Ghidra on a sample, export results to JSON.
    Returns path to JSON output file.
    """
    if project_dir is None:
        project_dir = Path(tempfile.mkdtemp(prefix='ghidra_proj_'))

    project_name = f"analysis_{sample_path.stem}"
    output_json = sample_path.with_suffix('.ghidra.json')

    cmd = [
        str(HEADLESS_BIN),
        str(project_dir),
        project_name,
        '-import', str(sample_path),
        '-postScript', 'ExportAnalysis.java', str(output_json),
        '-deleteProject',
        '-max-cpu', '4',
    ]

    try:
        proc = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            timeout=300  # 5 minute timeout
        )

        if output_json.exists():
            with open(output_json) as f:
                report = json.load(f)
                print(f"[GHIDRA] {report.get('function_count', 0)} functions, "
                      f"{report.get('string_count', 0)} strings")
            return output_json
        else:
            print(f"[GHIDRA] Error: {proc.stderr[:500]}")
            return None

    except subprocess.TimeoutExpired:
        print(f"[GHIDRA] Timeout analyzing {sample_path.name}")
        return None
    except Exception as e:
        print(f"[GHIDRA] Exception: {e}")
        return None

Stage 4: podman Sandbox

The dynamic stage runs the sample in an isolated container, capturing system calls and network traffic:

#!/usr/bin/env python3
"""sandbox_runner.py — podman-based dynamic analysis sandbox"""

import os
import json
import time
import shlex
import subprocess
import tempfile
from pathlib import Path
from datetime import datetime

SANDBOX_IMAGE = 'malware-sandbox:latest'
SANDBOX_TIMEOUT = 60  # seconds


class SandboxRunner:
    def __init__(self, sample_path: Path, timeout: int = SANDBOX_TIMEOUT):
        self.sample_path = sample_path.resolve()
        self.sample_name = sample_path.name
        self.timeout = timeout
        self.work_dir = Path(tempfile.mkdtemp(prefix='sandbox_'))
        self.results = {
            'sample': str(sample_path),
            'started_at': datetime.utcnow().isoformat(),
            'timeout': timeout,
        }

        # Copy sample into work directory (read-only inside container)
        import shutil
        self.container_sample = self.work_dir / self.sample_name
        shutil.copy2(sample_path, self.container_sample)

    def build_sandbox_image(self):
        """Build the analysis sandbox container image."""
        dockerfile = self.work_dir / 'Containerfile'
        dockerfile.write_text('''
FROM archlinux:latest
RUN pacman -Syu --noconfirm && pacman -S --noconfirm \
    strace tcpdump procps-ng binutils file curl wget
RUN useradd -m -s /bin/bash malware
USER malware
WORKDIR /home/malware
''')
        subprocess.run(['podman', 'build', '-t', SANDBOX_IMAGE,
                        str(self.work_dir)], check=True,
                       capture_output=True)

    def run_strace(self) -> Path:
        """Run the sample under strace, capture all syscalls."""
        output_path = self.work_dir / 'strace.log'

        # strace arguments: -f (follow forks), -v (verbose),
        # -o (output file), -e trace=all (all syscalls)
        strace_cmd = (
            f'strace -f -v -x -o /tmp/analysis/strace.log '
            f'-e trace=all '
            f'/tmp/analysis/{shlex.quote(self.sample_name)}'
        )

        return self._run_container(strace_cmd, output_path)

    def run_tcpdump(self) -> Path:
        """Run the sample while capturing network traffic."""
        output_path = self.work_dir / 'network.pcap'

        tcpdump_cmd = (
            f'tcpdump -i any -w /tmp/analysis/network.pcap '
            f'-s 65535 > /dev/null 2>&1 & '
            f'TCPDUMP_PID=$! ; '
            f'sleep 1 ; '
            f'/tmp/analysis/{shlex.quote(self.sample_name)} ; '
            f'EXIT_CODE=$? ; '
            f'sleep 2 ; '
            f'kill $TCPDUMP_PID 2>/dev/null ; '
            f'wait $TCPDUMP_PID 2>/dev/null ; '
            f'exit $EXIT_CODE'
        )

        return self._run_container(tcpdump_cmd, output_path)

    def run_combined(self) -> Path:
        """Run with both strace AND tcpdump simultaneously."""
        output_path = self.work_dir / 'combined.log'

        cmd = (
            f'tcpdump -i any -w /tmp/analysis/network.pcap '
            f'-s 65535 > /dev/null 2>&1 & '
            f'TCPDUMP_PID=$! ; '
            f'sleep 1 ; '
            f'strace -f -v -x -o /tmp/analysis/strace.log '
            f'-e trace=all '
            f'/tmp/analysis/{shlex.quote(self.sample_name)} ; '
            f'EXIT_CODE=$? ; '
            f'sleep 2 ; '
            f'kill $TCPDUMP_PID 2>/dev/null ; '
            f'wait $TCPDUMP_PID 2>/dev/null ; '
            f'echo "EXIT_CODE:$EXIT_CODE" > /tmp/analysis/exit_code.txt'
        )

        return self._run_container(cmd, output_path)

    def _run_container(self, cmd: str, output_path: Path) -> Path:
        """Execute command inside podman container and capture output."""
        container_name = f"sandbox_{self.sample_name[:20]}_{int(time.time())}"

        podman_cmd = [
            'podman', 'run',
            '--rm',
            '--name', container_name,
            '--network', 'none',          # No network (or 'bridge' for tcpdump)
            '--cpus', '1',
            '--memory', '512m',
            '--pids-limit', '100',
            '--security-opt', 'no-new-privileges',
            '--read-only', '--tmpfs', '/tmp:rw,noexec,nosuid,size=100m',
            '--tmpfs', '/var/tmp:rw,noexec,nosuid,size=100m',
            '--tmpfs', '/home/malware:rw,noexec,nosuid,size=50m',
            '-v', f'{self.work_dir}:/tmp/analysis:Z',
            SANDBOX_IMAGE,
            'bash', '-c', cmd,
        ]

        if 'tcpdump' in cmd:
            # Replace --network none with user-mode networking so tcpdump
            # has an interface to capture, while keeping inbound blocked.
            podman_cmd[6] = 'slirp4netns'

        print(f"[SANDBOX] Running: {container_name}")

        try:
            proc = subprocess.run(
                podman_cmd,
                capture_output=True,
                text=True,
                timeout=self.timeout + 10
            )
            self.results['container_stdout'] = proc.stdout[-5000:]
            self.results['container_stderr'] = proc.stderr[-5000:]
            self.results['returncode'] = proc.returncode

            # Parse exit code
            exit_file = self.work_dir / 'exit_code.txt'
            if exit_file.exists():
                exit_content = exit_file.read_text().strip()
                if 'EXIT_CODE:' in exit_content:
                    self.results['sample_exit_code'] = \
                        int(exit_content.split(':')[1])
        except subprocess.TimeoutExpired:
            self.results['timed_out'] = True
            # Force remove container
            subprocess.run(['podman', 'kill', container_name],
                           capture_output=True)
            subprocess.run(['podman', 'rm', '-f', container_name],
                           capture_output=True)

        # Collect results
        for artifact in ['strace.log', 'network.pcap', 'exit_code.txt']:
            path = self.work_dir / artifact
            if path.exists():
                self.results[f'{artifact}_size'] = path.stat().st_size

        return output_path

    def parse_strace(self) -> Dict:
        """Parse strace output to extract syscall statistics."""
        strace_path = self.work_dir / 'strace.log'
        if not strace_path.exists():
            return {}

        syscalls = {}
        with open(strace_path, errors='replace') as f:
            for line in f:
                # Format: pid syscall(args) = result
                if '(' in line and '=' in line:
                    syscall = line.split('(', 1)[0].strip()
                    # Remove PID prefix
                    parts = syscall.split()
                    if len(parts) > 1:
                        syscall = parts[-1]
                    syscalls[syscall] = syscalls.get(syscall, 0) + 1

        return {
            'total_syscalls': sum(syscalls.values()),
            'unique_syscalls': len(syscalls),
            'top_syscalls': dict(sorted(syscalls.items(),
                                        key=lambda x: x[1], reverse=True)[:20]),
        }

    def save_results(self) -> Path:
        """Save all sandbox results to JSON."""
        self.results['syscall_analysis'] = self.parse_strace()

        # List all output files
        artifacts = {}
        for f in self.work_dir.iterdir():
            if f.is_file():
                artifacts[f.name] = f.stat().st_size
        self.results['artifacts'] = artifacts

        output = self.sample_path.with_suffix('.sandbox.json')
        output.write_text(json.dumps(self.results, indent=2, default=str))
        print(f"[SANDBOX] Results: {output}")
        return output

    def cleanup(self):
        """Remove temporary work directory."""
        import shutil
        shutil.rmtree(self.work_dir, ignore_errors=True)

Stage 5: Correlation and Clustering

Group similar samples to identify malware families:

#!/usr/bin/env python3
"""correlate.py — Cluster similar samples"""

import json
from pathlib import Path
from collections import defaultdict

try:
    import ssdeep
except ImportError:
    ssdeep = None


class SampleCorrelator:
    def __init__(self, analysis_dir: Path):
        self.dir = Path(analysis_dir)
        self.samples = self._load_all()

    def _load_all(self) -> Dict:
        """Load all triage/sandbox/ghidra JSON results."""
        samples = {}
        for json_file in self.dir.rglob('*.triage.json'):
            try:
                data = json.loads(json_file.read_text())
                samples[json_file.stem.replace('.triage', '')] = data
            except:
                pass
        return samples

    def cluster_by_ssdeep(self, threshold: int = 80) -> Dict:
        """Cluster samples by ssdeep similarity (threshold 0-100)."""
        if not ssdeep:
            return {}

        clusters = []
        processed = set()

        for name_a, data_a in self.samples.items():
            if name_a in processed:
                continue

            ssdeep_a = data_a.get('ssdeep')
            if not ssdeep_a:
                continue

            cluster = {name_a}
            processed.add(name_a)

            for name_b, data_b in self.samples.items():
                if name_b in processed:
                    continue

                ssdeep_b = data_b.get('ssdeep')
                if not ssdeep_b:
                    continue

                try:
                    score = ssdeep.compare(ssdeep_a, ssdeep_b)
                    if score >= threshold:
                        cluster.add(name_b)
                        processed.add(name_b)
                except:
                    pass

            if len(cluster) > 1:
                clusters.append({
                    'members': list(cluster),
                    'size': len(cluster),
                    'representative': name_a,
                })

        return {'ssdeep_clusters': clusters}

    def cluster_by_imports(self, threshold: float = 0.7) -> Dict:
        """Cluster by Jaccard similarity of imported functions."""
        # Requires ghidra.json results
        ghidra_files = list(self.dir.rglob('*.ghidra.json'))
        if not ghidra_files:
            return {}

        imports_by_sample = {}
        for gf in ghidra_files:
            try:
                data = json.loads(gf.read_text())
                imports = {imp['name'] for imp in data.get('imports', [])}
                sample_name = gf.stem.replace('.ghidra', '')
                imports_by_sample[sample_name] = imports
            except:
                pass

        clusters = []
        processed = set()

        for name_a, imports_a in imports_by_sample.items():
            if name_a in processed or not imports_a:
                continue

            cluster = {name_a}
            processed.add(name_a)

            for name_b, imports_b in imports_by_sample.items():
                if name_b in processed or not imports_b:
                    continue

                intersection = imports_a & imports_b
                union = imports_a | imports_b
                if union:
                    jaccard = len(intersection) / len(union)
                    if jaccard >= threshold:
                        cluster.add(name_b)
                        processed.add(name_b)

            if len(cluster) > 1:
                clusters.append({
                    'members': list(cluster),
                    'size': len(cluster),
                    'common_imports': sorted(list(imports_a))[:20],
                })

        return {'import_clusters': clusters}

    def cluster_by_bucket(self) -> Dict:
        """Group by packer type and architecture."""
        buckets = defaultdict(list)

        for name, data in self.samples.items():
            packer = data.get('packer', {}).get('packer', 'unknown')
            arch = data.get('type', {}).get('arch', 'unknown')
            bucket_key = f"{arch}|{packer}"
            buckets[bucket_key].append(name)

        return {
            'buckets': {
                k: {'count': len(v), 'samples': v}
                for k, v in sorted(buckets.items())
            }
        }


if __name__ == '__main__':
    import sys
    corr = SampleCorrelator(Path(sys.argv[1]))
    results = {}
    results.update(corr.cluster_by_ssdeep())
    results.update(corr.cluster_by_imports())
    results.update(corr.cluster_by_bucket())
    print(json.dumps(results, indent=2))

#!/usr/bin/env python3
"""pipeline.py — Orchestrate the full analysis pipeline"""

import os
import json
import time
import signal
import threading
from pathlib import Path
from datetime import datetime

# Import stage modules
from ingest import watch_directory  # Stage 1
from triage import SampleTriage     # Stage 2
# Stage 3: ghidra_headless (subprocess)
# Stage 4: sandbox_runner (subprocess)
# Stage 5: correlate (background)

QUEUE_FILE = Path('/opt/samples/queue.jsonl')
PROCESSED_DIR = Path('/opt/samples/processed')
RESULTS_DIR  = Path('/opt/samples/results')
RESULTS_DIR.mkdir(parents=True, exist_ok=True)


class PipelineWorker:
    def __init__(self):
        self.running = True
        self.processed = set()

    def process_queue(self):
        """Process jobs from the queue file."""
        while self.running:
            if not QUEUE_FILE.exists():
                time.sleep(5)
                continue

            lines = QUEUE_FILE.read_text().strip().split('\n')
            for line in lines:
                if not line.strip():
                    continue

                try:
                    job = json.loads(line)
                    sample_path = Path(job['processed_path'])

                    if sample_path.name in self.processed:
                        continue

                    self.analyze_sample(sample_path, job)
                    self.processed.add(sample_path.name)

                except Exception as e:
                    print(f"[PIPELINE] Error processing job: {e}")

            time.sleep(5)

    def analyze_sample(self, sample_path: Path, meta: dict):
        """Run all analysis stages on one sample."""
        name = sample_path.name
        print(f"\n{'='*60}\n[PIPELINE] Analyzing: {name}\n{'='*60}")

        # Stage 2: Triage
        print(f"[2/4] Triage...")
        try:
            triage = SampleTriage(sample_path)
            triage_results = triage.run()
            triage_out = sample_path.with_suffix('.triage.json')
            triage_out.write_text(json.dumps(triage_results, indent=2))
        except Exception as e:
            print(f"  Triage failed: {e}")
            return

        # Skip deep analysis if benign
        if triage_results.get('verdict') == 'BENIGN':
            print(f"  Verdict: BENIGN — skipping deep analysis")
            return

        # Stage 3: Static Analysis (Ghidra)
        print(f"[3/4] Static analysis (Ghidra)...")
        try:
            from ghidra_headless import analyze_sample
            analyze_sample(sample_path, project_dir=RESULTS_DIR / 'ghidra_projects')
        except Exception as e:
            print(f"  Ghidra analysis failed: {e}")

        # Stage 4: Dynamic Analysis (Sandbox)
        print(f"[4/4] Dynamic analysis (Sandbox)...")
        try:
            from sandbox_runner import SandboxRunner
            runner = SandboxRunner(sample_path, timeout=60)
            runner.build_sandbox_image()
            runner.run_combined()
            runner.save_results()
            runner.cleanup()
        except Exception as e:
            print(f"  Sandbox analysis failed: {e}")

        print(f"[PIPELINE] Complete: {name}")

    def stop(self):
        self.running = False


def main():
    worker = PipelineWorker()

    # Start ingestion watcher in background
    ingest_thread = threading.Thread(target=watch_directory, daemon=True)
    ingest_thread.start()

    # Set up signal handlers
    def shutdown(sig, frame):
        worker.stop()
        print("\n[PIPELINE] Shutting down...")
    signal.signal(signal.SIGINT, shutdown)
    signal.signal(signal.SIGTERM, shutdown)

    # Start processing queue
    worker.process_queue()


if __name__ == '__main__':
    main()

YARA Rule Auto-Generation

When new malware families are discovered, generate YARA rules automatically:

#!/usr/bin/env python3
"""yara_gen.py — Auto-generate YARA rules from clustered samples"""

import json
import random
import hashlib
from pathlib import Path
from collections import Counter


def generate_yara_rule(cluster_name: str, samples: list,
                       analysis_dir: Path) -> str:
    """
    Generate a YARA rule that detects a malware cluster.
    Uses strings common to all samples but rare in benign files.
    """
    # Collect strings from all samples in the cluster
    cluster_strings = Counter()
    string_sources = {}

    for sample_name in samples:
        triage_path = analysis_dir / f"{sample_name}.triage.json"
        if not triage_path.exists():
            continue

        try:
            data = json.loads(triage_path.read_text())
            sample_strings = set(data.get('top_strings', []))
            for s in sample_strings:
                if len(s) >= 8 and len(s) <= 100:
                    cluster_strings[s] += 1
                    if s not in string_sources:
                        string_sources[s] = []
                    string_sources[s].append(sample_name)
        except:
            continue

    # Find strings present in most cluster members
    threshold = max(2, len(samples) // 2)
    candidate_strings = [
        s for s, count in cluster_strings.items()
        if count >= threshold
    ]

    if not candidate_strings:
        return None

    # Sort by string length (prefer longer for specificity)
    candidate_strings.sort(key=len, reverse=True)

    # Build YARA rule
    rule_id = hashlib.md5(cluster_name.encode()).hexdigest()[:8]
    escaped = candidate_strings[:15]  # Top 15 strings

    rule_parts = []
    rule_parts.append(f'rule malware_cluster_{rule_id} {{')
    rule_parts.append(f'    meta:')
    rule_parts.append(f'        description = "Auto-generated rule for {cluster_name}"')
    rule_parts.append(f'        date = "2026-06-26"')
    rule_parts.append(f'        hash = "{rule_id}"')
    rule_parts.append(f'    strings:')

    for i, s in enumerate(escaped):
        # Escape YARA-special characters in strings
        safe_s = s.replace('\\', '\\\\').replace('"', '\\"')
        rule_parts.append(f'        $s{i} = "{safe_s}"')

    # Require at least 3 of the strings (or all, if fewer)
    names = [f'$s{i}' for i in range(len(escaped))]
    required = min(3, len(names))
    rule_parts.append(f'    condition:')
    rule_parts.append(f'        {required} of ({", ".join(names)})')
    rule_parts.append(f'}}')

    return '\n'.join(rule_parts)


if __name__ == '__main__':
    import sys
    cluster_name = sys.argv[1]
    samples = sys.argv[2:]
    analysis_dir = Path(sys.argv[3]) if len(sys.argv) > 3 else Path('/opt/samples/processed')

    rule = generate_yara_rule(cluster_name, samples, analysis_dir)
    if rule:
        print(rule)
        output = Path(f'{cluster_name}.yar')
        output.write_text(rule)
        print(f'Rule written to {output}')
    else:
        print('Could not generate rule — insufficient common strings')

Challenge

Reimplement the ingestion stage in Go. Watch a directory for new files, compute each file's SHA-256, and append a job record to a queue file. Drop a sample into the watched directory and confirm that a queued job appears; the remaining stages read that queue.

package main

import (
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"time"
)

type Job struct {
	Name     string `json:"name"`
	SHA256   string `json:"sha256"`
	Size     int64  `json:"size"`
	Ingested string `json:"ingested_at"`
}

func hashFile(path string) (string, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return "", err
	}
	sum := sha256.Sum256(data)
	return hex.EncodeToString(sum[:]), nil
}

func main() {
	if len(os.Args) != 3 {
		fmt.Fprintln(os.Stderr, "usage: ingest <incoming-dir> <queue.jsonl>")
		os.Exit(1)
	}
	incoming, queue := os.Args[1], os.Args[2]
	seen := map[string]bool{}

	fmt.Printf("[*] watching %s\n", incoming)
	for {
		entries, err := os.ReadDir(incoming)
		if err != nil {
			fmt.Fprintln(os.Stderr, err)
			os.Exit(1)
		}
		for _, e := range entries {
			if e.IsDir() || seen[e.Name()] {
				continue
			}
			seen[e.Name()] = true
			info, _ := e.Info()
			sum, err := hashFile(filepath.Join(incoming, e.Name()))
			if err != nil {
				fmt.Fprintf(os.Stderr, "[-] %s: %v\n", e.Name(), err)
				continue
			}
			job := Job{
				Name:     e.Name(),
				SHA256:   sum,
				Size:     info.Size(),
				Ingested: time.Now().UTC().Format(time.RFC3339),
			}
			line, _ := json.Marshal(job)
			f, err := os.OpenFile(queue, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
			if err != nil {
				fmt.Fprintln(os.Stderr, err)
				continue
			}
			f.Write(append(line, '\n'))
			f.Close()
			fmt.Printf("[INGEST] %s sha256=%s\n", job.Name, job.SHA256[:16])
		}
		time.Sleep(2 * time.Second)
	}
}
