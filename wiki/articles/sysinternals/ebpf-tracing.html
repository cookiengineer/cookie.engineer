eBPF Tracing

eBPF -based kernel and user-space tracing with kprobes, uprobes, tracepoints, raw tracepoints, ring buffers, and bpftrace one-liners.

Overview

The Linux kernel exposes thousands of trace points. Every function entry, system call, scheduler event, network packet arrival, and memory allocation can be observed in flight. eBPF tracing provides safe, low-overhead access to all of these without patching kernel code, without kernel modules, and with guarantees that your instrumentation cannot crash the kernel. This article covers the tracing-specific mechanisms: kprobes, uprobes, tracepoints, raw tracepoints, ring buffers, and the bpftrace frontend.

Required Reading

  • eBPF - eBPF architecture, verifier, maps, helpers, BTF
  • Kernel Observability - kernel tracing infrastructure, /proc, /sys, debugfs
  • Processes - process lifecycle, /proc interface
  • Filesystems - VFS layer, inodes, file operations
  • ELF - ELF symbols, sections, relocations (uprobes rely on symbol resolution)

2. Kprobes

Kprobes allow attaching eBPF programs to any kernel function, at entry (kprobe) or return (kretprobe). They work by dynamically replacing the first instruction(s) of the target function with a breakpoint ( int3 on x86).

2.1 How Kprobes Work

Before instrumentation:
 ┌──────────────────┐
 │ __x64_sys_execve:│
 │    push rbp      │   ← first instruction
 │    mov rbp, rsp  │
 │    ...           │
 └──────────────────┘

After kprobe registered:
 ┌──────────────────┐
 │ __x64_sys_execve:│
 │    int3          │   ← breakpoint replaces push rbp
 │    mov rbp, rsp  │
 │    ...           │
 └──────────────────┘

When breakpoint hits:
   1. CPU traps to kernel (ring 0)
   2. kprobe handler saves registers → struct pt_regs
   3. Calls pre_handler (our BPF program)
   4. Single-steps original instruction (push rbp)
   5. Calls post_handler (if any)
   6. Returns to normal kernel execution

For kretprobes, the return address on the stack is replaced with a trampoline address. When the function returns, the trampoline fires the BPF program, then jumps to the original return address.

 ┌───────────────────────┐
 │ Function entry         │
 │ (kprobe fires here)    │ ◄── BPF program invoked
 └──────────+────────────┘
            │
            ▼
 ┌───────────────────────┐
 │ Function body          │
 └──────────+────────────┘
            │
            ▼
 ┌───────────────────────┐
 │ Function return        │
 │ (kretprobe fires here) │ ◄── BPF program invoked
 └───────+───────────────┘
         │
         ▼
    Original return address

2.2 Kprobe eBPF Program Structure

// trace_execve.bpf.c — trace execve calls with kprobe
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>

// Map to pass event data to user space
struct {
    __uint(type, BPF_MAP_TYPE_RINGBUF);
    __uint(max_entries, 256 * 1024);  // 256 KB
} events SEC(".maps");

// Event data structure
struct exec_event {
    u32 pid;
    u32 uid;
    char comm[16];
    char filename[256];
    u64 timestamp;
};

SEC("kprobe/__x64_sys_execve")
int kprobe__execve(struct pt_regs *ctx) {
    struct exec_event *event;

    // Reserve space in the ring buffer
    event = bpf_ringbuf_reserve(&events, sizeof(*event), 0);
    if (!event)
        return 0;

    // Fill in event data
    event->pid = bpf_get_current_pid_tgid() >> 32;
    event->uid = bpf_get_current_uid_gid();
    event->timestamp = bpf_ktime_get_ns();
    bpf_get_current_comm(&event->comm, sizeof(event->comm));

    // Read the first argument (filename pointer) from pt_regs
    // x86-64: RDI = arg0 = filename
    char *filename = (char *)PT_REGS_PARM1(ctx);
    bpf_probe_read_user_str(&event->filename, sizeof(event->filename), filename);

    // Submit to ring buffer
    bpf_ringbuf_submit(event, 0);

    return 0;
}

SEC("kretprobe/__x64_sys_execve")
int kretprobe__execve(struct pt_regs *ctx) {
    int ret = PT_REGS_RC(ctx);
    // ret = return value (fd for open, pid for fork, 0 for execve success)
    bpf_printk("execve returned: %d", ret);
    return 0;
}

3. Uprobes - User-Space Function Probing

Uprobes attach to user-space functions the same way kprobes attach to kernel functions. The key difference: the target is identified by a binary path and an offset (or symbol).

3.1 Uprobe Identification

uprobe target = file_path + inode:offset
              or
              file_path + symbol_name

Examples:
  /bin/bash:0x4a2b0        — offset in binary
  /bin/bash:readline       — function name
  /usr/lib/libc.so.6:malloc — library function

The kernel converts the symbol to an offset by reading the ELF file and resolving the symbol table. For PIE binaries, offsets are relative; the kernel adjusts at attach time.

 ┌───────────────────────────────────────┐
 │  /usr/lib/libc.so.6                    │
 │                                        │
 │  .dynsym: malloc     → 0x8a3b0        │
 │  .symtab:  __libc_malloc → 0x8a340    │
 │                                        │
 │  uprobe:/usr/lib/libc.so.6:malloc      │
 │    → kernel resolves to inode:0x8a3b0  │
 │                                        │
 │  When any process executes malloc:     │
 │    1. int3 at 0x8a3b0 in that process  │
 │    2. BPF program fires               │
 │    3. Original instruction single-step │
 │    4. Execution resumes                │
 └───────────────────────────────────────┘

3.2 Uprobe eBPF Program

// trace_malloc.bpf.c — trace malloc calls
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>

struct malloc_event {
    u32 pid;
    u64 size;
    u64 ptr;
    char comm[16];
};

struct {
    __uint(type, BPF_MAP_TYPE_RINGBUF);
    __uint(max_entries, 256 * 1024);
} events SEC(".maps");

SEC("uprobe//usr/lib/libc.so.6:malloc")
int uprobe_malloc(struct pt_regs *ctx) {
    struct malloc_event *event;

    event = bpf_ringbuf_reserve(&events, sizeof(*event), 0);
    if (!event)
        return 0;

    event->pid = bpf_get_current_pid_tgid() >> 32;
    event->size = PT_REGS_PARM1(ctx);  // first arg = size
    bpf_get_current_comm(&event->comm, sizeof(event->comm));

    bpf_ringbuf_submit(event, 0);
    return 0;
}

SEC("uretprobe//usr/lib/libc.so.6:malloc")
int uretprobe_malloc(struct pt_regs *ctx) {
    // Return value in PT_REGS_RC(ctx) = allocated pointer
    u64 ptr = PT_REGS_RC(ctx);

    // We can correlate size (from entry) with returned pointer
    // using a hash map keyed by pid/tid
    bpf_printk("malloc returned: %llx", ptr);
    return 0;
}

4. Tracepoints

Tracepoints are static, declared instrumentation points embedded in kernel source. Unlike kprobes, tracepoints are stable across kernel versions - they are part of the kernel's API contract.

4.1 Tracepoint Definition

// include/trace/events/sched.h
TRACE_EVENT(sched_process_exec,

    TP_PROTO(struct task_struct *p, pid_t old_pid,
             struct linux_binprm *bprm),

    TP_ARGS(p, old_pid, bprm),

    TP_STRUCT__entry(
        __string(filename, bprm->filename)
        __field(pid_t, pid)
        __field(pid_t, old_pid)
    ),

    TP_fast_assign(
        __assign_str(filename, bprm->filename);
        __entry->pid = p->pid;
        __entry->old_pid = old_pid;
    ),

    TP_printk("filename=%s pid=%d old_pid=%d",
              __get_str(filename), __entry->pid, __entry->old_pid)
);

This macro expands into: 1. A tracepoint struct definition (with fields for the event) 1. A fast assignment function that copies fields into the ring buffer 1. An eventfs interface entry 1. A format file that describes the binary layout

4.3 Tracepoint eBPF Program

// trace_openat.bpf.c — trace file opens via tracepoint
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>

struct openat_args {
    unsigned short common_type;
    unsigned char common_flags;
    unsigned char common_preempt_count;
    int common_pid;
    int __syscall_nr;
    int dfd;
    const char *filename;
    int flags;
    umode_t mode;
};

struct {
    __uint(type, BPF_MAP_TYPE_HASH);
    __uint(max_entries, 10240);
    __type(key, u32);    // pid
    __type(value, u64);  // count
} open_counter SEC(".maps");

SEC("tracepoint/syscalls/sys_enter_openat")
int trace_openat(struct openat_args *ctx) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;
    u64 *count = bpf_map_lookup_elem(&open_counter, &pid);
    u64 zero = 0;

    if (count) {
        (*count)++;
    } else {
        bpf_map_update_elem(&open_counter, &pid, &zero, BPF_NOEXIST);
    }

    // Print filename (readable via /sys/kernel/debug/tracing/trace_pipe)
    char fname[256];
    bpf_probe_read_user_str(&fname, sizeof(fname), ctx->filename);
    bpf_printk("PID %d openat: %s", pid, fname);

    return 0;
}

5. Raw Tracepoints

Raw tracepoints bypass the tracepoint argument marshalling overhead. Instead of writing arguments into the ring buffer format (a costly operation), raw tracepoints pass a pointer to the raw TP_PROTO arguments directly to the BPF program.

Standard tracepoint:
  Kernel calls trace_<name>()
    → copies args to ring buffer (TP_fast_assign)
    → BPF program reads ctx->field1, ctx->field2
    → ctx is the formatted ring buffer entry

Raw tracepoint:
  Kernel calls trace_<name>()
    → directly passes TP_PROTO args pointer to BPF program
    → BPF program casts ctx to the actual argument struct
    → No intermediate ring buffer, no copy

Performance difference:

Standard tracepoint: ~200-400 ns per event (arg copy overhead)
Raw tracepoint:      ~50-100 ns per event  (direct access)
						
// trace_sched_exec_raw.bpf.c — raw tracepoint version
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>

// Must match the TP_PROTO declaration exactly
struct sched_exec_args {
    u64 __do_not_use;  // padding for common fields
    struct task_struct *task;
    pid_t old_pid;
    struct linux_binprm *bprm;
};

SEC("raw_tracepoint/sched_process_exec")
int raw_trace_exec(struct bpf_raw_tracepoint_args *ctx) {
    struct sched_exec_args *args = (struct sched_exec_args *)ctx->args;
    struct task_struct *task = args->task;
    pid_t pid;

    bpf_probe_read_kernel(&pid, sizeof(pid), &task->pid);
    bpf_printk("sched_process_exec: pid=%d", pid);
    return 0;
}

6. Perf Events in BPF

BPF programs can interact with perf events for both sampling and output.

6.1 bpf_perf_event_output

Sends data to a user-space perf ring buffer (legacy, prefer BPF_MAP_TYPE_RINGBUF ):

struct {
    __uint(type, BPF_MAP_TYPE_PERF_EVENT_ARRAY);
    __uint(key_size, sizeof(int));
    __uint(value_size, sizeof(int));
} perf_map SEC(".maps");

// In BPF program:
struct event_data ev = { ... };
bpf_perf_event_output(ctx, &perf_map, BPF_F_CURRENT_CPU,
                       &ev, sizeof(ev));

6.2 bpf_perf_event_read

Reads the current value of a perf event counter (e.g., CPU cycles, cache misses):

// Read the perf event that triggered this BPF program
u64 count = bpf_perf_event_read(&perf_map, index);

// Read the perf event on a specific CPU
u64 cpu_count = bpf_perf_event_read(&perf_map, cpu);

7. Ring Buffers in Depth

The BPF_MAP_TYPE_RINGBUF map type is the modern data path from BPF to user space, replacing BPF_MAP_TYPE_PERF_EVENT_ARRAY for most use cases.

7.1 Why Ring Buffer?

Feature Perf Event Array Ring Buffer
Lock-free Yes (per-CPU) Yes
Variable-length records No (fixed-size entries) Yes
Memory efficiency Wastes unused entry space Compact
Multi-producer Per-CPU only Any CPU
API bpf_perf_event_output reserve+submit/discard
Wakeup on new data epoll only epoll + callback
User-space API libbpf perf_buffer ring_buffer or custom mmap

7.2 User-Space Consumer with Ring Buffer

// ringbuf_consumer.c — read events from BPF ring buffer
#include <bpf/libbpf.h>
#include <bpf/bpf.h>
#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <unistd.h>

static volatile sig_atomic_t running = 1;
void sig_handler(int sig) { running = 0; }

struct exec_event {
    u32 pid;
    u32 uid;
    char comm[16];
    char filename[256];
    u64 timestamp;
};

// Callback called for each event
static int handle_event(void *ctx, void *data, size_t data_sz) {
    struct exec_event *ev = (struct exec_event *)data;
    printf("EXEC pid=%d uid=%d comm=%s file=%s\n",
           ev->pid, ev->uid, ev->comm, ev->filename);
    return 0;
}

int main(int argc, char **argv) {
    struct bpf_object *obj = NULL;
    struct ring_buffer *rb = NULL;
    struct bpf_map *ringbuf_map;
    int err;

    // Open and load BPF object
    obj = bpf_object__open_file(argv[1], NULL);
    if (libbpf_get_error(obj)) {
        fprintf(stderr, "Failed to open BPF object\n");
        return 1;
    }
    err = bpf_object__load(obj);
    if (err) {
        fprintf(stderr, "Failed to load: %d\n", err);
        return 1;
    }

    // Attach all programs
    struct bpf_program *prog;
    bpf_object__for_each_program(prog, obj) {
        struct bpf_link *link = bpf_program__attach(prog);
        if (libbpf_get_error(link))
            fprintf(stderr, "Failed to attach %s\n", bpf_program__name(prog));
    }

    // Set up ring buffer consumer
    ringbuf_map = bpf_object__find_map_by_name(obj, "events");
    if (!ringbuf_map) {
        fprintf(stderr, "Map 'events' not found\n");
        return 1;
    }

    rb = ring_buffer__new(bpf_map__fd(ringbuf_map), handle_event, NULL, NULL);
    if (!rb) {
        fprintf(stderr, "Failed to create ring buffer\n");
        return 1;
    }

    signal(SIGINT, sig_handler);
    signal(SIGTERM, sig_handler);

    printf("Listening for events. Ctrl-C to exit.\n");
    while (running) {
        err = ring_buffer__poll(rb, 100 /* timeout ms */);
        if (err == -EINTR) break;
        if (err < 0) {
            fprintf(stderr, "Error polling ring buffer: %d\n", err);
            break;
        }
    }

    ring_buffer__free(rb);
    bpf_object__close(obj);
    return 0;
}

8. Tracing by Subsystem

8.1 File System Tracing

# Trace all file opens with bpftrace (one-liner)
bpftrace -e 'tracepoint:syscalls:sys_enter_openat { printf("%s %s\n", comm, str(args->filename)); }'

# Trace VFS operations directly (lower overhead than tracepoints)
bpftrace -e 'kprobe:vfs_open { printf("%s opened %s\n", comm, str(((struct path *)arg1)->dentry->d_name.name)); }'

bpftrace -e 'kprobe:vfs_read { printf("%s read %d bytes\n", comm, arg2); }'

bpftrace -e 'kprobe:vfs_write { printf("%s wrote %d bytes\n", comm, arg2); }'

# Count file operations by process
bpftrace -e '
    kprobe:vfs_read  { @reads[comm]  = count(); }
    kprobe:vfs_write { @writes[comm] = count(); }
    interval:s:5     { print(@reads, @writes); clear(@reads); clear(@writes); }'

# Trace which files are most opened
bpftrace -e '
    tracepoint:syscalls:sys_enter_openat {
        @files[str(args->filename)] = count();
    }
    END { print(@files); }'

# Trace file descriptor leaks (open without close)
bpftrace -e '
    tracepoint:syscalls:sys_enter_openat  { @open[pid, args->filename] = 1; }
    tracepoint:syscalls:sys_enter_close   { delete(@open[pid]); }
    END {
        printf("Unclosed files:\n");
        print(@open);
    }'
						
// vfs_stats.bpf.c — count VFS operations by type
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>

struct {
    __uint(type, BPF_MAP_TYPE_HASH);
    __uint(max_entries, 256);
    __type(key, u32);    // op type (0=open, 1=read, 2=write)
    __type(value, u64);  // count
} vfs_stats SEC(".maps");

SEC("kprobe/vfs_open")
int kp_vfs_open(struct pt_regs *ctx) {
    u32 key = 0;
    u64 *val = bpf_map_lookup_elem(&vfs_stats, &key);
    if (val) (*val)++;
    return 0;
}

SEC("kprobe/vfs_read")
int kp_vfs_read(struct pt_regs *ctx) {
    u32 key = 1;
    u64 *val = bpf_map_lookup_elem(&vfs_stats, &key);
    if (val) (*val)++;
    return 0;
}

SEC("kprobe/vfs_write")
int kp_vfs_write(struct pt_regs *ctx) {
    u32 key = 2;
    u64 *val = bpf_map_lookup_elem(&vfs_stats, &key);
    if (val) (*val)++;
    return 0;
}

// proc_lifecycle.bpf.c — track process lifecycle
#include <linux/bpf.h>
#include <bpf/bpf_helpers.h>
#include <bpf/bpf_tracing.h>

struct {
    __uint(type, BPF_MAP_TYPE_HASH);
    __uint(max_entries, 10240);
    __type(key, u32);   // pid
    __type(value, u64); // start_time_ns
} proc_start SEC(".maps");

struct {
    __uint(type, BPF_MAP_TYPE_RINGBUF);
    __uint(max_entries, 256 * 1024);
} events SEC(".maps");

struct proc_event {
    u32 pid;
    char comm[16];
    u64 duration_ns;   // 0 for exec, non-zero for exit
    u8 is_exit;
    u32 exit_code;
};

SEC("tracepoint/sched/sched_process_exec")
int proc_exec(void *ctx) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;
    u64 ts = bpf_ktime_get_ns();
    bpf_map_update_elem(&proc_start, &pid, &ts, BPF_ANY);

    struct proc_event *ev = bpf_ringbuf_reserve(&events, sizeof(*ev), 0);
    if (ev) {
        ev->pid = pid;
        ev->is_exit = 0;
        ev->duration_ns = 0;
        bpf_get_current_comm(&ev->comm, sizeof(ev->comm));
        bpf_ringbuf_submit(ev, 0);
    }
    return 0;
}

SEC("tracepoint/sched/sched_process_exit")
int proc_exit(void *ctx) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;
    u64 *start = bpf_map_lookup_elem(&proc_start, &pid);
    if (!start) return 0;

    u64 now = bpf_ktime_get_ns();
    struct proc_event *ev = bpf_ringbuf_reserve(&events, sizeof(*ev), 0);
    if (ev) {
        ev->pid = pid;
        ev->is_exit = 1;
        ev->duration_ns = now - *start;
        bpf_get_current_comm(&ev->comm, sizeof(ev->comm));
        bpf_ringbuf_submit(ev, 0);
    }

    bpf_map_delete_elem(&proc_start, &pid);
    return 0;
}

9. bpftrace

bpftrace is a high-level tracing language that compiles to eBPF . It is the spiritual successor to DTrace for Linux.

10. libbpf Skeletons

libbpf skeletons auto-generate C header files from BPF objects, containing struct definitions for maps and programs, as well as setup/teardown functions.

# Generate a skeleton from a BPF object file
bpftool gen skeleton counter.bpf.o > counter.skel.h

# The generated header includes:
# - struct counter_bpf { ... } with map pointers and program links
# - counter_bpf__open()    → opens the object
# - counter_bpf__load()    → loads programs and maps
# - counter_bpf__attach()  → attaches all programs
# - counter_bpf__destroy() → cleans up
						
// skeleton_loader.c — using a skeleton
#include "counter.skel.h"
#include <stdio.h>
#include <signal.h>

static volatile sig_atomic_t running = 1;
void sig_handler(int sig) { running = 0; }

int main() {
    struct counter_bpf *skel;
    int err;

    // 1. Open and load in one call
    skel = counter_bpf__open_and_load();
    if (!skel) {
        fprintf(stderr, "Failed to open and load skeleton\n");
        return 1;
    }

    // 2. Attach all programs
    err = counter_bpf__attach(skel);
    if (err) {
        fprintf(stderr, "Failed to attach: %d\n", err);
        counter_bpf__destroy(skel);
        return 1;
    }

    signal(SIGINT, sig_handler);

    // 3. Access maps directly via skeleton struct
    printf("Polling counter...\n");
    while (running) {
        u32 key = 0;
        u64 value;
        err = bpf_map__lookup_elem(skel->maps.counter,
                                    &key, sizeof(key),
                                    &value, sizeof(value), 0);
        if (!err)
            printf("\rcounter = %llu", (unsigned long long)value);
        sleep(1);
    }

    // 4. Cleanup
    counter_bpf__destroy(skel);
    return 0;
}
