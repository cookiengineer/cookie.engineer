Processes

A process is a running instance of a program, with its own memory, credentials, and lifecycle managed by the kernel.

Overview

Every command you run becomes a process. The kernel schedules it, gives it a virtual address space, tracks its open files and credentials, and exposes all of that through /proc . Understanding the process model is the foundation for debugging, privilege escalation, and container isolation. This article follows a process from the kernel's task_struct down to the /proc interface, then shows how to inspect and control processes with the standard tools.

What a Process Is

A process is a running instance of a program. The kernel represents every process with a task_struct (in /usr/src/linux/include/linux/sched.h ), which contains everything the kernel needs to schedule, pause, resume, and destroy the process.

A program is a dead file on disk. A process is that program brought to life - it has a memory address space, open file descriptors, a current working directory, credentials, signal handlers, and one or more threads. The kernel schedules processes (not programs) on the CPU . 

Program (disk)          Process (memory)
=============           =================
ELF binary on disk  ->  execve() loads into memory
                        Kernel creates task_struct
                        Memory regions mapped
                        /proc/PID/ entries created
                        Added to scheduler runqueue

Process Memory Layout

Every process in Linux sees a flat, contiguous virtual address space. The layout in memory (from low to high addresses) on x86-64: 

0x0000000000000000  +----------------------+
                    |  Text (code)         |  Read-only, executable
                    |  .text section       |  Shared between processes
                    |  Program instructions|
                    +----------------------+
                    |  Read-Only Data      |  .rodata section
                    |  String literals     |
                    |  Jump tables         |
                    +----------------------+
                    |  Data (initialized)  |  .data section
                    |  Global/static vars  |  Read-write
                    |  int x = 42;         |
                    +----------------------+
                    |  BSS (uninitialized) |  .bss section
                    |  Global/static vars  |  Zeroed at startup
                    |  int y;              |  No space in binary
                    +----------------------+
                    |  Heap                |  Grows upward (sbrk/mmap)
                    |  malloc() region     |  Dynamically sized
                    |  ...                 |
                    |  ...                 |
                    +----------------------+
                    |  Memory-mapped       |  Shared libraries, mmap
                    |  Regions             |  libc.so, ld.so
                    +----------------------+
                    |  ...                 |
                    |  Stack               |  Grows downward
                    |  Local variables     |  Function call frames
                    |  Return addresses    |
0x00007fffffffffff  +----------------------+  (approximate on x86-64)

/proc/PID Filesystem Deep Dive

/proc is not a real filesystem - it's a virtual interface to kernel data structures. Every directory named by a PID exposes the kernel's internal representation of that process. 

# Start a dummy process to inspect
sleep 3600 &
PID=$!
echo "Inspecting PID: $PID"

# -- cmdline: Complete command line (null-separated) --
cat /proc/$PID/cmdline | tr '\0' ' '
# Output: sleep 3600

# -- environ: Environment variables at process start --
cat /proc/$PID/environ | tr '\0' '\n' | head -5

# -- exe: Symlink to the executable binary --
readlink /proc/$PID/exe
# Output: /usr/bin/sleep

# -- cwd: Symlink to current working directory --
readlink /proc/$PID/cwd
# Output: /home/user

# -- root: Symlink to process root (chroot jail) --
readlink /proc/$PID/root
# Output: /  (normally)

# -- status: Human-readable process status --
cat /proc/$PID/status

# -- fd/: Directory of open file descriptors --
ls -la /proc/$PID/fd/
# Output:
# lrwx------ 1 user user 64 ... 0 -> /dev/pts/0
# lrwx------ 1 user user 64 ... 1 -> /dev/pts/0
# lrwx------ 1 user user 64 ... 2 -> /dev/pts/0

# -- maps: Virtual memory map --
cat /proc/$PID/maps

# -- smaps: Maps with detailed memory accounting --
cat /proc/$PID/smaps | head -40

kill $PID

Process States

The kernel tracks every process with a state field in the task_struct : 

        fork()
          |
    +-----v------+
    |            |   CPU assigned
    |   READY    |------------------+
    | (waiting   |                  |
    |  for CPU)  |<---------------+ |
    |            |   timeslice     | |
    +-----+------+   expired       | |
          |                        | |
          | scheduled +------------v-v--+
          |  on CPU   |                 |
          +---------->|    RUNNING      |
                      |  (on CPU)       |
                      +-+-----+------+--+
                        |     |      |
        I/O wait,       |     |      |  Signal (SIGSTOP)
        event wait      |     |      |  or ptrace attach
                        |     |      |
             +----------v-+    |   +--v-------+
             |  SLEEPING  |    |   | STOPPED  |
             | (S or D)   |    |   |   (T)    |
             +------------+    |   +----------+
                               |
                               |  exit() / signal
                          +----v------+
                          |  ZOMBIE   |
                          |   (Z)     |
                          +-----------+
State Letter Description
Running R Currently on CPU, or runnable in scheduler queue
Sleeping (interruptible) S Waiting for an event; can be woken by signals
Sleeping (uninterruptible) D Waiting for I/O; cannot be interrupted even by SIGKILL
Stopped T Stopped by SIGSTOP, SIGTSTP, or ptrace
Tracing stop t Stopped by ptrace (debugger)
Zombie Z Exited but parent has not called wait()
Dead X Process fully removed (transient, rarely visible)

Uninterruptible Sleep in D State

Processes enter D state when waiting for completion of a blocking I/O operation inside the kernel. While in D state, the process ignores all signals - including SIGKILL (9). This is why a kill -9 against a process stuck on a dead NFS mount does nothing. 

# Find processes in D state (stuck on I/O)
ps aux | awk '$8 ~ /D/'

# Or with /proc
for pid in /proc/[0-9]*; do
    state=$(awk '/^State:/ {print $2}' $pid/status 2>/dev/null)
    if [ "$state" = "D" ]; then
        echo "PID $(basename $pid) is in D state: $(cat $pid/comm 2>/dev/null)"
    fi
done

fork() - Creating a Process

fork() creates a new process on Linux. It is implemented on top of the clone() syscall and returns twice: once in the parent with the child's PID, and once in the child with 0 . The child is an exact copy of the calling process - same memory, file descriptors, and execution state. Thanks to Copy-on-Write ( CoW ), the kernel does not actually duplicate physical memory; it maps the child's page tables to the parent's pages as read-only. Only when either process writes to a page does the kernel make a private copy. 

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();

    if (pid == -1) {
        perror("fork");
        return 1;
    }

    if (pid == 0) {
        /* Child process - pid is 0 */
        printf("[CHILD]  PID: %d, PPID: %d\n", getpid(), getppid());
        printf("[CHILD]  fork() returned: %d\n", pid);
        _exit(0);  // Use _exit(), not exit() - avoids flushing parent's buffers twice
    } else {
        /* Parent process - pid is the child's PID */
        printf("[PARENT] PID: %d, Child PID: %d\n", getpid(), pid);
        printf("[PARENT] fork() returned: %d\n", pid);
        wait(NULL);  // Reap the child
    }
    return 0;
}

execve() - Replacing the Image

execve() replaces the entire process image with a new program. The PID stays the same. Open file descriptors (unless marked O_CLOEXEC ) survive the exec. Signal handlers are reset to defaults. The new program starts at _start . 

#include <stdio.h>
#include <unistd.h>

int main() {
    printf("[BEFORE] PID: %d\n", getpid());
    printf("[BEFORE] This process is about to be replaced\n");

    /* execve(path, argv, envp) */
    char *args[] = { "/usr/bin/id", NULL };
    char *env[]  = { NULL };

    execve("/usr/bin/id", args, env);

    /* If execve returns, it failed */
    perror("execve");
    return 1;
}
						
gcc -o exec-demo exec-demo.c && ./exec-demo
# The PID printed before and the one running /usr/bin/id are the same

wait() / waitpid() - Reaping Zombies

When a child process exits, it becomes a zombie. The kernel cannot free the process's task_struct until the parent acknowledges the exit by calling wait() (or waitpid() ). Zombies hold no memory, no CPU , no file descriptors - only a slot in the process table and the exit status. 

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <stdlib.h>

int main() {
    pid_t pid = fork();

    if (pid == 0) {
        printf("[CHILD] Exiting immediately\n");
        _exit(42);  // Exit status = 42
    }

    /* Parent */
    printf("[PARENT] Child PID: %d\n", pid);
    printf("[PARENT] Check with: ps aux | grep Z\n");
    sleep(5);  // Child is a zombie during this sleep

    int status;
    pid_t waited = wait(&status);

    if (WIFEXITED(status))
        printf("[PARENT] Child %d exited with status %d\n", waited, WEXITSTATUS(status));
    if (WIFSIGNALED(status))
        printf("[PARENT] Child %d killed by signal %d\n", waited, WTERMSIG(status));

    return 0;
}
						
gcc -o zombie-demo zombie-demo.c && ./zombie-demo
# In another terminal during the sleep:
ps aux | grep 'Z'

waitpid() Options

Flag Effect
WNOHANG Return immediately if no child has exited (non-blocking)
WUNTRACED Also return if child is stopped
WCONTINUED Also return if stopped child continues via SIGCONT
0 Block until any child changes state

Orphan processes: When a parent exits before its children, the kernel reparents the orphans to PID 1 (init/systemd). Init periodically calls wait() to reap them, preventing permanent zombies.

Process Credentials

Every process carries multiple user and group IDs. They control which files can be read/written/executed and which syscalls are permitted. 

+--------------------------------------------------+
|                  Process                          |
|                                                   |
|  Real UID (RUID):      1000  <- original user     |
|  Effective UID (EUID): 1000  <- used for checks   |
|  Saved UID (SUID):     1000  <- setuid saves here |
|  File system UID:      1000  <- for file access   |
|                                                   |
|  Real GID (RGID):      1000                       |
|  Effective GID (EGID): 1000                       |
|  Saved GID (SGID):     1000                       |
|  File system GID:      1000                       |
|                                                   |
|  Supplementary Groups: [10, 100, 1000]            |
|  Capabilities:         0x0000000000000000         |
+--------------------------------------------------+

When a setuid binary (permissions 4755 , owned by root) is executed:

  • RUID stays 1000 (the real user)
  • EUID becomes 0 (root, the file owner)
  • SUID becomes 0 (saved from previous EUID)

A process can drop privileges temporarily by swapping EUID with RUID, and later restore them via the SUID. 

#include <stdio.h>
#include <unistd.h>
#include <sys/types.h>

int main() {
    uid_t ruid, euid, suid;
    getresuid(&ruid, &euid, &suid);
    printf("RUID=%d  EUID=%d  SUID=%d\n", ruid, euid, suid);

    gid_t rgid, egid, sgid;
    getresgid(&rgid, &egid, &sgid);
    printf("RGID=%d  EGID=%d  SGID=%d\n", rgid, egid, sgid);
    return 0;
}

Verify credentials for any process:

cat /proc/$$/status | grep -E "^Uid|^Gid|^Groups"
# Uid:    1000    1000    1000    1000
# Gid:    1000    1000    1000    1000
# Format: Real  Effective  Saved  Filesystem

nice() and Priority

Linux uses dynamic priority scheduling. Every process has:

  • nice value (-20 to +19): User-visible. Lower = higher priority. Default is 0.
  • static priority (100-139): Kernel-internal = nice + 120.
  • dynamic priority : Adjusted by the scheduler based on sleep vs. CPU time.
# Run a command with specific niceness
nice -n 10 gzip largefile.dat

# Change niceness of running process
renice -n -5 -p 1234

# View nice values in ps/top
ps -eo pid,ni,comm | head -10
top  # Press 'r' to renice a process interactively

|  |                           |         |
|  |  PID 500 (bash)           |         |
|  |  Session leader           |         |

|  +--------------------------+         |
|                                       |
|  +--------------------------+         |

Job control (Ctrl+Z, fg, bg) works through process groups. A terminal's foreground process group receives keyboard signals (SIGINT, SIGTSTP ). Background groups can write to the terminal but reading causes SIGTTIN. 

# View process group and session info
ps -eo pid,ppid,pgid,sid,comm | head -10

# Column meanings:
# PID   = Process ID
# PPID  = Parent Process ID
# PGID  = Process Group ID (often same as PID of group leader)
# SID   = Session ID (often same as PID of session leader)

# Get the session leader of any process
cat /proc/1234/stat | awk '{print $6}'  # Field 6 is session ID

# Create a new session (daemon-like)
setsid bash              # New session, no controlling terminal
ps -eo pid,ppid,pgid,sid,comm | grep bash

Daemon Processes

A daemon is a background process with no controlling terminal. Traditional daemon initialization:

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <syslog.h>

void daemonize() {
    pid_t pid = fork();
    if (pid < 0)  exit(EXIT_FAILURE);
    if (pid > 0)  _exit(EXIT_SUCCESS);  // Parent exits

    // Child becomes session leader, loses controlling terminal
    if (setsid() < 0) exit(EXIT_FAILURE);

    // Fork again - child can never reacquire a controlling terminal
    pid = fork();
    if (pid < 0)  exit(EXIT_FAILURE);
    if (pid > 0)  _exit(EXIT_SUCCESS);

    // Set file creation mask
    umask(0);

    // Move to root directory
    chdir("/");

    // Close stdio and redirect to /dev/null
    close(STDIN_FILENO);
    close(STDOUT_FILENO);
    close(STDERR_FILENO);
    open("/dev/null", O_RDONLY);   // fd 0
    open("/dev/null", O_WRONLY);   // fd 1
    open("/dev/null", O_WRONLY);   // fd 2

    // Optionally connect to syslog
    openlog("mydaemon", LOG_PID, LOG_DAEMON);
    syslog(LOG_NOTICE, "Daemon started.");
    closelog();
}

int main() {
    daemonize();

    while (1) {
        // Actual daemon work
        sleep(60);
    }
    return 0;
}

Modern Linux systems use systemd service units instead of forking daemons. systemd handles logging (via stdout/stderr to journald), restarts, resource limits, and socket activation.

systemd cgroups for Management

systemd organizes processes into control groups (cgroups v2). Every service gets its own cgroup under /sys/fs/cgroup/system.slice/ . 

# Find the cgroup of a systemd service
systemctl status sshd | grep CGroup
# Output: CGroup: /system.slice/sshd.service

# Inspect resource usage of a cgroup
systemctl show sshd | grep -i cgroup

# List processes in a cgroup
ls /sys/fs/cgroup/system.slice/sshd.service/cgroup.procs

# All processes in systemd units
systemd-cgls

# CPU/memory usage per cgroup
systemd-cgtop

Tools for Process Inspection

# ps - snapshot of running processes
ps aux                          # Every process, BSD style
ps -ef                          # Every process, Unix style
ps -eo pid,ppid,user,state,pri,ni,vsz,rss,comm  # Custom columns
ps axfo pid,ppid,stat,comm      # Process tree with forest output

# Column meanings in ps aux:
# USER   = Owner
# PID    = Process ID
# %CPU   = CPU usage (percentage of one core)
# %MEM   = RSS as percentage of total physical memory
# VSZ    = Virtual memory size (kB)
# RSS    = Resident Set Size - physical memory used (kB)
# TTY    = Controlling terminal (? = none)
# STAT   = Process state code (see table above) + extra flags:
#          < = high priority, N = low priority, L = pages locked
#          s = session leader, l = multi-threaded, + = foreground
# START  = Start time
# TIME   = Total CPU time consumed
# COMMAND = Command with arguments

# top/htop - interactive process monitors
# htop has better UI: tree view, scrollable, mouse support, kill with F9

# pgrep/pkill - find/signal by name
pgrep -l sshd                    # List SSH daemon PIDs
pgrep -u root                    # All processes owned by root
pgrep -f "python server"         # Match full command line
pkill -HUP sshd                  # Send SIGHUP to sshd

# strace - trace system calls
strace -p 1234                   # Attach to running process
strace -f gcc hello.c            # Follow forks
strace -c ls                     # Summary counts of syscalls
strace -e trace=file cat /etc/hosts  # Only file-related syscalls

# lsof - list open files
lsof -p 1234                     # All open files for PID
lsof -i :80                      # Processes using port 80
lsof /var/log/syslog             # Processes with this file open

# fuser - identify processes using a file/socket
fuser -v /usr/bin/bash           # Processes using bash
fuser -n tcp 80                  # Processes on TCP port 80

Verification Checklist

# 1. Verify process memory layout
cat /proc/self/maps | head -20

# 2. Verify fork CoW - child sees same data, writes diverge
cat /proc/$$/maps | wc -l       # Parent's map entries
bash -c 'cat /proc/$$/maps | wc -l'  # Child's (same count initially)

# 3. Find all processes in D state (uninterruptible sleep)
ps aux | awk '$8 ~ /D/'

# 4. Find all zombie processes
ps aux | awk '$8 ~ /Z/'

# 5. Verify process credentials
cat /proc/$$/status | grep -E "^Uid:|^Gid:|^Groups:"

# 6. Check the cgroup hierarchy
cat /proc/$$/cgroup

# 7. List all process groups in current session
ps -eo pid,ppid,pgid,sid,stat,comm --sid $$ | head -20

# 8. Trace file I/O of a command
strace -e trace=file cat /etc/passwd 2>&1 | head -20

Challenge

Write a Go program that reconstructs the process tree from /proc . For each PID, read /proc/<pid>/status and extract Name , State , PPid , Uid , and VmRSS . Build a parent-to-child map, then print the tree rooted at PID 1 (or any process whose parent is unknown) with two spaces of indentation per level. Compare your output with ps axfo pid,ppid,comm . 

package main

import (
	"bufio"
	"fmt"
	"os"
	"path/filepath"
	"sort"
	"strconv"
	"strings"
)

type Process struct {
	PID   int
	PPID  int
	Name  string
	State string
	UID   int
	RSS   int
}

func readStatus(pid int) (Process, error) {
	p := Process{PID: pid}

	file, err := os.Open(fmt.Sprintf("/proc/%d/status", pid))
	if err != nil {
		return p, err
	}
	defer file.Close()

	scanner := bufio.NewScanner(file)
	for scanner.Scan() {
		key, value, found := strings.Cut(scanner.Text(), ":")
		if found == false {
			continue
		}
		value = strings.TrimSpace(value)
		switch key {
		case "Name":
			p.Name = value
		case "State":
			p.State = value[:1]
		case "PPid":
			p.PPID, _ = strconv.Atoi(value)
		case "Uid":
			if fields := strings.Fields(value); len(fields) > 0 {
				p.UID, _ = strconv.Atoi(fields[0])
			}
		case "VmRSS":
			if fields := strings.Fields(value); len(fields) > 0 {
				p.RSS, _ = strconv.Atoi(fields[0])
			}
		}
	}

	return p, scanner.Err()
}

func printTree(p Process, children map[int][]Process, depth int) {
	fmt.Printf("%s%d %s [%s] ppid=%d uid=%d rss=%dkB\n",
		strings.Repeat("  ", depth), p.PID, p.Name, p.State, p.PPID, p.UID, p.RSS)

	for _, child := range children[p.PID] {
		printTree(child, children, depth+1)
	}
}

func main() {
	entries, _ := filepath.Glob("/proc/[0-9]*")
	byPID := make(map[int]Process)
	children := make(map[int][]Process)

	for _, entry := range entries {
		pid, err := strconv.Atoi(filepath.Base(entry))
		if err != nil {
			continue
		}

		p, err := readStatus(pid)
		if err != nil {
			continue
		}

		byPID[p.PID] = p
		children[p.PPID] = append(children[p.PPID], p)
	}

	roots := make([]Process, 0)
	for _, p := range byPID {
		if _, ok := byPID[p.PPID]; ok == false {
			roots = append(roots, p)
		}
	}
	sort.Slice(roots, func(a, b int) bool { return roots[a].PID < roots[b].PID })

	for _, root := range roots {
		printTree(root, children, 0)
	}
}

Save it as proctree.go and run: 

go run proctree.go | head -20;
