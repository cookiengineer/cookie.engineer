Threads

POSIX threads, synchronization primitives, race conditions, deadlocks, thread-local storage, and the NPTL implementation on Linux.

Overview

A thread is the smallest unit of execution the kernel schedules. A process is the resource container that owns the address space, file descriptors, signal handlers, and credentials. Threads share that container, so communication is cheap but every concurrent write to shared memory is a potential race. After reading you can use the pthreads library, recognize race conditions and deadlocks, and explain how Linux clone() implements threads as processes.

Threads vs Processes

A thread is the smallest unit of execution that the kernel schedules. A process is a resource container - it owns the address space, file descriptors, signal handlers, and credentials. Threads live inside processes.

pthreads API

Linux implements POSIX threads via NPTL (Native POSIX Thread Library). All pthread functions are in libpthread , which is integrated into libc on modern glibc. Compile with -pthread (not -lpthread - the flag enables preprocessor defines too). 

#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <unistd.h>

void* thread_func(void* arg) {
    int id = *(int*)arg;
    printf("[THREAD %d] PID: %d, TID: %lu\n", id, getpid(), pthread_self());
    printf("[THREAD %d] Argument: %d\n", id, *(int*)arg);
    return (void*)(intptr_t)(id * 100);  // Return value via pthread_join
}

int main() {
    pthread_t threads[4];
    int ids[4];
    int rc;

    for (int i = 0; i < 4; i++) {
        ids[i] = i + 1;
        rc = pthread_create(&threads[i], NULL, thread_func, &ids[i]);
        if (rc) {
            fprintf(stderr, "pthread_create: %s\n", strerror(rc));
            return 1;
        }
    }

    printf("[MAIN] Created %d threads\n", 4);

    for (int i = 0; i < 4; i++) {
        void* retval;
        rc = pthread_join(threads[i], &retval);
        if (rc == 0)
            printf("[MAIN] Thread %d returned: %ld\n", i, (long)retval);
    }

    return 0;
}
						
gcc -pthread -o thread-demo thread-demo.c && ./thread-demo

Thread Synchronization: Mutex

A mutex (mutual exclusion) protects a critical section - code that must execute atomically with respect to other threads. 

#include <stdio.h>
#include <pthread.h>

#define ITERATIONS 1000000

int counter = 0;
pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;

void* increment(void* arg) {
    for (int i = 0; i < ITERATIONS; i++) {
        pthread_mutex_lock(&lock);
        counter++;  // Critical section: read-modify-write
        pthread_mutex_unlock(&lock);
    }
    return NULL;
}

int main() {
    pthread_t t1, t2;

    pthread_create(&t1, NULL, increment, NULL);
    pthread_create(&t2, NULL, increment, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    printf("Counter: %d (expected: %d)\n", counter, ITERATIONS * 2);
    return 0;
}
						
gcc -pthread -o mutex-demo mutex-demo.c && ./mutex-demo
# Output: Counter: 2000000 (expected: 2000000)  ← correct

Mutex Types

Type Attribute Behavior
Normal (PTHREAD_MUTEX_NORMAL) Default Deadlock on relock; undefined on unlock from other thread
Error-check (PTHREAD_MUTEX_ERRORCHECK) Explicit Returns EDEADLK on relock; EPERM on wrong-thread unlock
Recursive (PTHREAD_MUTEX_RECURSIVE) Explicit Allows relock by same thread; count-based unlock
Default (PTHREAD_MUTEX_DEFAULT) Implementation Undefined behavior on relock/unlock mistakes 
pthread_mutex_t rlock;
pthread_mutexattr_t attr;

pthread_mutexattr_init(&attr);
pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_RECURSIVE);
pthread_mutex_init(&rlock, &attr);

// Now the same thread can lock multiple times:
pthread_mutex_lock(&rlock);
pthread_mutex_lock(&rlock);    // OK — same thread
pthread_mutex_unlock(&rlock);
pthread_mutex_unlock(&rlock);

Race Conditions

A race condition occurs when two threads access shared data concurrently and at least one access is a write. The result depends on the unpredictable interleaving of thread execution. 

// RACY: Remove the mutex from the counter example and observe:
int counter = 0;

void* increment_racy(void* arg) {
    for (int i = 0; i < ITERATIONS; i++) {
        counter++;  // NOT atomic! Three assembly instructions:
                    // mov  eax, [counter]     ← load
                    // inc  eax               ← modify
                    // mov  [counter], eax     ← store
    }
    return NULL;
}
						
Timeline of two threads racing on counter++:

Thread 1          Thread 2          counter (memory)
────────          ────────          ────────────────
mov eax, [mem]                      0  (T1 reads 0)
                   mov eax, [mem]   0  (T2 reads 0)
                   inc eax          T2 has 1
inc eax                              T1 has 1
mov [mem], eax                      1  (T1 stores 1)
                   mov [mem], eax   1  (T2 stores 1)
                                    
Expected: 2 increments → counter = 2
Actual:   counter = 1  (one increment lost)

Verify with the race:

gcc -pthread -o race-demo race-demo.c && for i in $(seq 1 10); do ./race-demo; done
# Counter will be below 2000000 every time

Condition Variables

Condition variables allow threads to sleep until a predicate becomes true. They are always paired with a mutex. 

#include <stdio.h>
#include <pthread.h>
#include <stdlib.h>

pthread_mutex_t mutex = PTHREAD_MUTEX_INITIALIZER;
pthread_cond_t  cond  = PTHREAD_COND_INITIALIZER;
int ready = 0;
int data  = 0;

void* producer(void* arg) {
    pthread_mutex_lock(&mutex);
    data = 42;
    ready = 1;
    printf("[PRODUCER] Data prepared: %d\n", data);
    pthread_cond_signal(&cond);        // Wake one waiter
    // pthread_cond_broadcast(&cond);  // Wake all waiters
    pthread_mutex_unlock(&mutex);
    return NULL;
}

void* consumer(void* arg) {
    pthread_mutex_lock(&mutex);
    while (!ready) {  // ALWAYS loop — protects against spurious wakeups
        pthread_cond_wait(&cond, &mutex);  // Atomically: unlock + sleep
    }                                       // Re-acquires mutex on wake
    printf("[CONSUMER] Received: %d\n", data);
    pthread_mutex_unlock(&mutex);
    return NULL;
}

int main() {
    pthread_t prod, cons;
    pthread_create(&cons, NULL, consumer, NULL);
    sleep(1);  // Ensure consumer is waiting first
    pthread_create(&prod, NULL, producer, NULL);

    pthread_join(prod, NULL);
    pthread_join(cons, NULL);
    return 0;
}
						
gcc -pthread -o condvar-demo condvar-demo.c && ./condvar-demo

Spurious Wakeups

POSIX allows spurious wakeups: pthread_cond_wait may return even though no thread signaled. The while (!ready) loop re-checks the predicate, ensuring correctness.

Read-Write Locks

A read-write lock (rwlock) allows multiple readers concurrently but gives writers exclusive access. This optimizes read-heavy workloads. 

#include <pthread.h>
#include <stdio.h>

pthread_rwlock_t rwlock = PTHREAD_RWLOCK_INITIALIZER;
int shared_data = 0;

void* reader(void* arg) {
    pthread_rwlock_rdlock(&rwlock);
    printf("[READER %ld] Data: %d\n", (long)arg, shared_data);
    usleep(100000);  // Simulate work
    pthread_rwlock_unlock(&rwlock);
    return NULL;
}

void* writer(void* arg) {
    pthread_rwlock_wrlock(&rwlock);
    shared_data++;
    printf("[WRITER] Data updated to: %d\n", shared_data);
    usleep(200000);  // Simulate work
    pthread_rwlock_unlock(&rwlock);
    return NULL;
}
Lock Type Concurrent Behavior
rdlock Multiple readers can hold simultaneously
wrlock Writer gets exclusive access; blocks all readers and writers
rdlock during wrlock Readers block until writer releases
wrlock during rdlock Writer blocks until all readers release
wrlock during wrlock Second writer blocks (FIFO or reader-preference depending on impl)

Spinlocks

A spinlock busy-waits (spins in a loop) instead of sleeping. The kernel uses spinlocks in contexts where sleeping is impossible (interrupt handlers, while holding other spinlocks). In user space, spinlocks are rarely justified - the kernel scheduler can't preempt a spinning thread until its timeslice expires. 

#include <pthread.h>

pthread_spinlock_t spin;
pthread_spin_init(&spin, PTHREAD_PROCESS_PRIVATE);

pthread_spin_lock(&spin);
// Critical section — very short, no blocking operations!
pthread_spin_unlock(&spin);

pthread_spin_destroy(&spin);

Deadlock

Deadlock occurs when threads wait on each other in a cycle. Four conditions must all hold (Coffman conditions):

  1. Mutual Exclusion : Resources cannot be shared.
  2. Hold and Wait : Threads hold resources while waiting for others.
  3. No Preemption : Resources cannot be forcibly taken.
  4. Circular Wait : A cycle exists in the resource dependency graph.
pthread_mutex_t lockA = PTHREAD_MUTEX_INITIALIZER;
pthread_mutex_t lockB = PTHREAD_MUTEX_INITIALIZER;

void* thread1(void* arg) {
    pthread_mutex_lock(&lockA);
    sleep(1);                     // Give time for deadlock to form
    pthread_mutex_lock(&lockB);   // Thread 1: holds A, needs B
    printf("[T1] Got both locks\n");
    pthread_mutex_unlock(&lockB);
    pthread_mutex_unlock(&lockA);
    return NULL;
}

void* thread2(void* arg) {
    pthread_mutex_lock(&lockB);
    sleep(1);                     // Give time for deadlock to form
    pthread_mutex_lock(&lockA);   // Thread 2: holds B, needs A
    printf("[T2] Got both locks\n");  // ← Never reached
    pthread_mutex_unlock(&lockA);
    pthread_mutex_unlock(&lockB);
    return NULL;
}

Deadlock avoidance breaks one of the four conditions: - Lock ordering : Always acquire locks in the same order (A then B everywhere). - Try-lock + backoff : pthread_mutex_trylock() returns EBUSY instead of blocking. - Lock hierarchies : Assign levels; only acquire higher-level locks while holding lower ones. - Timeouts : Use pthread_mutex_timedlock() with a deadline. 

// Trylock pattern — avoids deadlock
while (1) {
    pthread_mutex_lock(&lockA);
    if (pthread_mutex_trylock(&lockB) == 0) {
        break;  // Got both
    }
    pthread_mutex_unlock(&lockA);  // Release A, retry later
    usleep(1000);                  // Backoff
}

Thread-Local Storage

Each thread can have private data that doesn't require locking because no other thread can access it. 

#include <stdio.h>
#include <pthread.h>

// Method 1: __thread keyword (GCC/Clang extension)
__thread int tls_counter = 0;

// Method 2: pthread_getspecific / pthread_setspecific
pthread_key_t key;

void destructor(void* value) {
    printf("[CLEANUP] Freeing TLS value for thread %lu\n", pthread_self());
    free(value);
}

void* tls_thread(void* arg) {
    // Method 1: __thread is zero-initialized per thread
    tls_counter++;
    printf("[TLS-__thread] Value: %d\n", tls_counter);

    // Method 2: pthread key has destructor for cleanup
    int* data = malloc(sizeof(int));
    *data = 42;
    pthread_setspecific(key, data);

    int* retrieved = pthread_getspecific(key);
    printf("[TLS-key] Value: %d\n", *retrieved);

    return NULL;
}

int main() {
    pthread_key_create(&key, destructor);
    pthread_t t1, t2;

    pthread_create(&t1, NULL, tls_thread, NULL);
    pthread_create(&t2, NULL, tls_thread, NULL);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    pthread_key_delete(key);
    return 0;
}

Thread-Local Storage Layout

The ELF binary has .tdata (initialized) and .tbss (zeroed) sections. Each thread gets a private copy; the thread-local storage block sits near the thread's stack and is addressed through the FS segment register on x86-64.

Linux Thread Implementation (NPTL)

On Linux, threads are not a separate kernel abstraction. They are processes that share more resources than usual. The clone() syscall with different flag combinations creates both processes and threads. 

#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <sched.h>
#include <unistd.h>
#include <sys/mman.h>
#include <string.h>

int thread_func(void* arg) {
    printf("[CLONE-CHILD] PID: %d, TID: %d\n", getpid(), gettid());
    printf("[CLONE-CHILD] Argument: %d\n", *(int*)arg);
    return 0;
}

int main() {
    // Allocate stack for the clone (stacks grow down on x86)
    const size_t STACK_SIZE = 1024 * 1024;
    void* stack = mmap(NULL, STACK_SIZE, PROT_READ | PROT_WRITE,
                       MAP_PRIVATE | MAP_ANONYMOUS | MAP_STACK, -1, 0);
    void* stack_top = (char*)stack + STACK_SIZE;

    int arg = 123;
    int flags = CLONE_VM        // Share address space
              | CLONE_FS        // Share filesystem info (cwd, root, umask)
              | CLONE_FILES     // Share file descriptor table
              | CLONE_SIGHAND   // Share signal handlers
              | CLONE_THREAD    // Place in same thread group (has TID)
              | CLONE_SYSVSEM   // Share SysV semaphore undo lists
              | CLONE_PARENT_SETTID
              | CLONE_CHILD_CLEARTID;

    pid_t tid = clone(thread_func, stack_top, flags, &arg);
    if (tid == -1) {
        perror("clone");
        return 1;
    }

    printf("[PARENT] Child thread ID: %d\n", tid);
    sleep(1);  // Wait for child to print (no pthread_join equivalent)
    return 0;
}

Kernel Threads

Kernel threads run entirely in kernel space, have no user-space memory, and exist only to perform kernel work. They appear in ps with [] brackets around their name. 

ps aux | grep '\[.*\]'
# Output:
# [kthreadd]        ← Master kernel thread spawner
# [ksoftirqd/0]     ← Softirq daemon per CPU
# [kworker/0:0]     ← Worker thread for deferred work
# [migration/0]     ← CPU task migration
# [rcu_sched]       ← RCU (Read-Copy-Update) grace period management
# [kcompactd0]      ← Memory compaction
# [kswapd0]         ← Swap daemon

Kernel threads are created by the kernel itself ( kthread_create() ) and all descend from kthreadd (PID 2). They appear in /proc like any other process, but their memory descriptor is NULL, so they never enter user space.

Inspecting Threads on a Running System

# Number of threads per process
ls -d /proc/[0-9]*/task/*/ 2>/dev/null | awk -F/ '{print $3}' | sort | uniq -c | sort -rn | head -10

# Or with ps
ps -eo pid,nlwp,comm | grep -v '^ *PID' | sort -k2 -rn | head -10

# Thread-level CPU usage
ps -eLf  # L flag: show threads separately

# Threads of a specific process
ls /proc/$$/task/
cat /proc/$$/task/$(pidof bash | cut -d' ' -f1)/status | grep -E "Tgid|Pid|PPid"

# top -H: show threads individually
top -H -p $$

# strace -f on a threaded program shows all thread syscalls
strace -f -p $(pidof multi-threaded-app)

# GDB: inspect threads
gdb -p $(pidof app)
(gdb) info threads
(gdb) thread 2
(gdb) bt

Verification Checklist

# 1. Count threads in current shell session
cat /proc/$$/status | grep Threads

# 2. Compare PID vs TID for a multi-threaded process
ps -eLf | head -20

# 3. Verify clone flags used by pthread_create
strace -e clone gcc -pthread -o /tmp/t test.c 2>&1 | grep clone || echo "N/A"

gcc -pthread -o /tmp/td thread-demo.c

cat /proc/$$/task/*/stat | awk '{print $1, $29}' | head -5
# Field 29 (started with kernel 4.?) is startstack

# 5. List kernel threads
ps -eo pid,comm | grep '\[.*\]'

# 6. Thread limit (kernel threads-max)

Challenge

Port the mutex counter to Go and prove the race with the race detector. The program below spawns 1000 goroutines that increment a shared counter without synchronization. Run it with go run -race main.go and observe the lost updates and the reported data race. 

package main

import (
	"fmt"
	"sync"
)

func main() {
	counter := 0

	var wg sync.WaitGroup
	for i := 0; i < 1000; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			counter++
		}()
	}
	wg.Wait()

	fmt.Println("counter:", counter)
}

Then fix it with a sync.Mutex and confirm the counter reaches 1000. As an extra step, replace the mutex with sync/atomic and compare the throughput of both versions with go test -bench .
